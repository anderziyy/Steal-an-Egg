local function make(className, properties, parent)
	local object = Instance.new(className)
	if className == "TextLabel" or className == "TextButton" or className == "TextBox" then
		pcall(function()
			object.AutoLocalize = false
		end)
	end
	for key, value in pairs(properties or {}) do
		object[key] = value
	end
	object.Parent = parent
	return object
end

local function applyEnglishOnly(root)
	local function visit(obj)
		pcall(function()
			if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
				obj.AutoLocalize = false
			end
		end)
		for _, child in ipairs(obj:GetChildren()) do
			visit(child)
		end
	end
	visit(root)
end

local function tween(object, time, properties, style, direction)
	local info = TweenInfo.new(time or 0.18, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out)
	local animation = TweenService:Create(object, info, properties)
	animation:Play()
	return animation
end

local cleanupCallbacks = {}
local cleanedUp = false
local pendingAction = nil

local function registerCleanup(callback)
	if type(callback) ~= "function" then
		return
	end
	table.insert(cleanupCallbacks, callback)
end

local function cleanupAll()
	if cleanedUp then
		return
	end
	cleanedUp = true

	for i = #cleanupCallbacks, 1, -1 do
		local callback = cleanupCallbacks[i]
		cleanupCallbacks[i] = nil
		pcall(callback)
	end
end

local function getViewport()
	local camera = workspace.CurrentCamera
	if not camera then
		return Vector2.new(1280, 720)
	end
	return camera.ViewportSize
end

local function updateGlobalScale()
	local viewport = getViewport()
	local reference = math.min(viewport.X, viewport.Y)
	local target = reference / 720
	RootScale.Scale = math.clamp(target, 0.82, 1.22)
end

updateGlobalScale()
if workspace.CurrentCamera then
	local viewportSizeConnection = workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(updateGlobalScale)
	registerCleanup(function()
		viewportSizeConnection:Disconnect()
	end)
end

local function createLockChoice(name, text, x)
	local Button = make("TextButton", {
		Name = name,
		Position = UDim2.fromScale(x, 0.48),
		Size = UDim2.fromScale(0.38, 0.18),
		BackgroundColor3 = COLORS.Black,
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Text = "",
		Active = true,
		ZIndex = 1003,
	}, LockPanel)
	make("UIStroke", {Thickness = 1.2, Color = COLORS.Gray3, Transparency = 0.05}, Button)

	local Label = make("TextLabel", {
		Position = UDim2.fromScale(0.08, 0),
		Size = UDim2.fromScale(0.76, 1),
		BackgroundTransparency = 1,
		Text = text,
		TextColor3 = COLORS.White,
		Font = Enum.Font.GothamSemibold,
		TextSize = 12,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextYAlignment = Enum.TextYAlignment.Center,
		ZIndex = 1004,
	}, Button)

	local Check = make("TextLabel", {
		AnchorPoint = Vector2.new(1, 0.5),
		Position = UDim2.fromScale(0.93, 0.5),
		Size = UDim2.fromScale(0.18, 0.70),
		BackgroundTransparency = 1,
		Text = "",
		TextColor3 = COLORS.White,
		Font = Enum.Font.GothamBold,
		TextSize = 14,
		TextXAlignment = Enum.TextXAlignment.Center,
		TextYAlignment = Enum.TextYAlignment.Center,
		ZIndex = 1004,
	}, Button)

	local state = false
	local function setState(value)
		state = value == true
		Button.BackgroundColor3 = state and COLORS.Gray2 or COLORS.Black
		Check.Text = state and "✓" or ""
	end

	Button.Activated:Connect(function()
		setState(not state)
		if name == "MenuChoice" then
			lockMenuSelected = state
		else
			lockButtonSelected = state
		end
	end)
	setState(false)
	return Button
end

local resultsScrollTouchStart = nil
local resultsScrollTouchCanvasY = nil

local function isResultsScrollingAllowed()
	return currentTab ~= "Credits" and ResultsScroll.Visible and ResultsScroll.ScrollingEnabled
end

local function getResultsMaxScrollY()
	return math.max(0, ResultsScroll.AbsoluteCanvasSize.Y - ResultsScroll.AbsoluteSize.Y)
end

local function setResultsScrollY(y)
	local maxY = getResultsMaxScrollY()
	ResultsScroll.CanvasPosition = Vector2.new(0, math.clamp(y or 0, 0, maxY))
end

local function isPointInsideResults(point)
	if not ResultsScroll.Visible then
		return false
	end
	local position = ResultsScroll.AbsolutePosition
	local size = ResultsScroll.AbsoluteSize
	return point.X >= position.X
		and point.X <= position.X + size.X
		and point.Y >= position.Y
		and point.Y <= position.Y + size.Y
end

local function updateResultsScrollMode()
	local enabled = currentTab ~= "Credits"
	ResultsScroll.ScrollingEnabled = enabled
	ResultsScroll.Active = enabled
	ResultsScroll.ScrollBarThickness = enabled and 3 or 0
	ResultsScroll.CanvasPosition = Vector2.new(0, 0)
end

resultsScrollWheelConnection = ResultsScroll.InputChanged:Connect(function(input)
	if not isResultsScrollingAllowed() then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseWheel then
		local delta = input.Position.Z
		if delta ~= 0 then
			setResultsScrollY(ResultsScroll.CanvasPosition.Y - (delta * ResultsScroll.AbsoluteSize.Y * 0.16))
		end
	elseif input.UserInputType == Enum.UserInputType.Touch then
		resultsScrollTouchStart = input.Position
		resultsScrollTouchCanvasY = ResultsScroll.CanvasPosition.Y
	end
end)

resultsScrollInputConnection = UserInputService.InputChanged:Connect(function(input)
	if not isResultsScrollingAllowed() then
		return
	end

	if input.UserInputType == Enum.UserInputType.Touch and resultsScrollTouchStart and resultsScrollTouchCanvasY then
		local deltaY = input.Position.Y - resultsScrollTouchStart.Y
		if math.abs(deltaY) >= 1 then
			setResultsScrollY(resultsScrollTouchCanvasY - deltaY)
		end
	elseif input.UserInputType == Enum.UserInputType.MouseWheel and isPointInsideResults(input.Position) then
		local delta = input.Position.Z
		if delta ~= 0 then
			setResultsScrollY(ResultsScroll.CanvasPosition.Y - (delta * ResultsScroll.AbsoluteSize.Y * 0.16))
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch then
		resultsScrollTouchStart = nil
		resultsScrollTouchCanvasY = nil
	end
end)

local function getVisibleItems(tabName, query)
	local normalized = string.lower(query or "")

	if normalized == "" then
		return datasets[tabName] or {}
	end

	-- Search globally across every tab, but only return real options.
	-- Category headers are never search results.
	local filtered = {}
	local searchOrder = {"Main", "Auto Buy", "Farm", "Config", "Credits"}
	for _, datasetName in ipairs(searchOrder) do
		for _, item in ipairs(datasets[datasetName] or {}) do
			if item.Type == "Category" then
				for _, child in ipairs(item.Items or {}) do
					local title = string.lower(child.Title or "")
					local description = string.lower(child.Description or "")
					if string.find(title, normalized, 1, true) or string.find(description, normalized, 1, true) then
						table.insert(filtered, child)
					end
				end
			else
				local title = string.lower(item.Title or "")
				local description = string.lower(item.Description or "")
				if string.find(title, normalized, 1, true) or string.find(description, normalized, 1, true) then
					table.insert(filtered, item)
				end
			end
		end
	end

	return filtered
end

local function safeRequire(moduleScript)
	if moduleScript == nil then
		return nil
	end
	local ok, result = pcall(require, moduleScript)
	if ok then
		return result
	end
	return nil
end

local function findEggTools(eggToolDisplay)
	local tools = {}
	local seen = {}

	local function scan(container)
		if not container then
			return
		end
		for _, child in ipairs(container:GetChildren()) do
			if child:IsA("Tool") and not seen[child] then
				local ok, isEgg = pcall(function()
					return eggToolDisplay and eggToolDisplay.IsEggTool and eggToolDisplay.IsEggTool(child)
				end)
				if ok and isEgg then
					seen[child] = true
					table.insert(tools, child)
				end
			end
		end
	end

	scan(LocalPlayer.Character)
	scan(LocalPlayer:FindFirstChildOfClass("Backpack"))
	return tools
end

local function getOwnPlotPlacement()
	local shared = game:GetService("ReplicatedStorage")
	local plotState = safeRequire(shared.Client:FindFirstChild("PlotState"))
	local eggRaycastModule = safeRequire(shared.Shared.Eggs:FindFirstChild("EggPlacementRaycast"))
	if not plotState or not eggRaycastModule or not eggRaycastModule.Raycast then
		return nil
	end

	local plot = plotState.ResolvePlot and plotState.ResolvePlot() or nil
	if not plot or not plot.PetArea or not plot.CenterPoint then
		return nil
	end

	local ok, raycastResult = pcall(function()
		return eggRaycastModule.Raycast(plot.PetArea, LocalPlayer.UserId)
	end)
	if not ok or raycastResult == nil or raycastResult.Instance ~= plot.PetArea then
		return nil
	end

	local relativeCFrame = plot.CenterPoint.CFrame:ToObjectSpace(CFrame.new(raycastResult.Position))
	return relativeCFrame
end

local function placeOneEgg()
	local shared = game:GetService("ReplicatedStorage")
	local eggToolDisplay = safeRequire(shared.Shared.Eggs:FindFirstChild("EggToolDisplay"))
	local eggState = safeRequire(shared.Client:FindFirstChild("EggState"))
	if not eggToolDisplay or not eggState or not eggState.PlantEgg then
		return false
	end

	local relativeCFrame = getOwnPlotPlacement()
	if not relativeCFrame then
		return false
	end

	local tools = findEggTools(eggToolDisplay)
	for _, tool in ipairs(tools) do
		local okUid, uid = pcall(function()
			return eggToolDisplay.GetToolUid and eggToolDisplay.GetToolUid(tool)
		end)
		if okUid and uid ~= nil then
			local okPlace, placed = pcall(function()
				return eggState.PlantEgg(uid, relativeCFrame)
			end)
			if okPlace and placed == true then
				local renderer = safeRequire(shared.Shared.Eggs:FindFirstChild("PlacedEggRenderer"))
				if renderer and renderer.Refresh then
					pcall(renderer.Refresh)
				end
				return true
			end
			-- A failed PlantEgg call means the current plot/egg state is not valid anymore.
			return false
		end
	end

	return false
end

local function setPlaceEggs(item, enabled)
	disconnectToggle(item)
	if not enabled then
		return
	end

	local running = true
	toggleConnections[item] = {

				task.wait(0.35)
			else
				task.wait(0.50)
			end
		end
	end)
end

local function setAntiAFK(item, enabled)
	if toggleConnections[item] then
		disconnectToggle(item)
	end

	if not enabled then
		return
	end

	local running = true
	toggleConnections[item] = {
		Disconnect = function()
			running = false
		end,
	}

	task.spawn(function()
		while running and not cleanedUp do
			task.wait(60)
			if not running or cleanedUp then
				break
			end

			local humanoid = getHumanoid()
			if humanoid and humanoid.Health > 0 then
				humanoid:Move(Vector3.new(1, 0, 0), true)
				task.wait(0.18)
				if running and not cleanedUp and humanoid.Parent and humanoid.Health > 0 then
					humanoid:Move(Vector3.zero, true)
				end
			end
		end
	end)
end

	end
	for _, child in ipairs(character:GetChildren()) do
		if child:IsA("Tool") and child:GetAttribute("ItemType") == "AssetEgg" then
			local uid = child:GetAttribute("UID")
			if typeof(uid) == "string" and uid ~= "" then
				return uid
			end
		end
	end
	return nil
end

local function eggUIDExistsInCharacterOrBackpack(uid)
	if typeof(uid) ~= "string" or uid == "" then
		return false
	end

	local function scan(container)
		if not container then
			return false
		end
		for _, child in ipairs(container:GetChildren()) do
			if child:IsA("Tool") and child:GetAttribute("ItemType") == "AssetEgg" and child:GetAttribute("UID") == uid then
				return true
			end
		end
		return false
	end

	return scan(LocalPlayer.Character) or scan(LocalPlayer:FindFirstChildOfClass("Backpack"))
end

local function getAvailableEggUID()
	local save = getSaveModule()
	if save and save.Get then
		local ok, data = pcall(save.Get)
		if ok and typeof(data) == "table" and typeof(data.EggInventory) == "table" then
			for uid, record in pairs(data.EggInventory) do
				if typeof(uid) == "string" and typeof(record) == "table" and record.Placement == nil then
					if eggUIDExistsInCharacterOrBackpack(uid) then
						return uid
					end
				end
			end
			for uid, record in pairs(data.EggInventory) do
				if typeof(uid) == "string" and typeof(record) == "table" and record.Placement == nil then
					return uid
				end
			end
		end
	end

	local eggToolDisplay = nil
	local shared = game:GetService("ReplicatedStorage")
	if shared.Shared and shared.Shared:FindFirstChild("Eggs") then
		eggToolDisplay = safeRequire(shared.Shared.Eggs:FindFirstChild("EggToolDisplay"))
	end
	for _, tool in ipairs(findEggTools(eggToolDisplay)) do
		local ok, uid = pcall(function()
			return eggToolDisplay and eggToolDisplay.GetToolUid and eggToolDisplay.GetToolUid(tool)
		end)
		if ok and typeof(uid) == "string" and uid ~= "" then
			return uid
		end
	end

	return nil
end

local function equipOneAvailableEgg(preferredUID)
	local equipped = getEquippedEggUID()
	if equipped then
		return equipped
	end

	local eggState = getEggStateModule()
	if not eggState or not eggState.WearEggTool then
		return nil
	end

	local uid = preferredUID
	if typeof(uid) ~= "string" or uid == "" then
		uid = getAvailableEggUID()
	end
	if typeof(uid) ~= "string" or uid == "" then
		return nil
	end

	local ok, equippedOk = pcall(function()
		return eggState.WearEggTool(uid)
	end)
	if ok and equippedOk then
		return uid
	end
	return nil
end

local function getOpenEggDependencies()
	local shared = game:GetService("ReplicatedStorage")

	local renderer = nil

		renderer = safeRequire(shared.Shared.Eggs:FindFirstChild("PlacedEggRenderer"))
	end
	return eggState, eggRecords, renderer

local function openReadyEggs()
	local eggState, eggRecords, renderer = getOpenEggDependencies()
	if not eggState or not eggState.ReadOwnerEggs or not eggRecords
		or not eggRecords.GrowthAlpha or not eggRecords.CurrentNightCredit
		or not renderer or not renderer.ActivateLocalEgg then
		return false
	end

	local ok, ownerEggs = pcall(function()
		return eggState.ReadOwnerEggs(LocalPlayer.UserId)
	end)
	if not ok or typeof(ownerEggs) ~= "table" then
		return false
	end

	local serverTime = workspace:GetServerTimeNow()
	local opened = false

	for uid, record in pairs(ownerEggs) do
		if typeof(uid) == "string" and typeof(record) == "table" and record.Placement then
			local ready = pcall(function()
				return eggRecords.GrowthAlpha(
					record,
					serverTime,
					record.GrowthSpeedMultiplier,
					eggRecords.CurrentNightCredit(record, serverTime, record.GrowthSpeedMultiplier),
					LocalPlayer
				) >= 1
			end)

			if ready then
				local activated, success = pcall(function()
					return renderer.ActivateLocalEgg(uid)
				end)
				if activated and success == true then
					opened = true
					task.wait(0.15)
				end
			end
		end
	end

	return opened
end

renderResults = function()
	local previousScrollY = ResultsScroll.CanvasPosition.Y
	clearResults()
	local query = SearchBox.Text
	local items = getVisibleItems(currentTab, query)
	if #items == 0 then
		if currentTab == "Main" and query == "" then
			setResultsScrollPositionY(0)
			return
		elseif query ~= "" then
			showEmptyState("No matching options found.")
		elseif currentTab == "Auto Buy" then
			setResultsScrollPositionY(0)
			return
		elseif currentTab == "Config" then
			showEmptyState("No options are available in Config.")
		elseif currentTab == "Credits" then
			showEmptyState("No options are available in Credits.")
		else
			showEmptyState("This tab has no options yet.")
		end
		setResultsScrollPositionY(0)
		return
	end

	local layoutIndex = 0
	if query ~= "" then
		-- Global search results are option-only; categories are never rendered.
		for _, item in ipairs(items) do
			layoutIndex += 1
			createStandardRow(item, layoutIndex)
		end
		task.defer(function()
			setResultsScrollPositionY(previousScrollY)
		end)
		return
	end

	for _, item in ipairs(items) do
		layoutIndex += 1
		if item.Type == "Category" then
			createCategoryRow(item, layoutIndex, query)
			if item.Expanded then
				layoutIndex += #(item.Items or {})
			end
		else
			createStandardRow(item, layoutIndex)
		end
	end

	task.defer(function()
		setResultsScrollPositionY(previousScrollY)
		applyEnglishOnly(ResultsScroll)
	end)
	applyEnglishOnly(ResultsScroll)
	-- Garantir que o canvas size seja recalculado corretamente para melhor scrolling
	task.wait()
	local listLayout = ResultsScroll:FindFirstChildOfClass("UIListLayout")
	if listLayout then
		local canvasYSize = listLayout.AbsoluteContentSize.Y
		if canvasYSize > 0 then
			ResultsScroll.CanvasSize = UDim2.fromScale(1, math.max(1, (canvasYSize + 10) / ResultsScroll.AbsoluteSize.Y))
		end
	end
end

local function setTab(name)
	currentTab = name
	updateResultsScrollMode()
	for tabName, button in pairs(tabButtons) do
		local selected = tabName == name
		tween(button, 0.14, {
			BackgroundColor3 = selected and COLORS.Gray2 or COLORS.Black2,
			BackgroundTransparency = selected and 0 or 0.15,
		})
		local icon = button:FindFirstChild("Icon")
		if icon and icon:IsA("ImageLabel") then
			tween(icon, 0.14, {ImageColor3 = selected and COLORS.White or COLORS.SoftWhite})
		end
		local label = button:FindFirstChild("Label")
		if label and label:IsA("TextLabel") then
			tween(label, 0.14, {TextColor3 = selected and COLORS.White or COLORS.SoftWhite})
		end
	end
	SearchBox.Text = ""
	SearchHint.Visible = true
	SearchClear.Visible = false
	renderResults()
end

	local position = guiObject.AbsolutePosition
	local size = guiObject.AbsoluteSize

		and point.Y >= position.Y and point.Y <= position.Y + size.Y
end

	if not ScreenGui.Parent then
		return
	end
	if Menu.Visible then
		hideMenu()
	else
		showMenu()
	end
end

local launcherDragging = false
local launcherDragInput = nil
local launcherDragStart = nil
local launcherStartPos = nil
local launcherDidDrag = false

local function clampLauncher(position)
	local viewport = getViewport()
	local x = math.clamp(position.X.Scale, 0.0, 0.925)
	local y = math.clamp(position.Y.Scale, 0.02, 0.92)

	if position.X.Scale < 0 then
		x = 0.015
	elseif position.X.Scale > 0.925 then
		x = 0.907
	end

	return UDim2.fromScale(x, y)
end

local function updateLauncherDrag(input)
	if lockEnabled and lockButtonSelected then
		return
	end
	local delta = input.Position - launcherDragStart
	if delta.Magnitude > 4 then
		launcherDidDrag = true
	end
	local viewport = getViewport()
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end
	local dx = delta.X / viewport.X
	local dy = delta.Y / viewport.Y
	local target = UDim2.fromScale(
		launcherStartPos.X.Scale + dx,
		launcherStartPos.Y.Scale + dy
	)
	LauncherFrame.Position = clampLauncher(target)
end

local function beginLauncherDrag(input)
	if lockEnabled and lockButtonSelected then
		return
	end
	launcherDragging = true
	launcherDidDrag = false
	launcherDragStart = input.Position
	launcherStartPos = LauncherFrame.Position
	launcherDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
end

local function endLauncherDrag()
	launcherDragging = false
	launcherDragInput = nil
	launcherDragStart = nil
	launcherStartPos = nil
end

Launcher.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		beginLauncherDrag(input)
	end
end)

local menuDragging = false
local menuDragInput = nil
local menuDragStart = nil
local menuStartPos = nil
local dragCorner = ""
local pointerDownAt = 0
local didDrag = false

local CornerSize = UDim2.fromScale(0.075, 0.11)
local corners = {}

local function makeCorner(name, position, anchor)
	local corner = make("TextButton", {
		Name = name,
		AnchorPoint = anchor,
		Position = position,
		Size = CornerSize,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Text = "",
		AutoButtonColor = false,
		Active = true,
		ZIndex = 10,
	}, Menu)
	return corner
end

corners.TopLeft = makeCorner("DragTopLeft", UDim2.fromScale(0, 0), Vector2.new(0, 0))
corners.TopRight = makeCorner("DragTopRight", UDim2.fromScale(1, 0), Vector2.new(1, 0))
corners.BottomLeft = makeCorner("DragBottomLeft", UDim2.fromScale(0, 1), Vector2.new(0, 1))
corners.BottomRight = makeCorner("DragBottomRight", UDim2.fromScale(1, 1), Vector2.new(1, 1))

for name, corner in pairs(corners) do
	corner.MouseEnter:Connect(function()
		tween(MenuStroke, 0.10, {Color = COLORS.White, Transparency = 0.05})
	end)
	corner.MouseLeave:Connect(function()
		if not menuDragging then
			tween(MenuStroke, 0.10, {Color = COLORS.Gray3, Transparency = 0.12})
		end
	end)
	corner.InputBegan:Connect(function(input)
		if lockEnabled and lockMenuSelected then
			return
		end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			pointerDownAt = os.clock()
			didDrag = false
			menuDragging = false
			menuDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
			menuDragStart = input.Position
			menuStartPos = Menu.Position
			dragCorner = name
		end
	end)
end

function updateMenuDrag(input)
	if lockEnabled and lockMenuSelected then
		return
	end
	if not menuDragStart or not menuStartPos then
		return
	end
	local delta = input.Position - menuDragStart
	if delta.Magnitude < 4 then
		return
	end
	if os.clock() - pointerDownAt < 0.06 then
		return
	end
	didDrag = true
	menuDragging = true

	local viewport = getViewport()
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end
	local dx = delta.X / viewport.X
	local dy = delta.Y / viewport.Y
	local x = menuStartPos.X.Scale + dx
	local y = menuStartPos.Y.Scale + dy
	Menu.Position = UDim2.fromScale(x, y)
end

function endMenuDrag()
	if not menuDragging then
		menuDragInput = nil
		return
	end

	local viewport = getViewport()
	local absolutePosition = Menu.AbsolutePosition
	local absoluteSize = Menu.AbsoluteSize
	local fullyLeft = absolutePosition.X + absoluteSize.X < 0
	local fullyRight = absolutePosition.X > viewport.X
	local fullyTop = absolutePosition.Y + absoluteSize.Y < 0
	local fullyBottom = absolutePosition.Y > viewport.Y

	if fullyLeft then
		local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
		tween(Menu, 0.20, {Position = UDim2.fromScale(0.02, y)})
	elseif fullyRight then
		local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
		tween(Menu, 0.20, {Position = UDim2.fromScale(0.98, y)})
	elseif fullyTop then
		tween(Menu, 0.20, {Position = UDim2.fromScale(math.clamp(Menu.Position.X.Scale, 0.02, 0.98), 0.05)})
	elseif fullyBottom then
		tween(Menu, 0.20, {Position = UDim2.fromScale(math.clamp(Menu.Position.X.Scale, 0.02, 0.98), 0.95)})
	end

	menuDragging = false
	menuDragInput = nil
	menuDragStart = nil
	menuStartPos = nil
	tween(MenuStroke, 0.12, {Color = COLORS.Gray3, Transparency = 0.12})
end

for name, corner in pairs(corners) do
	corner.Activated:Connect(function()
		if didDrag then
			didDrag = false
			return
		end
	end)
end

local inputChangedConnection = UserInputService.InputChanged:Connect(function(input)
	if launcherDragging and ((launcherDragInput and input == launcherDragInput) or (not launcherDragInput and input.UserInputType == Enum.UserInputType.MouseMovement)) then
		updateLauncherDrag(input)
	end
	if menuDragInput and input == menuDragInput then
		updateMenuDrag(input)
	elseif menuDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	elseif menuDragStart and not menuDragInput and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	end
end)

local inputEndedConnection = UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		if launcherDragging then
			endLauncherDrag()
		end
		if menuDragging then
			endMenuDrag()
		end
	end
end)

Launcher.Activated:Connect(function()
	if launcherDidDrag then
		launcherDidDrag = false
		return
	end
	toggleMenu()
end)
