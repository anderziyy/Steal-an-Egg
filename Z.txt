local function new(className, properties, parent)
    local obj = Instance.new(className)
    for key, value in pairs(properties or {}) do
        obj[key] = value
    end
    if parent then
        obj.Parent = parent
    end
    return obj
end

local function corner(parent, radiusScale)
    return new("UICorner", {CornerRadius = UDim.new(radiusScale or 0.08, 0)}, parent)
end

local function stroke(parent, color, transparency, thickness)
    return new("UIStroke", {
        Color = color or WHITE,
        Transparency = transparency or 0,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    }, parent)
end

local function tween(obj, info, props)
    local t = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

local function clampScale(value, minValue, maxValue)
    return math.clamp(value, minValue, maxValue)
end

local function viewportScale()
    local camera = Workspace.CurrentCamera
    local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
    local base = math.min(viewport.X / 1280, viewport.Y / 720)
    return clampScale(base, 0.72, 1.25)
end

local function fitGuiScale(guiObject)
    local uiScale = guiObject:FindFirstChildOfClass("UIScale")
    if not uiScale then
        uiScale = new("UIScale", {}, guiObject)
    end
    uiScale.Scale = viewportScale()
    return uiScale
end

local function nearestEdgeFallback()
    local pos = TopButtonHolder.Position
    local x, y = pos.X.Scale, pos.Y.Scale
    if x < 0 then x = 0 end
    if x > 0.925 then x = 0.925 end
    if y < 0 then y = 0 end
    if y > 0.925 then y = 0.925 end
    setOpenButtonPosition(x, y)
end

local draggingButton = false
local buttonDragStart
local buttonStartPos
local buttonDragInput

local function beginButtonDrag(input)
    draggingButton = true
    buttonDragStart = input.Position
    buttonStartPos = TopButtonHolder.Position
    buttonDragInput = input
end

local function updateButtonDrag(input)
    local camera = Workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    if viewport.X <= 0 or viewport.Y <= 0 then return end
    local delta = input.Position - buttonDragStart
    local nx = buttonStartPos.X.Scale + delta.X / viewport.X
    local ny = buttonStartPos.Y.Scale + delta.Y / viewport.Y
    setOpenButtonPosition(nx, ny)
end

local function endButtonDrag()
    if not draggingButton then return end
    draggingButton = false
    buttonDragInput = nil
    local camera = Workspace.CurrentCamera
    local viewport = camera and camera.ViewportSize or Vector2.new(1280, 720)
    local edge = 0.05
    local x = TopButtonHolder.Position.X.Scale
    local y = TopButtonHolder.Position.Y.Scale
    if x < edge then x = 0 end
    if x > 0.875 then x = 0.925 end
    if y < edge then y = 0.02 end
    if y > 0.875 then y = 0.925 end
    x = clampScale(x, 0, 0.925)
    y = clampScale(y, 0, 0.925)
    local visibleX = x
    local visibleY = y
    if visibleX < -0.001 or visibleY < -0.001 or visibleX > 0.926 or visibleY > 0.926 then
        visibleX = math.clamp(visibleX, 0, 0.925)
        visibleY = math.clamp(visibleY, 0, 0.925)
    end
    setOpenButtonPosition(visibleX, visibleY)
end

    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

    end

local function safeClipboard(text)
    local copied = false
    if type(setclipboard) == "function" then
        local ok = pcall(setclipboard, text)
        copied = ok
    elseif type(toclipboard) == "function" then
        local ok = pcall(toclipboard, text)
        copied = ok
    elseif type(Clipboard) == "table" and type(Clipboard.set) == "function" then
        local ok = pcall(Clipboard.set, text)
        copied = ok
    end
    return copied
end

local function randomServerHop()
    local moved = false
    local ok = pcall(function()
        local placeId = game.PlaceId
        local jobId = game.JobId
        if type(game.HttpGet) == "function" then
            local raw = game:HttpGet("https://games.roblox.com/v1/games/" .. tostring(placeId) .. "/servers/Public?sortOrder=Asc&limit=100")
            if raw and #raw > 0 then
                local data = game:GetService("HttpService"):JSONDecode(raw)
                local candidates = {}
                for _, server in ipairs(data.data or {}) do
                    if server.id and server.id ~= jobId and tonumber(server.playing or 0) < tonumber(server.maxPlayers or 0) then
                        table.insert(candidates, server.id)
                    end
                end
                if #candidates > 0 then
                    local selected = candidates[math.random(1, #candidates)]
                    TeleportService:TeleportToPlaceInstance(placeId, selected, Player)
                    moved = true
                    return
                end
            end
        end
        TeleportService:Teleport(placeId, Player)
        moved = true
    end)
    return ok and moved
end

        moved = true
    end)
    return ok and moved
end

local antiAfkEnabled = false
local antiAfkConnection
local lowModeEnabled = false
local lowModeSaved = {}

local function performAntiAfkPulse()
    local character = Player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    local state = humanoid:GetState()
    if state == Enum.HumanoidStateType.Dead then return end
    local oldAutoRotate = humanoid.AutoRotate
    humanoid.AutoRotate = false
    humanoid:Move(Vector3.new(0, 0, 0), true)
    task.wait(0.08)
    humanoid:Move(Vector3.new(0.1, 0, 0), true)
    task.wait(0.12)
    humanoid:Move(Vector3.new(-0.1, 0, 0), true)
    task.wait(0.08)
    humanoid:Move(Vector3.zero, true)
    humanoid.AutoRotate = oldAutoRotate
end

local function setAntiAfk(enabled)
    antiAfkEnabled = enabled
    if antiAfkConnection then
        antiAfkConnection:Disconnect()
        antiAfkConnection = nil
    end
    if enabled then
        antiAfkConnection = task.spawn(function()
            while antiAfkEnabled do
                task.wait(60)
                if antiAfkEnabled then
                    pcall(performAntiAfkPulse)
                end
            end
        end)
    end
end

local function saveVisual(obj, property)
    lowModeSaved[obj] = lowModeSaved[obj] or {}
    if lowModeSaved[obj][property] == nil then
        local ok, value = pcall(function() return obj[property] end)
        if ok then
            lowModeSaved[obj][property] = value
        end
    end
end

local function applyLowMode()
    table.clear(lowModeSaved)
    local lightingProps = {"GlobalShadows", "FogEnd", "Brightness", "EnvironmentDiffuseScale", "EnvironmentSpecularScale"}
    for _, prop in ipairs(lightingProps) do
        saveVisual(Lighting, prop)
    end
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 1000000
    Lighting.EnvironmentDiffuseScale = 0
    Lighting.EnvironmentSpecularScale = 0
    Lighting.Brightness = math.min(Lighting.Brightness, 1)

    for _, obj in ipairs(game:GetDescendants()) do
        if obj:IsA("BasePart") then
            saveVisual(obj, "Material")
            saveVisual(obj, "CastShadow")
            obj.Material = Enum.Material.SmoothPlastic
            obj.CastShadow = false
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
            saveVisual(obj, "Enabled")
            obj.Enabled = false
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            saveVisual(obj, "Transparency")
            obj.Transparency = 1
        elseif obj:IsA("SurfaceAppearance") then
            saveVisual(obj, "Enabled")
            obj.Enabled = false
        elseif obj:IsA("PostEffect") then
            saveVisual(obj, "Enabled")
            obj.Enabled = false
        end
    end
    lowModeEnabled = true
end

local function restoreLowMode()
    for obj, props in pairs(lowModeSaved) do
        if obj and obj.Parent then
            for property, value in pairs(props) do
                pcall(function()
                    obj[property] = value
                end)
            end
        end
    end
    table.clear(lowModeSaved)
    lowModeEnabled = false
end

local DISCORD_SERVER_LINK = "PASTE_YOUR_DISCORD_INVITE_LINK_HERE"

local function executeOption(info)
    if info.action == "copy_discord" then
        local discordLink = DISCORD_SERVER_LINK
        local copied = safeClipboard(discordLink)
        if copied then
            showNotification(132744109593099, "Discord server link copied.")
        else
            showNotification(132744109593099, "Copy the Discord link from the script configuration.")
        end
    elseif info.action == "copy_server" then
        local copied = safeClipboard(getCurrentServerCode())
        if copied then
            showNotification(88704800286791, "Current server ID copied.")
        else
            showNotification(88704800286791, "Clipboard access is unavailable in this environment.")
        end
    elseif info.action == "copy_game" then
        local copied = safeClipboard(getGameId())
        if copied then
            showNotification(88704800286791, "Game ID copied.")
        else
            showNotification(88704800286791, "Clipboard access is unavailable in this environment.")
        end
    elseif info.action == "auto_rejoin" then
        showConfirm("Are you sure you want to join a different server?", function()
            randomServerHop()
        end)
    elseif info.action == "rejoin_current" then
        showConfirm("Are you sure you want to reconnect to this server?", function()
            rejoinCurrentServer()
        end)
    elseif info.action == "anti_afk" then
        info.enabled = not info.enabled
        setAntiAfk(info.enabled)
    elseif info.action == "low_mode" then
        info.enabled = not info.enabled
        if info.enabled then
            applyLowMode()
        else
            restoreLowMode()
        end
    end
end

local function getAllSearchableOptions()
    local results = {}
    if #optionEntries > 0 then
        for _, entry in ipairs(optionEntries) do
            table.insert(results, entry)
        end
    end
    local extra = {}
    local oldSelected = selectedTab
    for _, tabData in ipairs(tabsData) do
        local tabName = tabData.name
        if tabName ~= oldSelected then
            if tabName == "Config" then
                table.insert(extra, {
                    data = configCategory.options[1], tab = "Config", category = "Player"
                })
                for i = 2, #configCategory.options do
                    table.insert(extra, {data = configCategory.options[i], tab = "Config", category = "Player"})
                end
            elseif tabName == "Credits" then
                for i, data in ipairs(creditsData) do
                    table.insert(extra, {data = data, tab = "Credits", category = ""})
                end
            end
        end
    end
    for _, entry in ipairs(extra) do
        local duplicate = false
        for _, existing in ipairs(results) do
            if existing.data == entry.data then
                duplicate = true
                break
            end
        end
        if not duplicate then
            table.insert(results, entry)
        end
    end
    return results
end

local function displaySearchResults(query)
    local lowered = string.lower(query)
    clearContent()
    if lowered == "" then
        buildSelectedTab(selectedTab)
        return
    end

    local seen = {}
    local matches = {}
    for _, entry in ipairs(getAllSearchableOptions()) do
        local data = entry.data
        local haystack = string.lower((data.title or "") .. " " .. (data.description or ""))
        if string.find(haystack, lowered, 1, true) and not seen[data.title] then
            seen[data.title] = true
            table.insert(matches, entry)
        end
    end

    if #matches == 0 then
        local none = new("Frame", {
            BackgroundColor3 = DARK,
            BorderSizePixel = 0,
            Size = UDim2.fromScale(0.98, 0.18),
            LayoutOrder = 1
        }, ContentScroll)
        corner(none, 0.06)
        stroke(none, GRAY, 0.55, 1)
        new("TextLabel", {
            BackgroundTransparency = 1,
            Size = UDim2.fromScale(0.9, 0.8),
            Position = UDim2.fromScale(0.05, 0.1),
            Text = "No matching options found.",
            TextColor3 = GRAY_3,
            TextScaled = true,
            Font = Enum.Font.Gotham,
            TextXAlignment = Enum.TextXAlignment.Center,
            TextYAlignment = Enum.TextYAlignment.Center
        }, none)
        return
    end

    for index, entry in ipairs(matches) do
        local data = entry.data
        local holder, button, visual = addOption(ContentScroll, {
            id = "Search" .. tostring(index),
            title = data.title,
            description = data.description .. " • " .. entry.tab,
            kind = data.kind,
            order = index
        })
        optionEntries[#optionEntries + 1] = {data = data, holder = holder, button = button, visual = visual, tab = entry.tab, category = entry.category}
        button.Activated:Connect(function()
            if data.kind == "toggle" then
                executeOption(data)
                setToggleVisual(visual, data.enabled == true)
            elseif data.kind == "click" then
                executeOption(data)
            end
        end)
        if data.kind == "toggle" then
            setToggleVisual(visual, data.enabled == true)
        end
    end
end

local function pointOnCornerBand(inputPosition)
    local camera = Workspace.CurrentCamera
    if not camera then return false end
    local viewport = camera.ViewportSize
    local abs = MainFrame.AbsolutePosition
    local size = MainFrame.AbsoluteSize
    local x, y = inputPosition.X - abs.X, inputPosition.Y - abs.Y
    local marginX = math.max(size.X * 0.14, 1)
    local marginY = math.max(size.Y * 0.14, 1)
    local inLeft = x >= 0 and x <= marginX
    local inRight = x >= size.X - marginX and x <= size.X
    local inTop = y >= 0 and y <= marginY
    local inBottom = y >= size.Y - marginY and y <= size.Y
    return (inLeft or inRight) and (inTop or inBottom)
end

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    if ConfirmOverlay.Visible then return end
    if pointOnCornerBand(input.Position) then
        menuDragging = true
        menuDragStart = input.Position
        menuStartPos = MainFrame.Position
        menuDragInput = input
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        menuDragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if menuDragging and input == menuDragInput then
        local camera = Workspace.CurrentCamera
        if not camera then return end
        local viewport = camera.ViewportSize
        local delta = input.Position - menuDragStart
        local nx = menuStartPos.X.Scale + delta.X / viewport.X
        local ny = menuStartPos.Y.Scale + delta.Y / viewport.Y
        local minX, maxX = -0.64, 1.64
        local minY, maxY = -0.64, 1.64
        MainFrame.Position = UDim2.fromScale(clampScale(nx, minX, maxX), clampScale(ny, minY, maxY))
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if menuDragging then
            menuDragging = false
            menuDragInput = nil
            local x = MainFrame.Position.X.Scale
            local y = MainFrame.Position.Y.Scale
            local width = MainFrame.Size.X.Scale
            local height = MainFrame.Size.Y.Scale
            local fullyOutsideLeft = x + width < 0
            local fullyOutsideRight = x > 1
            local fullyOutsideTop = y + height < 0
            local fullyOutsideBottom = y > 1
            if fullyOutsideLeft or fullyOutsideRight or fullyOutsideTop or fullyOutsideBottom then
                local targetX = math.clamp(x, -0.2, 0.2)
                local targetY = math.clamp(y, -0.2, 0.2)
                if fullyOutsideRight then targetX = 0.82 end
                if fullyOutsideLeft then targetX = -0.02 end
                if fullyOutsideBottom then targetY = 0.72 end
                if fullyOutsideTop then targetY = -0.02 end
                local returnX = fullyOutsideRight and 0.72 or (fullyOutsideLeft and -0.02 or targetX)
                local returnY = fullyOutsideBottom and 0.72 or (fullyOutsideTop and -0.02 or targetY)
                tween(MainFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
                    Position = UDim2.fromScale(returnX, returnY)
                })
            end
        end
    end
end)
