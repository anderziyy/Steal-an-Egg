local function make(className, props)
    local obj = Instance.new(className)
    for k, v in pairs(props or {}) do
        obj[k] = v
    end
    return obj
end

local function uiScale(parent, scale)
    local s = make("UIScale", {Scale = scale or 1})
    s.Parent = parent
    return s
end

local function stroke(parent, thickness, transparency)
    local s = make("UIStroke", {
        Color = THEME.Gray2,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
    s.Parent = parent
    return s
end

local function corner(parent, radiusScale)
    local c = make("UICorner", {CornerRadius = UDim.new(radiusScale or 0, 0)})
    c.Parent = parent
    return c
end

local function icon(parent, id, sizeScale, transparency)
    return make("ImageLabel", {
        Parent = parent,
        BackgroundTransparency = 1,
        Size = UDim2.fromScale(sizeScale, sizeScale),
        Image = asset(id),
        ImageTransparency = transparency or 0,
        ScaleType = Enum.ScaleType.Fit,
    })
end

local function button(parent, props)
    local b = make("TextButton", {
        Parent = parent,
        BackgroundColor3 = props.background or THEME.Panel2,
        BackgroundTransparency = props.transparency or 0,
        BorderSizePixel = 0,
        Text = props.text or "",
        TextColor3 = props.textColor or THEME.White,
        TextSize = props.textSize or 14,
        Font = props.font or Enum.Font.GothamMedium,
        AutoButtonColor = false,
        Active = true,
    })
    b.Size = props.size or UDim2.fromScale(1, 1)
    b.Position = props.position or UDim2.fromScale(0, 0)
    if props.stroke then
        stroke(b, props.stroke, props.strokeTransparency or 0)
    end
    return b
end

local function tween(instance, info, props)
    local t = TweenService:Create(instance, info, props)
    t:Play()
    return t
end

local function setVisibleAnimated(frame, visible)
    if visible then
        frame.Visible = true
        frame.BackgroundTransparency = 1
        tween(frame, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = 0})
    else
        local t = tween(frame, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {BackgroundTransparency = 1})
        t.Completed:Connect(function()
            frame.Visible = false
        end)
    end
end

        local h = openButton.AbsoluteSize.Y / gui.AbsoluteSize.Y

local function isOutsideCompletely(frame)
    local pos = frame.AbsolutePosition
    local size = frame.AbsoluteSize
    local view = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(0, 0)
    return pos.X + size.X <= 0 or pos.X >= view.X or pos.Y + size.Y <= 0 or pos.Y >= view.Y
end

UserInputService.InputChanged:Connect(function(input)
    if menuDragging and input == menuDragInput then
        local delta = input.Position - menuDragStart
        menu.Position = UDim2.fromScale(
            menuStartPos.X.Scale + delta.X / gui.AbsoluteSize.X,
            menuStartPos.Y.Scale + delta.Y / gui.AbsoluteSize.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if menuDragging and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
        menuDragging = false
        if isOutsideCompletely(menu) then
            local view = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1, 1)
            local pos = menu.AbsolutePosition
            local size = menu.AbsoluteSize
            local x = pos.X / view.X
            local y = pos.Y / view.Y
            local w = size.X / view.X
            local h = size.Y / view.Y
            local sideTarget = x + w <= 0 and 0.02 or 0.98
            local targetX = x + w <= 0 and 0.02 or 0.98 - w
            local targetY = math.clamp(y, 0.02, 0.98 - h)
            tween(menu, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Position = UDim2.fromScale(targetX, targetY)
            })
        else
            local view = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1, 1)
            local pos = menu.AbsolutePosition
            local size = menu.AbsoluteSize
            local x = math.clamp(pos.X / view.X, 0, 1 - size.X / view.X)
            local y = math.clamp(pos.Y / view.Y, 0, 1 - size.Y / view.Y)
            menu.Position = UDim2.fromScale(x, y)
        end
    end
end)

local function setLowMode(enabled)
    lowMode = enabled
    if enabled then
        savedState.Materials = {}
        savedState.Effects = {}
        savedState.PartTransparency = {}
        for _, descendant in ipairs(workspace:GetDescendants()) do
            if descendant:IsA("BasePart") then
                savedState.Materials[descendant] = descendant.Material
                descendant.Material = Enum.Material.SmoothPlastic
            elseif descendant:IsA("ParticleEmitter") or descendant:IsA("Trail") or descendant:IsA("Beam") or descendant:IsA("Smoke") or descendant:IsA("Fire") or descendant:IsA("Sparkles") then
                savedState.Effects[descendant] = descendant.Enabled
                descendant.Enabled = false
            end
        end
        Lighting.GlobalShadows = false
        Lighting.EnvironmentDiffuseScale = 0
        Lighting.EnvironmentSpecularScale = 0
        Lighting.FogEnd = 100000
    else
        for part, material in pairs(savedState.Materials) do
            if part and part.Parent then
                pcall(function() part.Material = material end)
            end
        end
        for effect, enabledValue in pairs(savedState.Effects) do
            if effect and effect.Parent then
                pcall(function() effect.Enabled = enabledValue end)
            end
        end
        savedState.Materials = {}
        savedState.Effects = {}
        pcall(function()
            Lighting.GlobalShadows = true
            Lighting.EnvironmentDiffuseScale = 1
            Lighting.EnvironmentSpecularScale = 1
        end)
    end
end

local function startAntiAfk()
    if antiAfkThread then
        return
    end
    antiAfkThread = task.spawn(function()
        while antiAfk do
            task.wait(60)
            if not antiAfk then
                break
            end
            local character = Player.Character
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                pcall(function()
                    humanoid:Move(Vector3.new(0.5, 0, 0), true)
                    task.wait(0.25)
                    humanoid:Move(Vector3.zero, true)
                end)
                if VirtualInputManager then
                    pcall(function()
                        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.W, false, game)
                        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.W, false, game)
                    end)
                end
            end
        end
        antiAfkThread = nil
    end)
end

local function performCopy(option)
    local value
    if option.Value == "Discord" then
        value = "https://discord.gg/72P6NhC5Pf"
    elseif option.Value == "JobId" then
        value = game.JobId
    elseif option.Value == "PlaceId" then
        value = tostring(game.PlaceId)
    end
    if value then
        local success = copyText(value)
        local message = success and option.Notice or ("Copied value: " .. value)
        notify(message, option.NoticeIcon or 132744109593099)
    end
end

local function performAction(action)
    if action == "current" then
        pcall(function()
            TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, Player)
        end)
    elseif action == "rejoin" then
        local ok = pcall(function()
            TeleportService:Teleport(game.PlaceId, Player)
        end)
        if not ok then
            pcall(function()
                TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, Player)
            end)
        end
    end
end

function renderTab(tabName)
    activeTab = tabName
    for _, child in ipairs(optionScroll:GetChildren()) do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end

    local tabData = data[tabName]
    if not tabData then
        return
    end

    local order = 1
    local entries = {}
    for _, category in ipairs(tabData.Categories) do
        table.insert(entries, {Kind = "Category", Data = category})
        for _, option in ipairs(category.Options) do
            table.insert(entries, {Kind = "Option", Data = option, Category = category})
        end
    end
    for _, option in ipairs(tabData.Options) do
        table.insert(entries, {Kind = "Option", Data = option})
    end

    for _, entry in ipairs(entries) do
        if entry.Kind == "Category" then
            local frame = createCategory(entry.Data, tabName, order)
            frame.LayoutOrder = order
            categoryFrames[entry.Data.Name] = frame
        elseif entry.Kind == "Option" then
            if not entry.Category or entry.Category.Open then
                local frame = createOptionCard(entry.Data, tabName)
                frame.LayoutOrder = order
                table.insert(optionFrames, {Frame = frame, Option = entry.Data, Tab = tabName})
                order = order + 1
                continue
            end
        end
        order = order + 1
    end

    local query = string.lower(searchBox.Text)
    if query ~= "" then
        applySearch(query)
    end
end

function applySearch(query)
    query = string.lower(query or "")
    if query == "" then
        renderTab(activeTab)
        return
    end

    for _, child in ipairs(optionScroll:GetChildren()) do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end

    local results = {}
    for _, tab in ipairs(tabDefinitions) do
        local tabData = data[tab.Name]
        if tabData then
            for _, category in ipairs(tabData.Categories) do
                for _, option in ipairs(category.Options) do
                    local haystack = string.lower((option.Name or "") .. " " .. (option.Description or ""))
                    if string.find(haystack, query, 1, true) then
                        table.insert(results, option)
                    end
                end
            end
            for _, option in ipairs(tabData.Options) do
                local haystack = string.lower((option.Name or "") .. " " .. (option.Description or ""))
                if string.find(haystack, query, 1, true) then
                    table.insert(results, option)
                end
            end
        end
    end

    local order = 1
    for _, option in ipairs(results) do
        local frame = createOptionCard(option, activeTab)
        frame.LayoutOrder = order
        order = order + 1
    end
end

searchBox:GetPropertyChangedSignal("Text"):Connect(function()
    local hasText = searchBox.Text ~= ""
    clearSearch.Visible = hasText

local function clampOpenButton()
    local view = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1, 1)
    local pos = openButton.AbsolutePosition
    local size = openButton.AbsoluteSize
    local x = math.clamp(pos.X / view.X, 0.005, 0.995 - size.X / view.X)
    local y = math.clamp(pos.Y / view.Y, 0.005, 0.995 - size.Y / view.Y)
    openButton.Position = UDim2.fromScale(x, y)
end
