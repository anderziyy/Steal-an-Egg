local function showNotification(message, icon)
	local notification = Instance.new("Frame")
	notification.Name = "Notification"
	notification.Size = UDim2.new(0.25, 0, 0.08, 0)
	notification.Position = UDim2.new(0.73, 0, 0.9 - #notifications * 0.1, 0)
	notification.BackgroundColor3 = COLORS.Secondary
	notification.BorderSizePixel = 1
	notification.BorderColor3 = COLORS.Border
	notification.Parent = playerGui:FindFirstChild("PhantomHubGui")
	notification.ZIndex = 3000
	
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, CORNER_RADIUS)
	corner.Parent = notification
	
	local iconLabel = Instance.new("ImageLabel")
	iconLabel.Name = "Icon"
	iconLabel.Size = UDim2.new(0.15, 0, 0.8, 0)
	iconLabel.Position = UDim2.new(0.02, 0, 0.1, 0)
	iconLabel.BackgroundTransparency = 1
	iconLabel.Image = icon
	iconLabel.Parent = notification
	iconLabel.ZIndex = 3000
	
	local messageLabel = Instance.new("TextLabel")
	messageLabel.Name = "Message"
	messageLabel.Size = UDim2.new(0.8, 0, 1, 0)
	messageLabel.Position = UDim2.new(0.18, 0, 0, 0)
	messageLabel.BackgroundTransparency = 1
	messageLabel.Text = message
	messageLabel.TextColor3 = COLORS.Text
	messageLabel.TextScaled = true
	messageLabel.TextWrapped = true
	messageLabel.Font = Enum.Font.Gotham
	messageLabel.Parent = notification
	messageLabel.ZIndex = 3000
	
	local progressBar = Instance.new("Frame")
	progressBar.Name = "Progress"
	progressBar.Size = UDim2.new(0, 0, 0.05, 0)
	progressBar.Position = UDim2.new(0, 0, 1, 0)
	progressBar.BackgroundColor3 = COLORS.Text
	progressBar.BorderSizePixel = 0
	progressBar.Parent = notification
	progressBar.ZIndex = 3000
	
	table.insert(notifications, notification)
	
	for i, notif in ipairs(notifications) do
		if i > 1 then
			local tween = game:GetService("TweenService"):Create(
				notif,
				TweenInfo.new(0.3),
				{Position = UDim2.new(0.73, 0, 0.9 - i * 0.1, 0)}
			)
			tween:Play()
		end
	end
	
	local tween = game:GetService("TweenService"):Create(
		progressBar,
		TweenInfo.new(3),
		{Size = UDim2.new(1, 0, 0.05, 0)}
	)
	tween:Play()
	
	tween.Completed:Connect(function()
		local outTween = game:GetService("TweenService"):Create(
			notification,
			TweenInfo.new(0.3),
			{Position = notification.Position + UDim2.new(0.3, 0, 0, 0)}
		)
		outTween:Play()
		outTween.Completed:Connect(function()
			notification:Destroy()
			table.remove(notifications, table.find(notifications, notification))
		end)
	end)
end

		end)
	end
	
	return optBtn
end

local function refreshOptions(menuElements)
	menuElements.optionsFrame:ClearAllChildren()
	
	local searchFrame = Instance.new("Frame")
	searchFrame.Name = "SearchBar"
	searchFrame.Size = UDim2.new(1, -10, 0.08, 0)
	searchFrame.Position = UDim2.new(0, 5, 0, 5)
	searchFrame.BackgroundColor3 = COLORS.Tertiary
	searchFrame.BorderSizePixel = 1
	searchFrame.BorderColor3 = COLORS.Border
	searchFrame.Parent = menuElements.optionsFrame
	searchFrame.LayoutOrder = -999
	searchFrame.ZIndex = 1000
	
	local searchCorner = Instance.new("UICorner")
	searchCorner.CornerRadius = UDim.new(0, CORNER_RADIUS)
	searchCorner.Parent = searchFrame
	
	local searchIcon = Instance.new("ImageLabel")
	searchIcon.Name = "SearchIcon"
	searchIcon.Size = UDim2.new(0.1, 0, 0.8, 0)
	searchIcon.Position = UDim2.new(0.02, 0, 0.1, 0)
	searchIcon.BackgroundTransparency = 1
	searchIcon.Image = "rbxassetid://73858257403039"
	searchIcon.Parent = searchFrame
	searchIcon.ZIndex = 1000
	
	menuElements.searchInput = Instance.new("TextBox")
	menuElements.searchInput.Name = "SearchInput"
	menuElements.searchInput.Size = UDim2.new(0.75, 0, 1, 0)
	menuElements.searchInput.Position = UDim2.new(0.12, 0, 0, 0)
	menuElements.searchInput.BackgroundTransparency = 1
	menuElements.searchInput.Text = searchText
	menuElements.searchInput.PlaceholderText = "Search..."
	menuElements.searchInput.PlaceholderColor3 = COLORS.TextMuted
	menuElements.searchInput.TextColor3 = COLORS.Text
	menuElements.searchInput.TextScaled = true
	menuElements.searchInput.Font = Enum.Font.Gotham
	menuElements.searchInput.Parent = searchFrame
	menuElements.searchInput.ZIndex = 1000
	
	menuElements.clearBtn = Instance.new("TextButton")
	menuElements.clearBtn.Name = "ClearBtn"
	menuElements.clearBtn.Size = UDim2.new(0.1, 0, 0.8, 0)
	menuElements.clearBtn.Position = UDim2.new(0.88, 0, 0.1, 0)
	menuElements.clearBtn.BackgroundTransparency = 1
	menuElements.clearBtn.Text = "X"
	menuElements.clearBtn.TextColor3 = COLORS.Text
	menuElements.clearBtn.TextScaled = true
	menuElements.clearBtn.Font = Enum.Font.GothamBold
	menuElements.clearBtn.Parent = searchFrame
	menuElements.clearBtn.Visible = #searchText > 0
	menuElements.clearBtn.ZIndex = 1000
	
	menuElements.clearBtn.MouseButton1Click:Connect(function()
		menuElements.searchInput.Text = ""
		searchText = ""
		menuElements.clearBtn.Visible = false
		refreshOptions(menuElements)
	end)
	
	menuElements.searchInput.Changed:Connect(function()
		searchText = menuElements.searchInput.Text
		menuElements.clearBtn.Visible = #searchText > 0
		refreshOptions(menuElements)
	end)
	
	local tabContent = CONTENT[selectedTab] or {}
	local totalItems = 0
	
	if selectedTab == "Config" and tabContent[1] and tabContent[1].type == "category" then
		local category = tabContent[1]
		if category.options then
			for _, option in ipairs(category.options) do
				if searchText == "" or string.lower(option.title):find(string.lower(searchText), 1, true) then
					createOptionButton(menuElements.optionsFrame, option, function()
						handleOptionAction(option.action)
					end)
					totalItems = totalItems + 1
				end
			end
		end
		
		if totalItems == 0 then
			createCategoryButton(menuElements.optionsFrame, category, function()
				refreshOptions(menuElements)
			end)
			
			if category.open then
				for _, option in ipairs(category.options) do
					createOptionButton(menuElements.optionsFrame, option, function()
						handleOptionAction(option.action)
					end)
					totalItems = totalItems + 1
				end
			end
		end
	else
		for _, item in ipairs(tabContent) do
			if searchText == "" then
				if item.type == "category" then
					createCategoryButton(menuElements.optionsFrame, item, function()
						refreshOptions(menuElements)
					end)
					totalItems = totalItems + 1
					
					if item.open then
						for _, option in ipairs(item.options) do
							createOptionButton(menuElements.optionsFrame, option, function()
								handleOptionAction(option.action)
							end)
							totalItems = totalItems + 1
						end
					end
				else
					createOptionButton(menuElements.optionsFrame, item, function()
						handleOptionAction(item.action)
					end)
					totalItems = totalItems + 1
				end
			else
				if item.type == "option" and string.lower(item.title):find(string.lower(searchText), 1, true) then
					createOptionButton(menuElements.optionsFrame, item, function()
						handleOptionAction(item.action)
					end)
					totalItems = totalItems + 1
				end
			end
		end
	end
	
	if totalItems >= 7 then
		menuElements.optionsFrame.CanvasSize = UDim2.new(0, 0, (totalItems * 0.15) + 0.5, 0)
	else
		menuElements.optionsFrame.CanvasSize = UDim2.new(0, 0, 1, 0)
	end
end

local function getScreenSize()

end

local function clampPosition(position, size)
	local screenSize = getScreenSize()
	local maxX = screenSize.X - (size.X.Scale * screenSize.X)
	local maxY = screenSize.Y - (size.Y.Scale * screenSize.Y)
	
	if position.X.Scale < 0 then
		position = UDim2.new(0, 0, position.Y.Scale, position.Y.Offset)
	elseif position.X.Scale > 1 then
		position = UDim2.new(1, -size.X.Offset, position.Y.Scale, position.Y.Offset)
	end
	
	if position.Y.Scale < 0 then
		position = UDim2.new(position.X.Scale, position.X.Offset, 0, 0)
	elseif position.Y.Scale > 1 then
		position = UDim2.new(position.X.Scale, position.X.Offset, 1, -size.Y.Offset)
	end
	
	return position
end

floatingButton.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isDraggingButton = true
		dragOffset = getUserInputPosition() - floatingButton.AbsolutePosition
		currentDragType = "button"
	end
end)

floatingButton.InputEnded:Connect(function(input, gameProcessed)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isDraggingButton = false
		currentDragType = nil
	end
end)

menuElements.main.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		local mouse = game:GetService("Mouse")
		local mousePos = Vector2.new(mouse.X, mouse.Y)
		local topBarAbsPos = menuElements.topBar.AbsolutePosition
		local topBarAbsSize = menuElements.topBar.AbsoluteSize
		
		if mousePos.Y < topBarAbsPos.Y + topBarAbsSize.Y then
			local distFromLeft = math.abs(mousePos.X - topBarAbsPos.X)
			local distFromRight = math.abs(mousePos.X - (topBarAbsPos.X + menuElements.main.AbsoluteSize.X))
			local distFromTop = math.abs(mousePos.Y - topBarAbsPos.Y)
			local distFromBottom = math.abs(mousePos.Y - (topBarAbsPos.Y + topBarAbsSize.Y))
			
			if (distFromLeft < 30 or distFromRight < 30 or distFromTop < 30 or distFromBottom < 30) and
			   distFromLeft < 100 and distFromRight < 100 then
				isDraggingMenu = true
				dragOffset = mousePos - menuElements.main.AbsolutePosition
				currentDragType = "menu"
			end
		end
	end
end)

UserInputService.InputEnded:Connect(function(input, gameProcessed)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isDraggingMenu = false
		if currentDragType == "menu" then
			currentDragType = nil
		end
	end
end)

local mouse = game:GetService("Mouse")

function getUserInputPosition()
	return Vector2.new(mouse.X, mouse.Y)
end

RunService.RenderStepped:Connect(function()
	if isDraggingButton and floatingButtonActive then
		local inputPos = getUserInputPosition()
		local newPosition = inputPos - dragOffset
		floatingButton.Position = UDim2.new(0, newPosition.X, 0, newPosition.Y)
		
		local screenSize = screenGui.AbsoluteSize
		local buttonSize = floatingButton.AbsoluteSize
		
		if newPosition.X < 0 then
			floatingButton.Position = UDim2.new(0, 0, floatingButton.Position.Y.Scale, floatingButton.Position.Y.Offset)
		elseif newPosition.X + buttonSize.X > screenSize.X then
			floatingButton.Position = UDim2.new(0, screenSize.X - buttonSize.X, floatingButton.Position.Y.Scale, floatingButton.Position.Y.Offset)
		end
		
		if newPosition.Y < 0 then
			floatingButton.Position = UDim2.new(floatingButton.Position.X.Scale, floatingButton.Position.X.Offset, 0, 0)
		elseif newPosition.Y + buttonSize.Y > screenSize.Y then
			floatingButton.Position = UDim2.new(floatingButton.Position.X.Scale, floatingButton.Position.X.Offset, 0, screenSize.Y - buttonSize.Y)
		end
	end
	
	if isDraggingMenu then
		local inputPos = getUserInputPosition()
		local newPosition = inputPos - dragOffset
		menuElements.main.Position = UDim2.new(0, newPosition.X, 0, newPosition.Y)
		
		local screenSize = screenGui.AbsoluteSize
		local menuSize = menuElements.main.AbsoluteSize
		
		if newPosition.X + menuSize.X < 5 then
			menuElements.main.Position = UDim2.new(0, -menuSize.X + 5, menuElements.main.Position.Y.Scale, menuElements.main.Position.Y.Offset)
		elseif newPosition.X > screenSize.X - 5 then
			menuElements.main.Position = UDim2.new(0, screenSize.X - 5, menuElements.main.Position.Y.Scale, menuElements.main.Position.Y.Offset)
		end
		
		if newPosition.Y < 0 then
			menuElements.main.Position = UDim2.new(menuElements.main.Position.X.Scale, menuElements.main.Position.X.Offset, 0, 0)
		elseif newPosition.Y + menuSize.Y > screenSize.Y then
			menuElements.main.Position = UDim2.new(menuElements.main.Position.X.Scale, menuElements.main.Position.X.Offset, 0, screenSize.Y - menuSize.Y)
		end
	end
end)
