local function make(className, properties, parent)
    local object = Instance.new(className)
    for key, value in pairs(properties or {}) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function preventTranslation(root)
    local function visit(object)
        pcall(function()
            if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
                object.AutoLocalize = false
            end
        end)
        for _, child in ipairs(object:GetChildren()) do
            visit(child)
        end
    end
    visit(root)
end

local function tween(object, duration, properties, style, direction)
    local info = TweenInfo.new(
        duration or 0.18,
        style or Enum.EasingStyle.Quint,
        direction or Enum.EasingDirection.Out
    )
    local animation = TweenService:Create(object, info, properties)
    animation:Play()
    return animation
end

local cleanupCallbacks = {}
local cleanedUp = false
local pendingAction = nil

local function registerCleanup(callback)
    if type(callback) ~= "function" then
        return
    end
    table.insert(cleanupCallbacks, callback)
end

local function cleanupAll()
    if cleanedUp then
        return
    end
    cleanedUp = true
    for index = #cleanupCallbacks, 1, -1 do
        local callback = cleanupCallbacks[index]
        cleanupCallbacks[index] = nil
        pcall(callback)
    end
end

local function getViewport()
    local camera = workspace.CurrentCamera
    if not camera then
        return Vector2.new(1280, 720)
    end
    return camera.ViewportSize
end

local function updateGlobalScale()
    local viewport = getViewport()
    local reference = math.min(viewport.X, viewport.Y)
    local target = reference / 720
    RootScale.Scale = math.clamp(target, 0.82, 1.22)
end

updateGlobalScale()

do
    local camera = workspace.CurrentCamera
    if camera then
        local connection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateGlobalScale)
        registerCleanup(function()
            connection:Disconnect()
        end)
    end
end

local function copyText(text)
    if typeof(setclipboard) == "function" then
        local ok = pcall(setclipboard, text)
        if ok then
            return true
        end
    end
    if typeof(toclipboard) == "function" then
        local ok = pcall(toclipboard, text)
        if ok then
            return true
        end
    end
    return false
end

local function getVisibleItems(tabName, query)
    local normalized = string.lower(query or "")
    if normalized == "" then
        return datasets[tabName] or {}
    end

    local filtered = {}
    for _, item in ipairs(datasets[tabName] or {}) do
        if item.Type == "Category" then
            for _, child in ipairs(item.Items or {}) do
                local title = string.lower(child.Title or "")
                local description = string.lower(child.Description or "")
                if string.find(title, normalized, 1, true)
                    or string.find(description, normalized, 1, true) then
                    table.insert(filtered, child)
                end
            end
        else
            local title = string.lower(item.Title or "")
            local description = string.lower(item.Description or "")
            if string.find(title, normalized, 1, true)
                or string.find(description, normalized, 1, true) then
                table.insert(filtered, item)
            end
        end
    end

    return filtered
end

local toggleStates = {}
local toggleConnections = {}

local function getHumanoid()
    local character = LocalPlayer.Character
    return character and character:FindFirstChildOfClass("Humanoid") or nil
end

local function setToggleState(item, enabled)
    toggleStates[item] = enabled

    if item.Title ~= "Anti AFK" then
        return
    end

    if toggleConnections[item] then
        toggleConnections[item]:Disconnect()
        toggleConnections[item] = nil
    end

    if enabled then
        local running = true
        local connection = {
            Disconnect = function()
                running = false
            end,
        }

        toggleConnections[item] = connection

        task.spawn(function()
            while running and not cleanedUp do
                task.wait(60)
                if not running or cleanedUp then
                    break
                end

                local humanoid = getHumanoid()
                if humanoid and humanoid.Health > 0 then
                    humanoid:Move(Vector3.new(1, 0, 0), true)
                    task.wait(0.18)
                    humanoid:Move(Vector3.zero, true)
                end
            end
        end)
    end
end

renderResults = function()
    for _, tabInfo in ipairs(tabs) do
        local container = resultContainers[tabInfo.Name]
        if container then
            container.Visible = tabInfo.Name == currentTab
        end
    end

    local container = resultContainers[currentTab]
    if not container then
        return
    end

    clearResults(container)

    local query = SearchBox.Text
    local items = getVisibleItems(currentTab, query)

    if #items == 0 then
        if query ~= "" then
            showEmptyState(container, "No matching options found.")
        elseif currentTab == "Credits" then
            showEmptyState(container, "No options are available in Credits.")
        elseif currentTab == "Config" then
            showEmptyState(container, "No options are available in Config.")
        else
            showEmptyState(container, "This tab has no options yet.")
        end
        return
    end

    local layoutIndex = 0

    if query ~= "" then
        for _, item in ipairs(items) do
            layoutIndex += 1
            createStandardRow(container, item, layoutIndex)
        end
        return
    end

    for _, item in ipairs(items) do
        layoutIndex += 1
        if item.Type == "Category" then
            createCategoryRow(container, item, layoutIndex, query)
            if item.Expanded then
                layoutIndex += #(item.Items or {})
            end
        else
            createStandardRow(container, item, layoutIndex)
        end
    end
end

    currentTab = name

        tween(button, 0.14, {
            BackgroundColor3 = selected and COLORS.Gray2 or COLORS.Black2,

        local icon = button:FindFirstChild("Icon")
        if icon and icon:IsA("ImageLabel") then
            tween(icon, 0.14, {
                ImageColor3 = selected and COLORS.White or COLORS.SoftWhite,

        end

        local label = button:FindFirstChild("Label")
        if label and label:IsA("TextLabel") then
            tween(label, 0.14, {
                TextColor3 = selected and COLORS.White or COLORS.SoftWhite,
            })
        end
    end

    SearchBox.Text = ""
    SearchHint.Visible = true
    SearchClear.Visible = false
    renderResults()
end

local closeFallbackConnection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then
        return

    if cleanedUp or not ScreenGui.Parent or not Menu.Visible or ConfirmOverlay.Visible then
        return

        and input.UserInputType ~= Enum.UserInputType.Touch then

    local size = CloseButton.AbsoluteSize

    local inside =
        input.Position.X >= position.X
        and input.Position.X <= position.X + size.X
        and input.Position.Y >= position.Y
        and input.Position.Y <= position.Y + size.Y

    if inside then
        requestCloseFromButton()
    end
end)

local launcherDragging = false
local launcherDragInput = nil
local launcherDragStart = nil
local launcherStartPos = nil
local launcherDidDrag = false

local function clampLauncher(position)
    local x = math.clamp(position.X.Scale, 0.0, 0.925)
    local y = math.clamp(position.Y.Scale, 0.02, 0.92)

    if position.X.Scale < 0 then
        x = 0.015
    elseif position.X.Scale > 0.925 then
        x = 0.907
    end

    return UDim2.fromScale(x, y)
end

local function updateLauncherDrag(input)
    if not launcherDragStart or not launcherStartPos then
        return
    end

    local delta = input.Position - launcherDragStart

    if delta.Magnitude > 4 then
        launcherDidDrag = true
    end

    local viewport = getViewport()
    if viewport.X <= 0 or viewport.Y <= 0 then
        return
    end

    local dx = delta.X / viewport.X
    local dy = delta.Y / viewport.Y

    LauncherFrame.Position = clampLauncher(UDim2.fromScale(
        launcherStartPos.X.Scale + dx,
        launcherStartPos.Y.Scale + dy
    ))
end

local function beginLauncherDrag(input)
    launcherDragging = true
    launcherDidDrag = false
    launcherDragStart = input.Position
    launcherStartPos = LauncherFrame.Position
    launcherDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
end

local function endLauncherDrag()
    launcherDragging = false
    launcherDragInput = nil
    launcherDragStart = nil
    launcherStartPos = nil
end

Launcher.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        beginLauncherDrag(input)
    end
end)

local function updateMenuDrag(input)
    if not menuDragStart or not menuStartPos then
        return
    end

    local delta = input.Position - menuDragStart

    if delta.Magnitude < 4 then
        return
    end

    if os.clock() - pointerDownAt < 0.06 then
        return
    end

    didDrag = true
    menuDragging = true

    local viewport = getViewport()
    if viewport.X <= 0 or viewport.Y <= 0 then
        return
    end

    local dx = delta.X / viewport.X
    local dy = delta.Y / viewport.Y

    Menu.Position = UDim2.fromScale(
        menuStartPos.X.Scale + dx,
        menuStartPos.Y.Scale + dy
    )
end

local function endMenuDrag()
    if not menuDragging then
        menuDragInput = nil
        return
    end

    local viewport = getViewport()
    local absolutePosition = Menu.AbsolutePosition
    local absoluteSize = Menu.AbsoluteSize

    local fullyLeft = absolutePosition.X + absoluteSize.X < 0
    local fullyRight = absolutePosition.X > viewport.X
    local fullyTop = absolutePosition.Y + absoluteSize.Y < 0
    local fullyBottom = absolutePosition.Y > viewport.Y

    if fullyLeft then
        local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
        tween(Menu, 0.20, {
            Position = UDim2.fromScale(0.02, y),
        })
    elseif fullyRight then
        local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
        tween(Menu, 0.20, {
            Position = UDim2.fromScale(0.98, y),
        })
    elseif fullyTop then
        tween(Menu, 0.20, {
            Position = UDim2.fromScale(
                math.clamp(Menu.Position.X.Scale, 0.02, 0.98),
                0.05
            ),
        })
    elseif fullyBottom then
        tween(Menu, 0.20, {
            Position = UDim2.fromScale(
                math.clamp(Menu.Position.X.Scale, 0.02, 0.98),
                0.95
            ),
        })
    end

    menuDragging = false
    menuDragInput = nil
    menuDragStart = nil
    menuStartPos = nil

    tween(MenuStroke, 0.12, {
        Color = COLORS.Gray3,
        Transparency = 0.12,
    })
end

local inputChangedConnection = UserInputService.InputChanged:Connect(function(input)
    if launcherDragging
        and (
            (launcherDragInput and input == launcherDragInput)
            or (not launcherDragInput and input.UserInputType == Enum.UserInputType.MouseMovement)
        ) then
        updateLauncherDrag(input)
    end

    if menuDragInput and input == menuDragInput then
        updateMenuDrag(input)
    elseif menuDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        updateMenuDrag(input)
    elseif menuDragStart and not menuDragInput
        and input.UserInputType == Enum.UserInputType.MouseMovement then
        updateMenuDrag(input)
    end
end)

local inputEndedConnection = UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    if launcherDragging then
        local shouldToggle = not launcherDidDrag
        endLauncherDrag()

        if shouldToggle then
            toggleMenu()
        end
    end

    if menuDragging then
        endMenuDrag()
    end
end)

    })

SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
    local hasText = SearchBox.Text ~= ""
    SearchHint.Visible = not hasText
    SearchClear.Visible = hasText
    renderResults()
end)

SearchClear.Activated:Connect(function()
    SearchBox.Text = ""
    SearchBox:CaptureFocus()
end)

local currentCameraConnection
workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    if currentCameraConnection then
        currentCameraConnection:Disconnect()
        currentCameraConnection = nil
    end

    local camera = workspace.CurrentCamera
    if camera then
        currentCameraConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(updateGlobalScale)
    end
end)

registerCleanup(function()
    if currentCameraConnection then
        currentCameraConnection:Disconnect()
        currentCameraConnection = nil
    end
end)

registerCleanup(function()
    for _, connection in pairs(toggleConnections) do
        if connection and connection.Disconnect then
            pcall(function()
                connection:Disconnect()
            end)
        end
    end
    toggleConnections = {}

    pcall(function()
        inputChangedConnection:Disconnect()
    end)

    pcall(function()
        inputEndedConnection:Disconnect()
    end)
end)
