local function asset(id)
    return "rbxassetid://" .. tostring(id)

local function clamp(value, minimum, maximum)
    return math.max(minimum, math.min(maximum, value))
end

local function scaleUDim2(x, y)

local function makeToggle(parent, optionData)
    local button = Instance.new("TextButton")

    button.ZIndex = 20
    button.Parent = parent

    local buttonStroke = Instance.new("UIStroke")
    buttonStroke.Color = Color3.fromRGB(85, 85, 85)
    buttonStroke.Thickness = 1
    buttonStroke.Transparency = 0.15
    buttonStroke.Parent = button

    local knob = Instance.new("Frame")
    knob.BackgroundColor3 = Color3.fromRGB(175, 175, 175)
    knob.BorderSizePixel = 0
    knob.Size = UDim2.new(0.42, 0, 0.74, 0)
    knob.AnchorPoint = Vector2.new(0.5, 0.5)
    knob.Position = UDim2.new(0.27, 0, 0.5, 0)
    knob.ZIndex = 21
    knob.Parent = button

    local knobCorner = Instance.new("UICorner")
    knobCorner.CornerRadius = UDim.new(1, 0)
    knobCorner.Parent = knob

    local value = optionState[optionData.name] == true

    local function render()
        value = optionState[optionData.name] == true
        button.BackgroundColor3 = value and Color3.fromRGB(235, 235, 235) or Color3.fromRGB(42, 42, 42)
        knob.BackgroundColor3 = value and Color3.fromRGB(25, 25, 25) or Color3.fromRGB(175, 175, 175)

    end

    button.Activated:Connect(function()
        optionState[optionData.name] = not value
        render()
    end)

    render()
end

function renderTab(query)
    clearResults()
    local normalized = string.lower(query or "")
    for index, categoryData in ipairs(data[currentTab] or {}) do
        local shouldShow = normalized == ""
        if not shouldShow then
            for _, optionData in ipairs(categoryData.options) do
                if string.find(string.lower(optionData.name), normalized, 1, true) then
                    shouldShow = true
                    break
                end
            end
        end
        if shouldShow then
            makeCategory(index, categoryData, normalized)
        end
    end
end

        button.ImageLabel.ImageColor3 = selected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(165, 165, 165)
    end
    renderTab(searchBox.Text)
end

local function updateResponsiveSize()
    local camera = workspace.CurrentCamera
    if not camera then
        return
    end
    viewport = camera.ViewportSize

    local xScale, yScale = squareScale(74)
    openButton.Size = scaleUDim2(xScale, yScale)

    local availableWidth = viewport.X * 0.78
    local availableHeight = viewport.Y * 0.72
    local targetHeight = math.min(availableHeight, availableWidth / 1.72)
    local targetWidth = targetHeight * 1.72

    local menuX = targetWidth / math.max(viewport.X, 1)
    local menuY = targetHeight / math.max(viewport.Y, 1)

    mainFrame.Size = scaleUDim2(menuX, menuY)

    local currentPos = mainFrame.Position
    local px = currentPos.X.Scale
    local py = currentPos.Y.Scale

    mainFrame.Position = scaleUDim2(
        clamp(px, 0.5 * menuX, 1 - 0.5 * menuX),
        clamp(py, 0.5 * menuY, 1 - 0.5 * menuY)
    )

    local confirmWidth = math.min(targetWidth * 0.68, viewport.X * 0.62)
    local confirmHeight = confirmWidth / 1.7
    confirmation.Size = scaleUDim2(
        confirmWidth / math.max(viewport.X, 1),
        confirmHeight / math.max(viewport.Y, 1)
    )
    confirmation.Position = UDim2.new(0.5, 0, 0.5, 0)
end

updateResponsiveSize()

local function fullyOutside(position, size)
    local x = position.X.Scale
    local y = position.Y.Scale
    local sx = size.X.Scale
    local sy = size.Y.Scale
    return x + sx <= 0 or x >= 1 or y + sy <= 0 or y >= 1
end

local function returnToVisibleSide(frame)
    local x = frame.Position.X.Scale
    local y = frame.Position.Y.Scale
    local sx = frame.Size.X.Scale
    local sy = frame.Size.Y.Scale

    if x + sx <= 0 then
        frame.Position = scaleUDim2(0, clamp(y, 0, 1 - sy))
    elseif x >= 1 then
        frame.Position = scaleUDim2(1 - sx, clamp(y, 0, 1 - sy))
    elseif y + sy <= 0 then
        frame.Position = scaleUDim2(clamp(x, 0, 1 - sx), 0)
    elseif y >= 1 then
        frame.Position = scaleUDim2(clamp(x, 0, 1 - sx), 1 - sy)
    end
end

local function makeDraggable(frame, hitTargets)
    local dragging = false
    local dragInput = nil
    local dragStart = nil
    local startPosition = nil
    local moved = false

    local function begin(input)
        if dragging then
            return
        end
        dragging = true
        dragInput = input
        dragStart = input.Position
        startPosition = frame.Position
        moved = false
    end

    local function update(input)
        if not dragging or not dragStart or not startPosition then
            return
        end

        local delta = input.Position - dragStart
        local camera = workspace.CurrentCamera
        if not camera then
            return
        end

        local viewportSize = camera.ViewportSize
        local dx = delta.X / math.max(viewportSize.X, 1)
        local dy = delta.Y / math.max(viewportSize.Y, 1)

        if math.abs(delta.X) > 4 or math.abs(delta.Y) > 4 then
            moved = true
        end

        frame.Position = scaleUDim2(
            startPosition.X.Scale + dx,
            startPosition.Y.Scale + dy
        )
    end

    local function finish()
        if not dragging then
            return
        end
        dragging = false
        dragInput = nil
        if fullyOutside(frame.Position, frame.Size) then
            returnToVisibleSide(frame)
        else
            local sx = frame.Size.X.Scale
            local sy = frame.Size.Y.Scale
            frame.Position = scaleUDim2(
                clamp(frame.Position.X.Scale, -sx + 0.002, 0.998),
                clamp(frame.Position.Y.Scale, -sy + 0.002, 0.998)
            )
        end
    end

    for _, target in ipairs(hitTargets) do
        target.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                begin(input)
            end
        end)

        target.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                dragInput = input
            end
        end)
    end

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            update(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            finish()
        end
    end)

    return function()
        return moved
    end
end

local openDragging = false
local openDragStart = nil
local openStartPosition = nil
local openDragInput = nil
local openMoved = false

openButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        openDragging = true
        openDragStart = input.Position
        openStartPosition = openButton.Position
        openDragInput = input
        openMoved = false
    end
end)

openButton.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        openDragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not openDragging or input ~= openDragInput then
        return
    end

    local camera = workspace.CurrentCamera
    if not camera then
        return
    end

    local delta = input.Position - openDragStart
    if math.abs(delta.X) > 4 or math.abs(delta.Y) > 4 then
        openMoved = true
    end

    local viewportSize = camera.ViewportSize
    openButton.Position = scaleUDim2(
        openStartPosition.X.Scale + delta.X / math.max(viewportSize.X, 1),
        openStartPosition.Y.Scale + delta.Y / math.max(viewportSize.Y, 1)
    )
end)

UserInputService.InputEnded:Connect(function(input)
    if not openDragging then
        return
    end
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        openDragging = false

        local sx = openButton.Size.X.Scale
        local sy = openButton.Size.Y.Scale
        local x = openButton.Position.X.Scale
        local y = openButton.Position.Y.Scale

        if x + sx <= 0 then
            openButton.Position = scaleUDim2(0, clamp(y, 0, 1 - sy))
        elseif x >= 1 then
            openButton.Position = scaleUDim2(1 - sx, clamp(y, 0, 1 - sy))
        elseif y + sy <= 0 then
            openButton.Position = scaleUDim2(clamp(x, 0, 1 - sx), 0)
        elseif y >= 1 then
            openButton.Position = scaleUDim2(clamp(x, 0, 1 - sx), 1 - sy)
        else
            openButton.Position = scaleUDim2(x, y)
        end

        if not openMoved then
            setMenuVisible(not menuVisible)
        end
    end
end)

end
