local function make(className, props, parent)
    local obj = Instance.new(className)
    for key, value in pairs(props or {}) do
        obj[key] = value
    end
    obj.Parent = parent
    return obj
end

local function corner(parent, radius)
    return make("UICorner", {
        CornerRadius = UDim.new(0, radius or 8),
    }, parent)
end

local function stroke(parent, color, thickness, transparency)
    return make("UIStroke", {
        Color = color or Theme.Line,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
    }, parent)
end

local function pad(parent, left, top, right, bottom)
    return make("UIPadding", {
        PaddingLeft = UDim.new(0, left or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingBottom = UDim.new(0, bottom or 0),
    }, parent)
end

local function tween(instance, time, props, style, direction)
    local info = TweenInfo.new(time, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out)
    local t = TweenService:Create(instance, info, props)
    t:Play()
    return t
end

local function setScalePosition(guiObject, x, y, anchorX, anchorY)
    guiObject.AnchorPoint = Vector2.new(anchorX or 0, anchorY or 0)
    guiObject.Position = UDim2.fromScale(x, y)
end

local function setPhysicalSquareSize(guiObject, widthScale)
    local camera = workspace.CurrentCamera
    if not camera then
        guiObject.Size = UDim2.fromScale(widthScale, widthScale)
        return
    end
    local viewport = camera.ViewportSize
    if viewport.Y <= 0 then
        guiObject.Size = UDim2.fromScale(widthScale, widthScale)
        return
    end
    local yScale = widthScale * viewport.X / viewport.Y
    guiObject.Size = UDim2.fromScale(widthScale, yScale)
end

local function connectViewportSquare(guiObject, widthScale)
    local camera = workspace.CurrentCamera
    if not camera then
        return
    end
    local function refresh()
        setPhysicalSquareSize(guiObject, widthScale)
    end
    refresh()
    camera:GetPropertyChangedSignal("ViewportSize"):Connect(refresh)
end

local function optionMatches(optionName, search)
    if search == "" then
        return true
    end
    return string.find(string.lower(optionName), string.lower(search), 1, true) ~= nil
end


local function clamp01(value)
    return math.clamp(value, 0, 1)
end

local function dragGui(gui, beginInput, startPosition, startSize)
    local camera = workspace.CurrentCamera
    if not camera then
        return
    end
    local viewport = camera.ViewportSize
    if viewport.X <= 0 or viewport.Y <= 0 then
        return
    end

    local dx = beginInput.Position.X - startPosition.X
    local dy = beginInput.Position.Y - startPosition.Y
    local px = beginInput.Position.X
    local py = beginInput.Position.Y

    local currentPos = gui.Position
    local currentX = currentPos.X.Scale
    local currentY = currentPos.Y.Scale

    local deltaXScale = dx / viewport.X
    local deltaYScale = dy / viewport.Y

    local newX = currentX + deltaXScale
    local newY = currentY + deltaYScale

    local sizeX = startSize.X.Scale
    local sizeY = startSize.Y.Scale
    local anchor = gui.AnchorPoint

    local minX = anchor.X * sizeX
    local maxX = 1 - (1 - anchor.X) * sizeX
    local minY = anchor.Y * sizeY
    local maxY = 1 - (1 - anchor.Y) * sizeY

    newX = math.clamp(newX, minX, maxX)
    newY = math.clamp(newY, minY, maxY)

    gui.Position = UDim2.fromScale(newX, newY)

    return px, py
end

local function addDraggable(gui, hitRegion, options)
    options = options or {}
    local dragging = false
    local inputObject
    local startPosition
    local startGuiPosition
    local startGuiSize
    local dragThreshold = options.Threshold or 5
    local pressedAt = 0

    hitRegion.Active = true

    hitRegion.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then
            return
        end

        pressedAt = os.clock()
        dragging = false
        inputObject = input
        startPosition = input.Position
        startGuiPosition = gui.Position
        startGuiSize = gui.Size

        local movedConnection
        movedConnection = UserInputService.InputChanged:Connect(function(changed)
            if changed ~= inputObject and changed.UserInputType ~= Enum.UserInputType.MouseMovement
                and changed.UserInputType ~= Enum.UserInputType.Touch then
                return
            end

            if not inputObject then
                return
            end

            local delta = changed.Position - startPosition
            if math.abs(delta.X) + math.abs(delta.Y) >= dragThreshold then
                if not dragging then
                    dragging = true
                end
            end

            if dragging then
                local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize
                if viewport then
                    local targetX = startGuiPosition.X.Scale + delta.X / viewport.X
                    local targetY = startGuiPosition.Y.Scale + delta.Y / viewport.Y
                    local sizeX = startGuiSize.X.Scale
                    local sizeY = startGuiSize.Y.Scale
                    local anchor = gui.AnchorPoint
                    local minX = anchor.X * sizeX
                    local maxX = 1 - (1 - anchor.X) * sizeX
                    local minY = anchor.Y * sizeY
                    local maxY = 1 - (1 - anchor.Y) * sizeY
                    targetX = math.clamp(targetX, minX, maxX)
                    targetY = math.clamp(targetY, minY, maxY)
                    gui.Position = UDim2.fromScale(targetX, targetY)
                end
            end
        end)

        local endConnection
        endConnection = UserInputService.InputEnded:Connect(function(ended)
            if ended ~= inputObject then
                return
            end
            inputObject = nil
            movedConnection:Disconnect()
            endConnection:Disconnect()

            if dragging then
                task.defer(function()
                    local finalX = gui.Position.X.Scale
                    local finalY = gui.Position.Y.Scale
                    local sizeX = gui.Size.X.Scale
                    local sizeY = gui.Size.Y.Scale
                    local anchor = gui.AnchorPoint
                    local minX = anchor.X * sizeX
                    local maxX = 1 - (1 - anchor.X) * sizeX
                    local minY = anchor.Y * sizeY
                    local maxY = 1 - (1 - anchor.Y) * sizeY
                    gui.Position = UDim2.fromScale(
                        math.clamp(finalX, minX, maxX),
                        math.clamp(finalY, minY, maxY)
                    )
                end)
            end
        end)
    end)
end

local openerWasDragged = false
local openerDragging = false
local openerInput
local openerStart
local openerStartGui
local openerSize

OpenButton.InputBegan:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    openerWasDragged = false
    openerDragging = false
    openerInput = input
    openerStart = input.Position
    openerStartGui = OpenButton.Position
    openerSize = OpenButton.Size
end)

UserInputService.InputChanged:Connect(function(input)
    if not openerInput then
        return
    end
    if input ~= openerInput and input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    local delta = input.Position - openerStart
    if math.abs(delta.X) + math.abs(delta.Y) >= 4 then
        openerDragging = true
        openerWasDragged = true
    end
    if openerDragging then
        local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize
        if viewport then
            local sizeX = openerSize.X.Scale
            local sizeY = openerSize.Y.Scale
            local targetX = openerStartGui.X.Scale + delta.X / viewport.X
            local targetY = openerStartGui.Y.Scale + delta.Y / viewport.Y
            targetX = math.clamp(targetX, 0, 1 - sizeX)
            targetY = math.clamp(targetY, 0, 1 - sizeY)
            OpenButton.Position = UDim2.fromScale(targetX, targetY)
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input == openerInput then
        openerInput = nil
        openerDragging = false
    end
end)

        Menu.Visible = true
        local oldSize = Menu.Size
        Menu.Size = UDim2.fromScale(oldSize.X.Scale * 0.94, oldSize.Y.Scale * 0.94)
        Menu.BackgroundTransparency = 0.1
        tween(Menu, 0.18, {
            Size = oldSize,
            BackgroundTransparency = 0,
        })
        menuVisible = true
    else
        local oldSize = Menu.Size
        local out = tween(Menu, 0.15, {
            Size = UDim2.fromScale(oldSize.X.Scale * 0.94, oldSize.Y.Scale * 0.94),
            BackgroundTransparency = 0.1,
        })
        out.Completed:Connect(function()
            Menu.Visible = false
            Menu.Size = oldSize
            Menu.BackgroundTransparency = 0
        end)
        menuVisible = false
    end
    task.delay(0.20, function()
        menuAnimating = false
    end)
end

OpenButton.Activated:Connect(function()
    if openerWasDragged then
        openerWasDragged = false
        return
    end
    if menuVisible then
        setMenuVisible(false)
    else
        setMenuVisible(true)
    end
end)

    setMenuVisible(false)
end)

Close.Activated:Connect(function()
    if rootFolder and rootFolder.Parent then
        rootFolder:Destroy()
    end
end)

local camera = workspace.CurrentCamera
if camera then
    camera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
        local viewport = camera.ViewportSize
        if viewport.X <= 0 or viewport.Y <= 0 then
            return
        end

        for _, gui in ipairs({OpenButton}) do
            if gui and gui.Parent then
                setPhysicalSquareSize(gui, 0.065)
            end
        end

        local function clampGui(gui)
            if not gui or not gui.Parent then
                return
            end
            local sizeX = gui.Size.X.Scale
            local sizeY = gui.Size.Y.Scale
            local anchor = gui.AnchorPoint
            local x = gui.Position.X.Scale
            local y = gui.Position.Y.Scale
            x = math.clamp(x, anchor.X * sizeX, 1 - (1 - anchor.X) * sizeX)
            y = math.clamp(y, anchor.Y * sizeY, 1 - (1 - anchor.Y) * sizeY)
            gui.Position = UDim2.fromScale(x, y)
        end

        clampGui(OpenButton)
        clampGui(Menu)
    end)
end
