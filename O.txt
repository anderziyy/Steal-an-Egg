local function protectGui(gui)
	pcall(function()
		gui.ZIndexBehavior = Enum.ZIndexBehavior.Global
	end)
	pcall(function()
		gui.DisplayOrder = 999999
	end)
	pcall(function()
		gui.ResetOnSpawn = false
	end)
	pcall(function()
		gui.AutoLocalize = false
	end)
	pcall(function()
		gui.IgnoreGuiInset = true
	end)
	pcall(function()
		gui.ScreenInsets = Enum.ScreenInsets.None
	end)
	pcall(function()
		gui.SafeAreaCompatibility = Enum.SafeAreaCompatibility.FullscreenExtension
	end)
end

local function setNoLocalization(instance)
	pcall(function()
		instance.AutoLocalize = false
	end)
end

local function setText(instance, text)
	instance.Text = text
	setNoLocalization(instance)
end

local function create(instanceClass, props, parent)
	local object = Instance.new(instanceClass)
	for key, value in pairs(props or {}) do
		object[key] = value
	end
	if parent then
		object.Parent = parent
	end
	return object
end

local function createStroke(parent, color, thickness, transparency)
	return create("UIStroke", {
		Color = color or Theme.Border,
		Thickness = thickness or 1,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

local function createAspect(parent, ratio, mode)
	return create("UIAspectRatioConstraint", {
		AspectRatio = ratio,
		AspectType = mode or Enum.AspectType.FitWithinMaxSize,
		DominantAxis = Enum.DominantAxis.Width,
	}, parent)
end

local function viewport()
	local camera = workspace.CurrentCamera
	if camera then
		return camera.ViewportSize
	end
	return Vector2.new(1280, 720)
end

local function scaleFromPixels(pxX, pxY)
	local view = viewport()
	local x = 0
	local y = 0
	if view.X > 0 then
		x = pxX / view.X
	end
	if view.Y > 0 then
		y = pxY / view.Y
	end
	return fromScale(x, y)
end

local function getGuiPositionScale(guiObject)
	return Vector2.new(guiObject.Position.X.Scale, guiObject.Position.Y.Scale)
end

local function setGuiPositionScale(guiObject, x, y)
	guiObject.Position = UDim2.new(x, 0, y, 0)
end

local function clamp(v, low, high)
	return math.max(low, math.min(high, v))
end

local function tween(instance, info, goal)
	local created = TweenService:Create(instance, info, goal)
	created:Play()
	return created
end

local function tweenFast(instance, goal)
	return tween(instance, TweenInfo.new(0.16, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), goal)
end

local function tweenSoft(instance, goal)
	return tween(instance, TweenInfo.new(0.28, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), goal)
end

local function addHover(button, baseColor, hoverColor)
	button.MouseEnter:Connect(function()
		if button.Active then
			tweenFast(button, {BackgroundColor3 = hoverColor})
		end
	end)
	button.MouseLeave:Connect(function()
		tweenFast(button, {BackgroundColor3 = baseColor})
	end)
end

local function setTabVisual(tabKey)
	for key, entry in pairs(tabButtons) do
		local selected = key == tabKey
		entry.Button.BackgroundColor3 = selected and Theme.PanelStrong or Theme.PanelAlt
		entry.Label.TextColor3 = selected and Theme.White or Theme.WhiteSoft
		entry.Icon.ImageColor3 = selected and Theme.White or Theme.Gray
	end
end

		local contentHeight = TabsLayout.AbsoluteContentSize.Y

			TabsViewport.CanvasSize = fromScale(0, math.max(1, contentHeight / visibleHeight))
		else
			TabsViewport.CanvasSize = fromScale(0, 1)
		end
	end)
end

local function performAction(action)
	if action == "copy_discord" then
		local copied = false
		local clipboardMethods = {
			function() return setclipboard(DISCORD_LINK) end,
			function() return toclipboard(DISCORD_LINK) end,
			function() return set_clipboard(DISCORD_LINK) end,
		}
		for _, method in ipairs(clipboardMethods) do
			local ok = pcall(method)
			if ok then
				copied = true
				break
			end
		end
		if copied then
			showNotification(DISCORD_NOTIFY_ICON, "Discord invite copied to clipboard.")
		else
			showNotification(DISCORD_NOTIFY_ICON, "Clipboard access is unavailable in this environment.")
		end
	elseif action == "copy_server" then
		local value = tostring(game.JobId or "")
		local copied = false
		local clipboardMethods = {
			function() return setclipboard(value) end,
			function() return toclipboard(value) end,
			function() return set_clipboard(value) end,
		}
		for _, method in ipairs(clipboardMethods) do
			local ok = pcall(method)
			if ok then
				copied = true
				break
			end
		end
		if copied then
			showNotification(SERVER_ICON, "Current server ID copied.")
		else
			showNotification(SERVER_ICON, "Clipboard access is unavailable in this environment.")
		end
	elseif action == "copy_game" then
		local value = tostring(game.PlaceId or "")
		local copied = false
		local clipboardMethods = {
			function() return setclipboard(value) end,
			function() return toclipboard(value) end,
			function() return set_clipboard(value) end,
		}
		for _, method in ipairs(clipboardMethods) do
			local ok = pcall(method)
			if ok then
				copied = true
				break
			end
		end
		if copied then
			showNotification(CLICK_ICON, "Game ID copied.")
		else
			showNotification(CLICK_ICON, "Clipboard access is unavailable in this environment.")
		end
	elseif action == "random_rejoin" then
		hopToRandomServer()
	elseif action == "same_rejoin" then
		rejoinSameServer()
	end
end

function renderActiveTab()
	if state.ModalOpen then
		closeConfirm()
	end
	removeDynamicChildren()
	EmptyState.Visible = false
	local query = string.lower(state.SearchQuery or "")
	local order = 1
	local isSearching = query ~= ""
	local matches = {}

	if isSearching then
		for _, tab in ipairs(tabs) do
			local info = data[tab.Key]
			if info then
				for _, option in ipairs(info.Options or {}) do
					local hay = string.lower((option.Title or "") .. " " .. (option.Description or ""))
					if string.find(hay, query, 1, true) then
						table.insert(matches, option)
					end
				end
			end
		end
		if #matches == 0 then
			local noResult = create("Frame", {
				Name = "NoSearchResults",
				BackgroundColor3 = Theme.Panel,
				BorderSizePixel = 0,
				Size = fromScale(0.96, 0.32),
				LayoutOrder = order,
				ZIndex = 212,
			}, ContentViewport)
			order = order + 1
			createStroke(noResult, Theme.BorderSoft, 1, 0.15)
			createScaleText(noResult, "No matching options", {
				Size = fromScale(0.8, 0.28),
				Position = fromScale(0.1, 0.25),
				TextXAlignment = Enum.TextXAlignment.Center,
				Font = Enum.Font.GothamBold,
				TextColor3 = Theme.White,
				ZIndex = 213,
			})
			createScaleText(noResult, "Search results include options from every tab.", {
				Size = fromScale(0.82, 0.24),
				Position = fromScale(0.09, 0.54),
				TextXAlignment = Enum.TextXAlignment.Center,
				Font = Enum.Font.Gotham,
				TextColor3 = Theme.Gray,
				ZIndex = 213,
			})
		else
			for _, option in ipairs(matches) do
				createOptionRow(option, order)
				order = order + 1
			end
		end
		updateContentCanvas(#matches + 1 > 6)
		return
	end

	local info = data[state.ActiveTab]
	if not info then
		updateContentCanvas(false)
		return
	end

	if #info.Categories == 0 and #info.Options == 0 then
		EmptyState.Visible = true
		EmptyState.Parent = ContentViewport
		EmptyState.LayoutOrder = 1
		updateContentCanvas(false)
		return
	end

	local optionsByCategory = {}
	for _, option in ipairs(info.Options) do
		if option.Category then
			optionsByCategory[option.Category] = optionsByCategory[option.Category] or {}
			table.insert(optionsByCategory[option.Category], option)
		end
	end

	local categoryIds = {}
	for _, category in ipairs(info.Categories) do
		categoryIds[category.Id] = true
		createCategoryRow(category, order)
		order = order + 1
		if state.CategoryOpen[category.Id] then
			for _, option in ipairs(optionsByCategory[category.Id] or {}) do
				createOptionRow(option, order)
				order = order + 1
			end
		end
	end
	for _, option in ipairs(info.Options) do
		if not option.Category or not categoryIds[option.Category] then
			createOptionRow(option, order)
			order = order + 1
		end
	end
	local count = getOptionCount(state.ActiveTab)
	updateContentCanvas(count > 6)
end

local function calculateResponsiveSizes()
	local view = viewport()
	if view.X <= 0 or view.Y <= 0 then
		return
	end
	local buttonPx = math.clamp(math.min(view.X, view.Y) * 0.115, 62, 86)
	local buttonScaleX = buttonPx / view.X
	local buttonScaleY = buttonPx / view.Y
	OpenButton.Size = fromScale(buttonScaleX, buttonScaleY)
	ButtonGlow.Size = fromScale(buttonScaleX * 1.14, buttonScaleY * 1.14)
	ButtonGlow.Position = fromScale(-buttonScaleX * 0.07, -buttonScaleY * 0.07)

	local menuPxW = math.clamp(math.min(view.X * 0.78, 620), 315, 620)
	local menuPxH = math.clamp(math.min(view.Y * 0.68, 430), 265, 430)
	if view.X < 430 then
		menuPxW = view.X * 0.92
	end
	if view.Y < 520 then
		menuPxH = view.Y * 0.77
	end
	menuWidthScale = clamp(menuPxW / view.X, 0.5, 0.94)
	menuHeightScale = clamp(menuPxH / view.Y, 0.44, 0.84)
	if menuWidthScale > 0.9 then
		menuWidthScale = 0.9
	end
	if menuHeightScale > 0.86 then
		menuHeightScale = 0.86
	end
	local menuX = clamp(state.LastMenuPosition.X, 0.02, 0.98 - menuWidthScale)
	local menuY = clamp(state.LastMenuPosition.Y, 0.02, 0.98 - menuHeightScale)
	if not state.MenuDragging then
		state.LastMenuPosition = Vector2.new(menuX, menuY)
	end
	Menu.Size = fromScale(menuWidthScale, menuHeightScale)
	if not state.MenuDragging then
		Menu.Position = fromScale(menuX, menuY)
	end
	if not state.MenuOpen then
		state.LastMenuPosition = Vector2.new(menuX, menuY)
	end
	Notifications.Size = fromScale(0.31, 0.43)
	Notifications.Position = fromScale(0.675, 0.545)
	if view.X < 600 then
		Notifications.Size = fromScale(0.48, 0.43)
		Notifications.Position = fromScale(0.49, 0.545)
	end
end

local function keepButtonInBounds(animate)
	local view = viewport()
	local abs = OpenButton.AbsoluteSize
	if view.X <= 0 or view.Y <= 0 then
		return
	end
	local current = getGuiPositionScale(OpenButton)
	local maxX = 1 - abs.X / view.X
	local maxY = 1 - abs.Y / view.Y
	local x = clamp(current.X, 0.005, math.max(0.005, maxX - 0.005))
	local y = clamp(current.Y, 0.005, math.max(0.005, maxY - 0.005))
	if x ~= current.X or y ~= current.Y then
		state.LastButtonPosition = Vector2.new(x, y)
		if animate then
			tweenFast(OpenButton, {Position = fromScale(x, y)})
		else
			setGuiPositionScale(OpenButton, x, y)
		end
	end
end

local function getMenuBounds()
	local view = viewport()
	local absPos = Menu.AbsolutePosition
	local absSize = Menu.AbsoluteSize
	local left = absPos.X
	local top = absPos.Y
	local right = absPos.X + absSize.X
	local bottom = absPos.Y + absSize.Y
	return left, top, right, bottom, absSize, view
end

local function clampMenuPosition(x, y)
	local view = viewport()
	local w = Menu.AbsoluteSize.X
	local h = Menu.AbsoluteSize.Y
	local minX = -w / view.X + 0.006
	local maxX = 1 - 0.006
	local minY = -h / view.Y + 0.006
	local maxY = 1 - 0.006
	return clamp(x, minX, maxX), clamp(y, minY, maxY)
end

local function recoverMenuIfFullyOffscreen()
	local left, top, right, bottom, absSize, view = getMenuBounds()
	if view.X <= 0 or view.Y <= 0 then
		return
	end
	local fullyLeft = right <= 0
	local fullyRight = left >= view.X
	local fullyTop = bottom <= 0
	local fullyBottom = top >= view.Y
	if fullyLeft or fullyRight or fullyTop or fullyBottom then
		local currentY = state.LastMenuPosition.Y
		local targetX
		if fullyRight then
			targetX = 1 - absSize.X / view.X - 0.02
		elseif fullyLeft then
			targetX = 0.02
		else
			targetX = clamp(state.LastMenuPosition.X, 0.02, 0.98 - absSize.X / view.X)
		end
		local targetY = currentY
		if fullyTop then
			targetY = 0.02
		elseif fullyBottom then
			targetY = 0.98 - absSize.Y / view.Y
		else
			targetY = clamp(targetY, 0.02, 0.98 - absSize.Y / view.Y)
		end
		targetX, targetY = clampMenuPosition(targetX, targetY)
		state.LastMenuPosition = Vector2.new(targetX, targetY)
		tweenSoft(Menu, {Position = fromScale(targetX, targetY)})
	end
end

local dragConnections = {}
local dragSinkName = "PhantomHubDragSink"

local function bindDragSink()
	pcall(function()
		ContextActionService:BindActionAtPriority(dragSinkName, function()
			return Enum.ContextActionResult.Sink
		end, false, Enum.ContextActionPriority.High.Value, Enum.UserInputType.MouseButton1, Enum.UserInputType.MouseMovement, Enum.UserInputType.Touch)
	end)
end

local function unbindDragSink()
	pcall(function()
		ContextActionService:UnbindAction(dragSinkName)
	end)
end

local function startButtonDrag(input)
	state.ButtonDragging = true
	state.ButtonDragMoved = false
	bindDragSink()
	local startPos = getGuiPositionScale(OpenButton)
	local startInput = input.Position
	local activeInput = input
	local connection
	connection = UserInputService.InputChanged:Connect(function(changed)
		if not state.ButtonDragging then
			return
		end
		if activeInput.UserInputType == Enum.UserInputType.Touch and changed.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		if activeInput.UserInputType == Enum.UserInputType.MouseButton1 and changed.UserInputType ~= Enum.UserInputType.MouseMovement then
			return
		end
		local delta = changed.Position - startInput
		if delta.Magnitude > 5 then
			state.ButtonDragMoved = true
		end
		local view = viewport()
		if view.X <= 0 or view.Y <= 0 then
			return
		end
		local nx = startPos.X + delta.X / view.X
		local ny = startPos.Y + delta.Y / view.Y
		local abs = OpenButton.AbsoluteSize
		local maxX = 1 - abs.X / view.X
		local maxY = 1 - abs.Y / view.Y
		nx = clamp(nx, -abs.X / view.X + 0.003, 1 - 0.003)
		ny = clamp(ny, -abs.Y / view.Y + 0.003, 1 - 0.003)
		OpenButton.Position = fromScale(nx, ny)
		state.LastButtonPosition = Vector2.new(nx, ny)
	end)
	table.insert(dragConnections, connection)
end

local function stopButtonDrag()
	if not state.ButtonDragging then
		return
	end
	state.ButtonDragging = false
	unbindDragSink()
	for i = #dragConnections, 1, -1 do
		local connection = dragConnections[i]
		if connection then
			connection:Disconnect()
		end
		table.remove(dragConnections, i)
	end
	local view = viewport()
	local abs = OpenButton.AbsoluteSize
	local pos = getGuiPositionScale(OpenButton)
	local x = pos.X
	local y = pos.Y
	local fullyLeft = (x + abs.X / view.X) <= 0
	local fullyRight = x >= 1
	local fullyTop = (y + abs.Y / view.Y) <= 0
	local fullyBottom = y >= 1
	if fullyLeft or fullyRight or fullyTop or fullyBottom then
		if fullyRight then
			x = 1 - abs.X / view.X - 0.02
		elseif fullyLeft then
			x = 0.02
		else
			x = clamp(x, 0.02, 0.98 - abs.X / view.X)
		end
		if fullyTop then
			y = 0.03
		elseif fullyBottom then
			y = 0.9 - abs.Y / view.Y
		else
			y = clamp(y, 0.02, 0.98 - abs.Y / view.Y)
		end
		state.LastButtonPosition = Vector2.new(x, y)
		tweenSoft(OpenButton, {Position = fromScale(x, y)})
	else

local function startMenuDrag(input)
	if state.ModalOpen then
		return
	end
	state.MenuDragging = true
	state.MenuDragMoved = false
	bindDragSink()
	local startPos = getGuiPositionScale(Menu)
	state.MenuDragOriginalY = startPos.Y
	local startInput = input.Position
	local activeInput = input
	local connection
	connection = UserInputService.InputChanged:Connect(function(changed)
		if not state.MenuDragging then
			return
		end
		if activeInput.UserInputType == Enum.UserInputType.Touch and changed.UserInputType ~= Enum.UserInputType.Touch then
			return
		end
		if activeInput.UserInputType == Enum.UserInputType.MouseButton1 and changed.UserInputType ~= Enum.UserInputType.MouseMovement then
			return
		end
		local delta = changed.Position - startInput
		if delta.Magnitude > 5 then
			state.MenuDragMoved = true
		end
		local view = viewport()
		if view.X <= 0 or view.Y <= 0 then
			return
		end
		local nx = startPos.X + delta.X / view.X
		local ny = startPos.Y + delta.Y / view.Y
		Menu.Position = fromScale(nx, ny)
		state.LastMenuPosition = Vector2.new(nx, ny)
	end)
	table.insert(dragConnections, connection)
end

local function requestFunction()
	local candidates = {
		(rawget(_G, "request")),
		(rawget(_G, "http_request")),
		(rawget(_G, "requestAsync")),
		(rawget(_G, "syn") and rawget(_G, "syn").request),
		(rawget(_G, "fluxus") and rawget(_G, "fluxus").request),
	}
	for _, candidate in ipairs(candidates) do
		if type(candidate) == "function" then
			return candidate
		end
	end
	return nil
end

local function getPublicServers(placeId)
	local results = {}
	local cursor = ""
	local req = requestFunction()
	for page = 1, 6 do
		local url = "https://games.roblox.com/v1/games/" .. tostring(placeId) .. "/servers/Public?sortOrder=Asc&limit=100"
		if cursor ~= "" then
			url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
		end
		local body = nil
		if req then
			local ok, response = pcall(req, {Url = url, Method = "GET", Headers = {['User-Agent'] = "Roblox"}})
			if ok and type(response) == "table" then
				body = response.Body or response.body
			end
		end
		if not body then
			local ok, response = pcall(function()
				return game:HttpGet(url)
			end)
			if ok then
				body = response
			end
		end
		if type(body) ~= "string" or body == "" then
			break
		end
		local ok, decoded = pcall(function()
			return HttpService:JSONDecode(body)
		end)
		if not ok or type(decoded) ~= "table" then
			break
		end
		for _, server in ipairs(decoded.data or {}) do
			if type(server) == "table" and server.id and tonumber(server.playing or 0) < tonumber(server.maxPlayers or 0) then
				if server.id ~= game.JobId then
					table.insert(results, server.id)
				end
			end
		end
		cursor = decoded.nextPageCursor or ""
		if cursor == "" then
			break
		end
	end
	return results
end

function hopToRandomServer()
	if state.ServerHopBusy then
		return
	end
	state.ServerHopBusy = true
	showNotification(SERVER_ICON, "Searching for an available public server...")
	task.spawn(function()
		local servers = getPublicServers(game.PlaceId)
		if #servers > 0 then
			local selected = servers[math.random(1, #servers)]
			local ok, err = pcall(function()
				TeleportService:TeleportToPlaceInstance(game.PlaceId, selected, LocalPlayer)
			end)
			if not ok then
				showNotification(SERVER_ICON, "Server teleport failed: " .. tostring(err))
			end
		else
			local ok, err = pcall(function()
				TeleportService:Teleport(game.PlaceId, LocalPlayer)
			end)
			if not ok then
				showNotification(SERVER_ICON, "Random server fallback failed: " .. tostring(err))
			end
		end
		state.ServerHopBusy = false
	end)
end

	end)
end

local antiAfkThreadToken = 0
local function refreshCharacterConnections()
	local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	local humanoid = character:FindFirstChildOfClass("Humanoid") or character:WaitForChild("Humanoid")
	return character, humanoid
end

local function startAntiAfkLoop()
	antiAfkThreadToken = antiAfkThreadToken + 1
	local token = antiAfkThreadToken
	task.spawn(function()
		while state.Toggles.anti_afk and token == antiAfkThreadToken do
			task.wait(60)
			if not state.Toggles.anti_afk or token ~= antiAfkThreadToken then
				break
			end
			local character = LocalPlayer.Character
			local humanoid = character and character:FindFirstChildOfClass("Humanoid")
			if humanoid and humanoid.Health > 0 then
				pcall(function()
					humanoid:Move(Vector3.new(0, 0, -1), true)
				end)
				task.wait(0.18)
				pcall(function()
					humanoid:Move(Vector3.zero, true)
				end)
			end
		end
	end)
end

		end
		if humanoid.Jump then
			pcall(function()
				humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
			end)
		end
	end)
end
