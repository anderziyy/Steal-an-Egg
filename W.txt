local function tween(obj, info, props)
    local ok, t = pcall(function()
        return TweenService:Create(obj, info, props)
    end)

local function keepFullyOnScreen(gui, anchorSide)
    local camera = workspace.CurrentCamera
    if not camera then return end
    local view = camera.ViewportSize
    local abs = gui.AbsoluteSize
    local pos = gui.AbsolutePosition
    local x = pos.X
    local y = pos.Y
    local margin = math.max(6, math.floor(math.min(view.X, view.Y) * 0.012))
    local fullyVisible = x >= -abs.X + margin and x <= view.X - margin and y >= margin and y <= view.Y - abs.Y - margin
    if fullyVisible then return end
    local right = x > (view.X - abs.X) / 2
    local targetX = right and (view.X - abs.X - margin) or margin
    local targetY = math.clamp(y, margin, view.Y - abs.Y - margin)
    local newScaleX = targetX / math.max(1, view.X)
    local newScaleY = targetY / math.max(1, view.Y)
    tween(gui, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Position = UDim2.fromScale(math.clamp(newScaleX, 0, 1), math.clamp(newScaleY, 0, 1))
    })
end

local function beginScaleDrag(gui, input)
    local startPos = gui.Position
    local startInput = input.Position
    local dragging = true
    local moveConn
    local endConn
    moveConn = UserInputService.InputChanged:Connect(function(changed)
        if not dragging then return end
        if changed.UserInputType ~= Enum.UserInputType.MouseMovement and changed.UserInputType ~= Enum.UserInputType.Touch then return end
        local delta = changed.Position - startInput
        local camera = workspace.CurrentCamera
        if not camera then return end
        local vp = camera.ViewportSize
        local dx = delta.X / math.max(1, vp.X)
        local dy = delta.Y / math.max(1, vp.Y)
        gui.Position = UDim2.fromScale(startPos.X.Scale + dx, startPos.Y.Scale + dy)
    end)
    endConn = input.Changed:Connect(function()
        if input.UserInputState == Enum.UserInputState.End or input.UserInputState == Enum.UserInputState.Cancel then
            dragging = false
            moveConn:Disconnect()
            endConn:Disconnect()
            keepFullyOnScreen(gui)
        end
    end)
end

local openDragConnection = nil
OpenButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        local dragStart = input.Position
        local moved = false
        local started = false
        local moveConn
        local endConn
        moveConn = UserInputService.InputChanged:Connect(function(changed)
            if changed.UserInputType ~= Enum.UserInputType.MouseMovement and changed.UserInputType ~= Enum.UserInputType.Touch then return end
            local delta = changed.Position - dragStart
            if delta.Magnitude > 7 and not started then
                started = true
                moved = true
                beginScaleDrag(OpenButton, input)
                if moveConn then moveConn:Disconnect() end
            end
        end)
        endConn = input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End or input.UserInputState == Enum.UserInputState.Cancel then
                if moveConn then moveConn:Disconnect() end
                if endConn then endConn:Disconnect() end
                if not moved then
                    OpenButton:SetAttribute("Tap", not OpenButton:GetAttribute("Tap"))
                end
            end
        end)
    end
end)

function refreshOptions()
    for _, child in ipairs(OptionsScroll:GetChildren()) do
        if child:IsA("Frame") and child ~= OptionsLayout then child:Destroy() end
    end
    optionWidgets = {}
    local query = searchTerm:lower()
    local shownAny = false
    local list = currentMainOptions
    if query ~= "" then
        list = {}
        for _, tab in ipairs(tabs) do
            for _, item in ipairs(settingsData[tab.name] or {}) do
                if item.kind == "category" then
                    for _, sub in ipairs(item.options or {}) do
                        local hay = (sub.title .. " " .. sub.description):lower()
                        if hay:find(query, 1, true) then
                            table.insert(list, sub)
                        end
                    end
                elseif item.kind ~= "category" then
                    local hay = (item.title .. " " .. item.description):lower()
                    if hay:find(query, 1, true) then
                        table.insert(list, item)
                    end
                end
            end
        end
        for i, opt in ipairs(list) do
            opt.order = i
            addOptionWidget(OptionsScroll, opt)
            shownAny = true
        end
    else
        for _, item in ipairs(list) do
            if item.kind == "category" then
                addOptionWidget(OptionsScroll, {id = item.id .. "__categorySpacer", title = "", description = "", kind = "visual", order = item.order, callback = function() end})
                local spacer = OptionsScroll:FindFirstChild(item.id .. "__categorySpacer")
                if spacer then spacer:Destroy() end
                local categoryFrame = createCategory(OptionsScroll, item)
                if categoryOpen[item.id] then
                    for _, sub in ipairs(item.options or {}) do
                        sub.order = (item.order * 100) + (_)
                        addOptionWidget(OptionsScroll, sub)
                    end
                    shownAny = true
                end
            else
                item.order = item.order or 1
                addOptionWidget(OptionsScroll, item)
                shownAny = true
            end
        end
    end
    OptionsScroll.CanvasPosition = Vector2.new(0, 0)
end

    local funcs = {setclipboard, toclipboard, set_clipboard, copyclipboard}
    for _, fn in ipairs(funcs) do
        if type(fn) == "function" then
            local ok = pcall(fn, text)
            if ok then
                copied = true
                break
            end
        end
    end
    if not copied then
        pcall(function()
            if typeof(setclipboard) == "function" then setclipboard(text) copied = true end
        end)
    end
    return copied
end

local function applyLowMode()
    lowModeState = {}
    for _, inst in ipairs(game:GetDescendants()) do
        if inst:IsA("BasePart") then
            lowModeState[inst] = {Material = inst.Material, Reflectance = inst.Reflectance, CastShadow = inst.CastShadow}
            inst.Material = Enum.Material.SmoothPlastic
            inst.Reflectance = 0
            inst.CastShadow = false
        elseif inst:IsA("Decal") or inst:IsA("Texture") then
            lowModeState[inst] = {Transparency = inst.Transparency}
            inst.Transparency = 1
        elseif inst:IsA("ParticleEmitter") or inst:IsA("Trail") or inst:IsA("Smoke") or inst:IsA("Fire") or inst:IsA("Sparkles") then
            lowModeState[inst] = {Enabled = inst.Enabled}
            inst.Enabled = false
        elseif inst:IsA("PostEffect") then
            lowModeState[inst] = {Enabled = inst.Enabled}
            inst.Enabled = false
        elseif inst:IsA("SurfaceAppearance") then
            lowModeState[inst] = {Parent = inst.Parent}
            inst.Parent = nil
        end
    end
end

local function restoreLowMode()
    for inst, state in pairs(lowModeState) do
        if inst and inst.Parent ~= nil then
            for property, value in pairs(state) do
                pcall(function() inst[property] = value end)
            end
        end
    end
    lowModeState = {}
end

local antiAfkThread
local jumpConn
local jumpHold = false

local function setAntiAfk(state)
    antiAfk = state
    if antiAfkThread then
        antiAfk = false
        antiAfkThread = nil
    end
    if state then
        antiAfkThread = task.spawn(function()
            while antiAfk do
                task.wait(60)
                if not antiAfk then break end
                local char = LocalPlayer.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                if hum then
                    local root = char:FindFirstChild("HumanoidRootPart")
                    if root then
                        local cf = root.CFrame
                        hum:Move(Vector3.new(0, 0, -1), true)
                        task.wait(0.22)
                        hum:Move(Vector3.zero, true)
                        pcall(function() root.CFrame = cf end)
                    end
                end
            end
        end)
    end
end

local function setInfiniteJump(state)
    infiniteJump = state
    if jumpConn then jumpConn:Disconnect() jumpConn = nil end
    if state then
        jumpConn = UserInputService.JumpRequest:Connect(function()
            local char = LocalPlayer.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum:ChangeState(Enum.HumanoidStateType.Jumping)
            end
        end)
    end
end
