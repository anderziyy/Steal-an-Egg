local function new(className, props, parent)
    local obj = Instance.new(className)
    for key, value in pairs(props or {}) do
        obj[key] = value
    end
    if parent then
        obj.Parent = parent
    end
    return obj
end

local function tween(instance, time, properties)
    local info = TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    TweenService:Create(instance, info, properties):Play()
end

local function setButtonVisual(btn, selected)
    btn.BackgroundColor3 = selected and Color3.fromRGB(48, 48, 48) or Color3.fromRGB(26, 26, 26)

    local label = btn:FindFirstChild("Label")
    if icon then

    end
    if label then
        label.TextColor3 = selected and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(175, 175, 175)
    end
end

local function refreshFilter()
    local query = string.lower(searchBox.Text)
    placeholder.Visible = query == ""
    clearSearch.Visible = query ~= ""
    for _, row in ipairs(optionRows) do
        local match = query == "" or string.find(string.lower(row.Name), query, 1, true) ~= nil
        row.Visible = match
    end
end

searchBox:GetPropertyChangedSignal("Text"):Connect(refreshFilter)

    searchBox.Text = ""
    searchBox:CaptureFocus()
end)

local draggingButton = false
local buttonDragInput
local buttonDragStart
local buttonStartPos
local buttonMoved = false
local savedButtonPosition = button.Position

local function viewportInfo()
    local camera = workspace.CurrentCamera
    local size = camera and camera.ViewportSize or Vector2.new(1280, 720)
    local inset = GuiService:GetGuiInset()
    return size, inset
end

local function updateScaleSizing()
    local size = select(1, viewportInfo())
    local minAxis = math.max(math.min(size.X, size.Y), 1)
    local buttonPixels = math.clamp(64, 48, 80)
    local buttonUnitX = buttonPixels / math.max(size.X, 1)
    local buttonUnitY = buttonPixels / math.max(size.Y, 1)
    button.Size = UDim2.new(buttonUnitX, 0, buttonUnitY, 0)
end

updateScaleSizing()

local function clampScale(value, minValue, maxValue)
    return math.clamp(value, minValue, maxValue)
end

local function clampButtonPosition(position)
    local size = select(1, viewportInfo())
    local absSize = button.AbsoluteSize
    local halfW = absSize.X / 2 / math.max(size.X, 1)
    local halfH = absSize.Y / 2 / math.max(size.Y, 1)
    return UDim2.new(
        clampScale(position.X.Scale, halfW, 1 - halfW),
        0,
        clampScale(position.Y.Scale, halfH, 1 - halfH),
        0
    )
end

local function updateButtonDrag(input)
    local delta = input.Position - buttonDragStart
    local size = select(1, viewportInfo())
    local newX = buttonStartPos.X.Scale + delta.X / math.max(size.X, 1)
    local newY = buttonStartPos.Y.Scale + delta.Y / math.max(size.Y, 1)
    button.Position = UDim2.new(newX, 0, newY, 0)
    buttonMoved = math.abs(delta.X) + math.abs(delta.Y) > 4
end

button.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        draggingButton = true
        buttonMoved = false
        buttonDragStart = input.Position
        buttonStartPos = button.Position
        buttonDragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if draggingButton and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        buttonDragInput = input
        updateButtonDrag(input)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if draggingButton and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
        draggingButton = false
        buttonDragInput = nil
        if button.Position.X.Scale < -button.AbsoluteSize.X / math.max(select(1, viewportInfo()).X, 1) or button.Position.X.Scale > 1 + button.AbsoluteSize.X / math.max(select(1, viewportInfo()).X, 1) or button.Position.Y.Scale < -button.AbsoluteSize.Y / math.max(select(1, viewportInfo()).Y, 1) or button.Position.Y.Scale > 1 + button.AbsoluteSize.Y / math.max(select(1, viewportInfo()).Y, 1) then
            local current = button.Position
            local targetX = current.X.Scale < 0.5 and 0.06 or 0.94
            local targetY = clampScale(current.Y.Scale, 0.06, 0.94)
            button.Position = UDim2.new(targetX, 0, targetY, 0)
        else
            button.Position = clampButtonPosition(button.Position)
        end
        savedButtonPosition = button.Position
    end
end)

button.Activated:Connect(function()
    if buttonMoved then
        return
    end
    setMenuVisible(not menuOpen)
end)

local menuDragActive = false
local menuDragStart
local menuStartPos
local menuDragInput

local function menuIsCompletelyOutside()
    local pos = menu.AbsolutePosition
    local size = menu.AbsoluteSize
    local view = select(1, viewportInfo())
    return pos.X + size.X <= 0 or pos.Y + size.Y <= 0 or pos.X >= view.X or pos.Y >= view.Y
end

local function snapMenuIntoView()
    if not menuIsCompletelyOutside() then
        return
    end
    local pos = menu.AbsolutePosition
    local size = menu.AbsoluteSize
    local view = select(1, viewportInfo())
    local viewW = math.max(view.X, 1)
    local viewH = math.max(view.Y, 1)
    local x = menu.Position.X.Scale
    local y = menu.Position.Y.Scale
    if pos.X + size.X <= 0 then
        x = (size.X * 0.5) / viewW
    elseif pos.X >= view.X then
        x = 1 - (size.X * 0.5) / viewW
    end
    if pos.Y + size.Y <= 0 then
        y = (size.Y * 0.5) / viewH
    elseif pos.Y >= view.Y then
        y = 1 - (size.Y * 0.5) / viewH
    end
    menu.Position = UDim2.new(x, 0, y, 0)
end

local cornerSizeX = 0.08
local cornerSizeY = 0.15
local cornerPositions = {
    {Vector2.new(0, 0), UDim2.new(0, 0, 0, 0)},
    {Vector2.new(1, 0), UDim2.new(1, 0, 0, 0)},
    {Vector2.new(0, 1), UDim2.new(0, 1, 1, 0)},
    {Vector2.new(1, 1), UDim2.new(1, 0, 1, 0)},
}

for index, info in ipairs(cornerPositions) do
    local anchor = info[1]
    local pos = info[2]
    local handle = new("Frame", {
        Name = "CornerDrag_" .. index,
        AnchorPoint = anchor,
        Position = pos,
        Size = UDim2.new(cornerSizeX, 0, cornerSizeY, 0),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Active = true,
        ZIndex = 35,
    }, menu)

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            menuDragActive = true
            menuDragStart = input.Position
            menuStartPos = menu.Position
            menuDragInput = input
        end
    end)
end

UserInputService.InputChanged:Connect(function(input)
    if menuDragActive and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - menuDragStart
        local view = select(1, viewportInfo())
        local x = menuStartPos.X.Scale + delta.X / math.max(view.X, 1)
        local y = menuStartPos.Y.Scale + delta.Y / math.max(view.Y, 1)
        menu.Position = UDim2.new(x, 0, y, 0)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if menuDragActive and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
        menuDragActive = false
        menuDragInput = nil
        snapMenuIntoView()
    end
end)
