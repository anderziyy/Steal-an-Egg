local function make(className, properties, parent)
	local object = Instance.new(className)
	for key, value in pairs(properties or {}) do
		object[key] = value
	end
	object.Parent = parent
	return object
end

local function applyEnglishOnly(root)
	local function visit(obj)
		pcall(function()
			if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
				obj.AutoLocalize = false
			end
		end)
		for _, child in ipairs(obj:GetChildren()) do
			visit(child)
		end
	end
	visit(root)
end

local function tween(object, time, properties, style, direction)
	local info = TweenInfo.new(time or 0.18, style or Enum.EasingStyle.Quint, direction or Enum.EasingDirection.Out)
	local animation = TweenService:Create(object, info, properties)
	animation:Play()
	return animation
end

local cleanupCallbacks = {}
local cleanedUp = false
local pendingAction = nil

local function registerCleanup(callback)
	if type(callback) ~= "function" then
		return
	end
	table.insert(cleanupCallbacks, callback)
end

local function cleanupAll()
	if cleanedUp then
		return
	end
	cleanedUp = true

	for i = #cleanupCallbacks, 1, -1 do
		local callback = cleanupCallbacks[i]
		cleanupCallbacks[i] = nil
		pcall(callback)
	end
end

local scrollDragging = false
local scrollDragInput = nil
local scrollDragStartY = 0
local scrollStartCanvasY = 0
local scrollDidDrag = false

local function isPointInside(guiObject, point)
	local position = guiObject.AbsolutePosition
	local size = guiObject.AbsoluteSize
	return point.X >= position.X and point.X <= position.X + size.X
		and point.Y >= position.Y and point.Y <= position.Y + size.Y
end

local globalInputBeganConnection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if cleanedUp or not ScreenGui.Parent or not Menu.Visible then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		if isPointInside(ResultsScroll, input.Position) then
			scrollDragging = true
			scrollDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
			scrollDragStartY = input.Position.Y
			scrollStartCanvasY = ResultsScroll.CanvasPosition.Y
			scrollDidDrag = false
		end
	end
end)
registerCleanup(function()
	globalInputBeganConnection:Disconnect()
end)

end

local function getVisibleItems(tabName, query)
	local normalized = string.lower(query or "")

	if normalized == "" then
		return datasets[tabName] or {}
	end

	local filtered = {}
	local searchOrder = {"Main", "Auto Buy", "Farm", "Config", "Credits"}
	for _, datasetName in ipairs(searchOrder) do
		for _, item in ipairs(datasets[datasetName] or {}) do
			if item.Type == "Category" then
				for _, child in ipairs(item.Items or {}) do
					local title = string.lower(child.Title or "")
					local description = string.lower(child.Description or "")
					if string.find(title, normalized, 1, true) or string.find(description, normalized, 1, true) then
						table.insert(filtered, child)
					end
				end
			else
				local title = string.lower(item.Title or "")
				local description = string.lower(item.Description or "")
				if string.find(title, normalized, 1, true) or string.find(description, normalized, 1, true) then
					table.insert(filtered, item)
				end
			end
		end
	end

	return filtered
end

local toggleStates = {}
local toggleConnections = {}

local function getHumanoid()
	local character = LocalPlayer.Character
	return character and character:FindFirstChildOfClass("Humanoid") or nil
end

local function setToggleState(item, enabled)
	toggleStates[item] = enabled
	if item.Title ~= "Idle Movement" then return end

	if toggleConnections[item] then
		toggleConnections[item]:Disconnect()
		toggleConnections[item] = nil
	end

	if enabled then
		local running = true
		local connection = {Disconnect = function() running = false end}
		toggleConnections[item] = connection
		task.spawn(function()
			while running and not cleanedUp do
				task.wait(60)
				if not running or cleanedUp then break end
				local humanoid = getHumanoid()
				if humanoid and humanoid.Health > 0 then
					humanoid:Move(Vector3.new(1, 0, 0), true)
					task.wait(0.18)
					humanoid:Move(Vector3.zero, true)
				end
			end
		end)
	end
end

renderResults = function()
	local previousScrollY = ResultsScroll.CanvasPosition.Y
	clearResults()
	local query = SearchBox.Text
	local items = getVisibleItems(currentTab, query)
	if #items == 0 then
		if query ~= "" then
			showEmptyState("No matching options found.")
		elseif currentTab == "Config" then
			showEmptyState("No options are available in Config.")
		elseif currentTab == "Credits" then
			showEmptyState("No options are available in Credits.")
		else
			showEmptyState("This tab has no options yet.")
		end
		setResultsScrollPositionY(0)
		return
	end

	local layoutIndex = 0
	if query ~= "" then
		for _, item in ipairs(items) do
			layoutIndex += 1
			createStandardRow(item, layoutIndex)
		end
		task.defer(function()
			setResultsScrollPositionY(previousScrollY)
		end)
		return
	end

	for _, item in ipairs(items) do
		layoutIndex += 1
		if item.Type == "Category" then
			createCategoryRow(item, layoutIndex, query)
			if item.Expanded then
				layoutIndex += #(item.Items or {})
			end
		else
			createStandardRow(item, layoutIndex)
		end
	end

	task.defer(function()
		setResultsScrollPositionY(previousScrollY)
	end)
end

		end
	end

	renderResults()
end

local launcherDragging = false
local launcherDragInput = nil
local launcherDragStart = nil
local launcherStartPos = nil
local launcherDidDrag = false

local function clampLauncher(position)
	local viewport = getViewport()
	local x = math.clamp(position.X.Scale, 0.0, 0.925)
	local y = math.clamp(position.Y.Scale, 0.02, 0.92)

	if position.X.Scale < 0 then
		x = 0.015
	elseif position.X.Scale > 0.925 then
		x = 0.907
	end

	return UDim2.fromScale(x, y)
end

local function updateLauncherDrag(input)
	local delta = input.Position - launcherDragStart
	if delta.Magnitude > 4 then
		launcherDidDrag = true
	end
	local viewport = getViewport()
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end
	local dx = delta.X / viewport.X
	local dy = delta.Y / viewport.Y
	local target = UDim2.fromScale(
		launcherStartPos.X.Scale + dx,
		launcherStartPos.Y.Scale + dy
	)
	LauncherFrame.Position = clampLauncher(target)
end

local function beginLauncherDrag(input)
	launcherDragging = true
	launcherDidDrag = false
	launcherDragStart = input.Position
	launcherStartPos = LauncherFrame.Position
	launcherDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
end

local function endLauncherDrag()
	launcherDragging = false
	launcherDragInput = nil
	launcherDragStart = nil
	launcherStartPos = nil
end

Launcher.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		beginLauncherDrag(input)
	end
end)

function updateMenuDrag(input)
	if not menuDragStart or not menuStartPos then
		return
	end
	local delta = input.Position - menuDragStart
	if delta.Magnitude < 4 then
		return
	end
	if os.clock() - pointerDownAt < 0.06 then
		return
	end
	didDrag = true
	menuDragging = true

	local viewport = getViewport()
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end
	local dx = delta.X / viewport.X
	local dy = delta.Y / viewport.Y
	local x = menuStartPos.X.Scale + dx
	local y = menuStartPos.Y.Scale + dy
	Menu.Position = UDim2.fromScale(x, y)
end

function endMenuDrag()
	if not menuDragging then
		menuDragInput = nil
		return
	end

	local viewport = getViewport()
	local absolutePosition = Menu.AbsolutePosition
	local absoluteSize = Menu.AbsoluteSize
	local fullyLeft = absolutePosition.X + absoluteSize.X < 0
	local fullyRight = absolutePosition.X > viewport.X
	local fullyTop = absolutePosition.Y + absoluteSize.Y < 0
	local fullyBottom = absolutePosition.Y > viewport.Y

	if fullyLeft then
		local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
		tween(Menu, 0.20, {Position = UDim2.fromScale(0.02, y)})
	elseif fullyRight then
		local y = math.clamp(Menu.Position.Y.Scale, 0.05, 0.95)
		tween(Menu, 0.20, {Position = UDim2.fromScale(0.98, y)})
	elseif fullyTop then
		tween(Menu, 0.20, {Position = UDim2.fromScale(math.clamp(Menu.Position.X.Scale, 0.02, 0.98), 0.05)})
	elseif fullyBottom then
		tween(Menu, 0.20, {Position = UDim2.fromScale(math.clamp(Menu.Position.X.Scale, 0.02, 0.98), 0.95)})
	end

	menuDragging = false
	menuDragInput = nil
	menuDragStart = nil
	menuStartPos = nil
	tween(MenuStroke, 0.12, {Color = COLORS.Gray3, Transparency = 0.12})
end

for name, corner in pairs(corners) do
	corner.Activated:Connect(function()
		if didDrag then
			didDrag = false
			return
		end
	end)
end

local inputChangedConnection = UserInputService.InputChanged:Connect(function(input)
	if launcherDragging and ((launcherDragInput and input == launcherDragInput) or (not launcherDragInput and input.UserInputType == Enum.UserInputType.MouseMovement)) then
		updateLauncherDrag(input)
	end
	if menuDragInput and input == menuDragInput then
		updateMenuDrag(input)
	elseif menuDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	elseif menuDragStart and not menuDragInput and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	end
	
	if scrollDragging and ((scrollDragInput and input == scrollDragInput) or (not scrollDragInput and input.UserInputType == Enum.UserInputType.MouseMovement)) then
		local deltaY = scrollDragStartY - input.Position.Y
		if math.abs(deltaY) > 5 then
			scrollDidDrag = true
		end
		ResultsScroll.CanvasPosition = Vector2.new(0, scrollStartCanvasY + deltaY)
	end
end)

local inputEndedConnection = UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		if launcherDragging then
			endLauncherDrag()
		end
		if menuDragging then
			endMenuDrag()
		end
		if scrollDragging then
			scrollDragging = false
			scrollDragInput = nil
		end
	end
end)

Launcher.Activated:Connect(function()
	if launcherDidDrag then
		launcherDidDrag = false
		return
	end
	toggleMenu()
end)

local function updateSearchVisual()
	local hasText = SearchBox.Text ~= ""
	SearchHint.Visible = not hasText
	SearchClear.Visible = hasText
end

local searchTextConnection = SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
	updateSearchVisual()
	renderResults()
end)

SearchClear.Activated:Connect(function()
	SearchBox.Text = ""
	SearchBox:CaptureFocus()
end)

applyEnglishOnly(ScreenGui)

setTab("Main")
Menu.Visible = false

local renderSteppedConnection = RunService.RenderStepped:Connect(function()
	if not ScreenGui.Parent then
		return
	end
	updateGlobalScale()
end)

registerCleanup(function()
	for _, connection in pairs(toggleConnections) do
		if connection and connection.Disconnect then connection:Disconnect() end
	end
end)

registerCleanup(function()
	for _, connection in pairs(toggleConnections) do
		pcall(function()
			connection:Disconnect()
		end)
	end
	toggleConnections = {}
	inputChangedConnection:Disconnect()
	inputEndedConnection:Disconnect()
	searchTextConnection:Disconnect()
	renderSteppedConnection:Disconnect()
end)
