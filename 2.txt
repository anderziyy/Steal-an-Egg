local function getParentContainer()
    local candidates = {}
    local okCore = pcall(function()
        table.insert(candidates, CoreGui)
    end)
    if not okCore then
        table.insert(candidates, LocalPlayer:WaitForChild("PlayerGui"))
    end

    for _, parent in ipairs(candidates) do
        local accepted = pcall(function()
            local probe = Instance.new("Folder")
            probe.Name = "__PhantomHubProbe"
            probe.Parent = parent
            probe:Destroy()
        end)
        if accepted then
            return parent
        end
    end

    return LocalPlayer:WaitForChild("PlayerGui")
end

local function clamp01(value)
    if value < 0 then
        return 0
    end
    if value > 1 then
        return 1
    end
    return value
end

local function smoothTween(instance, properties, duration, style, direction)
    local info = TweenInfo.new(
        duration or 0.16,
        style or Enum.EasingStyle.Quart,
        direction or Enum.EasingDirection.Out,
        0,

        0
    )
    local tween = TweenService:Create(instance, info, properties)
    tween:Play()
    return tween
end

local function getAbsoluteViewport()
    local camera = workspace.CurrentCamera
    if camera then
        local viewport = camera.ViewportSize
        if viewport.X > 0 and viewport.Y > 0 then
            return viewport
        end
    end
    return Vector2.new(1280, 720)
end

local function pixelToScaleDelta(parent, pixelDelta)
    local absolute = parent.AbsoluteSize
    local x = absolute.X > 0 and pixelDelta.X / absolute.X or 0
    local y = absolute.Y > 0 and pixelDelta.Y / absolute.Y or 0
    return x, y
end

local function scaleToPixel(parent, scaleX, scaleY)
    local absolute = parent.AbsoluteSize
    return Vector2.new(absolute.X * scaleX, absolute.Y * scaleY)
end

local function clampFrameScale(frame, desiredX, desiredY, minimumX, minimumY, maximumX, maximumY)
    local sizeX = frame.Size.X.Scale
    local sizeY = frame.Size.Y.Scale
    local maxX = maximumX
    local maxY = maximumY
    if maxX == nil then
        maxX = 1 - sizeX
    end
    if maxY == nil then
        maxY = 1 - sizeY
    end
    local x = math.clamp(desiredX, minimumX or 0, maxX)
    local y = math.clamp(desiredY, minimumY or 0, maxY)
    return x, y
end

local function attachScaleDrag(target, dragHandle, config)
    config = config or {}
    local active = false
    local dragging = false
    local activeInput = nil
    local dragStart = nil
    local startPosition = nil
    local moveConnection = nil
    local endConnection = nil
    local clickThreshold = config.ClickThreshold or 5

    local function finish()
        active = false
        dragging = false
        activeInput = nil
        dragStart = nil
        startPosition = nil
        if moveConnection then
            moveConnection:Disconnect()
            moveConnection = nil
        end
        if endConnection then
            endConnection:Disconnect()
            endConnection = nil
        end
    end

    local function move(input)
        if not active or not dragStart or not startPosition then
            return
        end
        local delta = input.Position - dragStart
        if math.abs(delta.X) >= clickThreshold or math.abs(delta.Y) >= clickThreshold then
            dragging = true
        end
        if not dragging then
            return
        end
        local dx, dy = pixelToScaleDelta(target.Parent, delta)
        local desiredX = startPosition.X.Scale + dx
        local desiredY = startPosition.Y.Scale + dy
        local x, y = clampFrameScale(
            target,
            desiredX,
            desiredY,
            config.MinX or 0,
            config.MinY or 0,
            config.MaxX,
            config.MaxY
        )
        target.Position = UDim2.fromScale(x, y)
        if config.OnDrag then
            config.OnDrag(x, y)
        end
    end

    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then
            return
        end
        active = true
        activeInput = input
        dragStart = input.Position
        startPosition = target.Position

        moveConnection = UserInputService.InputChanged:Connect(function(changed)
            if changed.UserInputType == Enum.UserInputType.MouseMovement
                or changed.UserInputType == Enum.UserInputType.Touch then
                move(changed)
            end
        end)

        endConnection = UserInputService.InputEnded:Connect(function(ended)
            if ended == activeInput
                or ended.UserInputType == Enum.UserInputType.MouseButton1
                or ended.UserInputType == Enum.UserInputType.Touch then
                local wasDragging = dragging
                finish()
                if config.OnRelease then
                    config.OnRelease(wasDragging)
                end
            end
        end)
    end)

    return {
        IsDragging = function()
            return dragging
        end,
        Stop = finish,
    }
end

local menuVisible = false
local menuTweenToken = 0
local confirmVisible = false

local function setMenuVisible(visible, animate)
    menuVisible = visible
    menuTweenToken += 1
    local token = menuTweenToken

    if visible then
        Menu.Visible = true
        if animate then
            MenuScale.Scale = 0.94
            smoothTween(MenuScale, {Scale = 1}, 0.18, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        else
            MenuScale.Scale = 1
        end
    else
        if animate then
            local tween = smoothTween(MenuScale, {Scale = 0.94}, 0.13, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            tween.Completed:Connect(function()
                if token == menuTweenToken and not menuVisible then
                    Menu.Visible = false
                    MenuScale.Scale = 1
                end
            end)
        else
            Menu.Visible = false
            MenuScale.Scale = 1
        end
    end
end

local function clampMainButtonToScreen()
    local x = MainButton.Position.X.Scale
    local y = MainButton.Position.Y.Scale
    local sizeX = MainButton.Size.X.Scale
    local sizeY = MainButton.Size.Y.Scale
    local clampedX = math.clamp(x, 0, 1 - sizeX)
    local clampedY = math.clamp(y, 0, 1 - sizeY)
    if clampedX ~= x or clampedY ~= y then
        MainButton.Position = UDim2.fromScale(clampedX, clampedY)
    end
end

local function clampMenuToScreen()
    local x = Menu.Position.X.Scale
    local y = Menu.Position.Y.Scale
    local sizeX = Menu.Size.X.Scale
    local sizeY = Menu.Size.Y.Scale
    Menu.Position = UDim2.fromScale(
        math.clamp(x, 0, 1 - sizeX),
        math.clamp(y, 0, 1 - sizeY)
    )
end

local function normalizedPositionForRightEdge(frame)
    local sizeX = frame.Size.X.Scale
    return math.clamp(1 - sizeX, 0, 1 - sizeX)
end

local function normalizedPositionForLeftEdge(frame)
    return 0
end

local function restoreMainButtonNearBoundary()
    local position = MainButton.Position
    local sizeX = MainButton.Size.X.Scale
    local sizeY = MainButton.Size.Y.Scale
    local centerX = position.X.Scale + sizeX / 2
    local targetX
    if centerX >= 0.5 then
        targetX = normalizedPositionForRightEdge(MainButton)
    else
        targetX = normalizedPositionForLeftEdge(MainButton)
    end
    local targetY = math.clamp(position.Y.Scale, 0, 1 - sizeY)
    MainButton.Position = UDim2.fromScale(targetX, targetY)
end

local function restoreMenuNearBoundary()
    local position = Menu.Position
    local sizeX = Menu.Size.X.Scale
    local sizeY = Menu.Size.Y.Scale
    local viewportCenter = position.X.Scale + sizeX / 2
    local targetX
    if viewportCenter >= 0.5 then
        targetX = math.clamp(1 - sizeX, 0, 1 - sizeX)
    else
        targetX = 0
    end
    local targetY = math.clamp(position.Y.Scale, 0, 1 - sizeY)
    Menu.Position = UDim2.fromScale(targetX, targetY)
end

local lastScreenSize = getAbsoluteViewport()
local function handleViewportChanged()
    local now = getAbsoluteViewport()
    if now.X <= 0 or now.Y <= 0 then
        return
    end
    local old = lastScreenSize
    lastScreenSize = now
    if old.X == now.X and old.Y == now.Y then
        return
    end
    clampMainButtonToScreen()
    clampMenuToScreen()
end

    local camera = workspace.CurrentCamera
    if camera then
        cameraConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(handleViewportChanged)
    end
end)

workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    pcall(function()
        if cameraConnection then
            cameraConnection:Disconnect()
            cameraConnection = nil
        end
        local camera = workspace.CurrentCamera
        if camera then
            cameraConnection = camera:GetPropertyChangedSignal("ViewportSize"):Connect(handleViewportChanged)
        end
    end)
    handleViewportChanged()
end)

local function keepWithinViewportLoop()
    while ScreenGui.Parent do
        if not mainDrag.IsDragging() then
            local pos = MainButton.Position
            local sizeX = MainButton.Size.X.Scale
            local sizeY = MainButton.Size.Y.Scale
            if pos.X.Scale < -0.02 or pos.X.Scale > 1.02 - sizeX or pos.Y.Scale < -0.02 or pos.Y.Scale > 1.02 - sizeY then
                restoreMainButtonNearBoundary()
            else
                clampMainButtonToScreen()
            end
        end

        if menuVisible and not confirmVisible then
            local anyDragging = false
            for _, dragger in ipairs(menuDraggers) do
                if dragger.IsDragging() then
                    anyDragging = true
                    break
                end
            end
            if not anyDragging then
                local pos = Menu.Position
                local sizeX = Menu.Size.X.Scale
                local sizeY = Menu.Size.Y.Scale
                if pos.X.Scale < -0.08 or pos.X.Scale > 1.08 - sizeX or pos.Y.Scale < -0.08 or pos.Y.Scale > 1.08 - sizeY then
                    restoreMenuNearBoundary()
                else
                    clampMenuToScreen()
                end
            end
        end

        task.wait(0.10)
    end
end

task.spawn(keepWithinViewportLoop)

local function enforceScaleOnly(guiObject)
    local size = guiObject.Size
    local position = guiObject.Position
    if size.X.Offset ~= 0 or size.Y.Offset ~= 0 then
        guiObject.Size = UDim2.fromScale(size.X.Scale, size.Y.Scale)

    if position.X.Offset ~= 0 or position.Y.Offset ~= 0 then
        guiObject.Position = UDim2.fromScale(position.X.Scale, position.Y.Scale)
    end
end

local function searchFutureOptions(query)
    local normalized = string.lower(query or "")
    local count = 0
    for _, child in ipairs(ContentScroll:GetChildren()) do
        if child:IsA("Frame") and child ~= EmptyState then
            local searchable = string.lower(child:GetAttribute("SearchText") or "")
            local matches = normalized ~= "" and string.find(searchable, normalized, 1, true) ~= nil
            child.Visible = matches
            if matches then
                count += 1
            end
        end
    end
    return count
end

local function clearFutureOptions()
    for _, child in ipairs(ContentScroll:GetChildren()) do
        if child:IsA("Frame") and child ~= EmptyState then
            child:Destroy()
        end
    end
end

local function rebuildSearchState()
    local query = string.lower(SearchBox.Text or "")
    if query == "" then
        for _, child in ipairs(ContentScroll:GetChildren()) do
            if child:IsA("Frame") and child ~= EmptyState then
                child.Visible = false
            end
        end
        EmptyState.Visible = true
        EmptyTitle.Text = "No options yet"
        setEmptyStateForTab(ActiveTab)
        return
    end
    local count = searchFutureOptions(query)
    EmptyState.Visible = count == 0
    EmptyTitle.Text = count == 0 and "No matching options" or "No options yet"
    EmptyDescription.Text = count == 0 and "Try another search term." or ""
end

local function initializeFutureOptionLayer()
    clearFutureOptions()
    rebuildSearchState()
end

local function pulseTab(entry)
    if not entry or not entry.Button.Parent then
        return
    end
    local stroke = entry.Stroke
    if stroke then
        local original = stroke.Transparency
        smoothTween(stroke, {Transparency = 0.10}, 0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        task.delay(0.08, function()
            if stroke.Parent then
                smoothTween(stroke, {Transparency = original}, 0.14, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            end
        end)
    end
end

local function reserveOptionRegistry()
    return {
        Main = {},
        ["Auto Buy"] = {},
        Farm = {},
        Config = {},
        Credits = {},
    }

local OptionRegistry = reserveOptionRegistry()

local function getAllOptionsForSearch(query)
    local matches = {}
    local normalized = string.lower(query or "")
    if normalized == "" then
        return matches
    end
    for tabName, options in pairs(OptionRegistry) do
        for _, option in ipairs(options) do
            local haystack = string.lower((option.Title or "") .. " " .. (option.Description or ""))
            if string.find(haystack, normalized, 1, true) then
                local copy = table.clone(option)
                copy.SourceTab = tabName
                table.insert(matches, copy)
            end
        end
    end
    table.sort(matches, function(a, b)
        return string.lower(a.Title or "") < string.lower(b.Title or "")
    end)
    return matches
end

local function rebuildGlobalSearchResults(query)
    local results = getAllOptionsForSearch(query)
    clearFutureOptions()
    if #results == 0 then
        EmptyState.Visible = true
        EmptyTitle.Text = query == "" and "No options yet" or "No matching options"
        EmptyDescription.Text = query == "" and "This tab is ready for future options." or "Try another search term."
        return 0
    end
    EmptyState.Visible = false
    for index, option in ipairs(results) do
        local frame = Instance.new("Frame")
        frame.Name = "SearchResult_" .. tostring(index)
        frame.BackgroundColor3 = COLORS.Panel
        frame.BorderSizePixel = 0
        frame.Size = UDim2.fromScale(0.92, 0.15)
        frame.LayoutOrder = index
        frame.ZIndex = 24
        frame:SetAttribute("SearchText", string.lower((option.Title or "") .. " " .. (option.Description or "")))
        frame.Parent = ContentScroll
        addStroke(frame, 1, 0.70, COLORS.Gray)

        local resultTitle = makeText(
            frame,
            option.Title or "Option",
            UDim2.fromScale(0.05, 0.07),
            UDim2.fromScale(0.72, 0.39),
            13,
            Enum.Font.GothamBold,
            COLORS.White
        )

        local resultDescription = makeText(
            frame,
            option.Description or "",
            UDim2.fromScale(0.05, 0.48),
            UDim2.fromScale(0.82, 0.35),
            10,
            Enum.Font.Gotham,
            COLORS.Gray3
        )
        resultDescription.TextWrapped = true

        local source = makeText(
            frame,
            option.SourceTab or "",
            UDim2.fromScale(0.74, 0.07),
            UDim2.fromScale(0.21, 0.28),
            9,
            Enum.Font.GothamMedium,
            COLORS.Gray2
        )
        source.TextXAlignment = Enum.TextXAlignment.Right
    end
    return #results
end

local globalSearchBound = false
local function bindGlobalSearchOnce()
    if globalSearchBound then
        return
    end
    globalSearchBound = true
    SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
        local text = SearchBox.Text or ""
        if text == "" then
            clearFutureOptions()
            EmptyState.Visible = true
            EmptyTitle.Text = "No options yet"
            setEmptyStateForTab(ActiveTab)
            return
        end
        local count = rebuildGlobalSearchResults(text)
        if count == 0 then
            EmptyState.Visible = true
        end
    end)
end
