local function copyText(text)
    local ok = false
    pcall(function()
        if typeof(setclipboard) == "function" then
            setclipboard(text)
            ok = true
        end
    end)
    if not ok then
        pcall(function()
            if typeof(toclipboard) == "function" then
                toclipboard(text)
                ok = true
            end
        end)
    end
    return ok
end

local function getRequestFunction()
    local candidates = {
        typeof(request) == "function" and request or nil,
        typeof(http_request) == "function" and http_request or nil,
        typeof(syn) == "table" and typeof(syn.request) == "function" and syn.request or nil,
        typeof(fluxus) == "table" and typeof(fluxus.request) == "function" and fluxus.request or nil,
    }
    for _, fn in ipairs(candidates) do
        if fn then return fn end
    end
    return nil
end

local function getRandomServerId()
    local req = getRequestFunction()
    if not req then return nil end
    local cursor = ""
    local servers = {}
    for _ = 1, 8 do
        local url = "https://games.roblox.com/v1/games/" .. tostring(game.PlaceId) .. "/servers/Public?sortOrder=Asc&limit=100"
        if cursor ~= "" then
            url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
        end
        local ok, response = pcall(function()
            return req({
                Url = url,
                Method = "GET",
                Headers = { ["Content-Type"] = "application/json" },
            })
        end)
        if not ok or not response then break end
        local body = response.Body or response.body
        if type(body) ~= "string" then break end
        local decodedOk, decoded = pcall(function() return HttpService:JSONDecode(body) end)
        if not decodedOk or type(decoded) ~= "table" then break end
        for _, server in ipairs(decoded.data or {}) do
            if server.id and server.playing and server.maxPlayers and server.playing < server.maxPlayers and server.id ~= game.JobId then
                table.insert(servers, server.id)
            end
        end
        cursor = decoded.nextPageCursor or ""
        if cursor == "" or #servers > 12 then break end
    end
    if #servers == 0 then return nil end
    return servers[math.random(1, #servers)]
end

local function teleportRandom()
    local target = getRandomServerId()
    if target then
        pcall(function()
            TeleportService:TeleportToPlaceInstance(game.PlaceId, target, player)
        end)
        return
    end
    pcall(function()
        TeleportService:Teleport(game.PlaceId, player)
    end)
end

local antiAfkThread
local infinityConnection

local function setAntiAfk(enabled)
    toggleStates.antiAfk = enabled
    if antiAfkThread then
        task.cancel(antiAfkThread)
        antiAfkThread = nil
    end
    if enabled then
        antiAfkThread = task.spawn(function()
            while toggleStates.antiAfk do
                task.wait(60)
                if not toggleStates.antiAfk then break end
                local hum = getHumanoid()
                if hum and hum.Health > 0 then
                    hum:Move(Vector3.new(0, 0, -1), false)
                    task.wait(0.18)
                    hum:Move(Vector3.zero, false)
                end
            end
        end)
    end
end

local function setInfinityJump(enabled)
    toggleStates.infinityJump = enabled
    if infinityConnection then
        infinityConnection:Disconnect()
        infinityConnection = nil
    end
    if enabled then
        infinityConnection = UserInputService.JumpRequest:Connect(function()
            local hum = getHumanoid()
            if hum and hum.Health > 0 then
                hum:ChangeState(Enum.HumanoidStateType.Jumping)
            end
        end)
    end
end

local function matches(option, query)
    if query == "" then return true end
    local hay = string.lower((option.title or "") .. " " .. (option.description or ""))
    return string.find(hay, string.lower(query), 1, true) ~= nil
end

local buttonPress = false
local buttonDragging = false
local buttonStartPosition
local buttonStartInput
local buttonStartPoint
local dragThreshold = 7

local function inputPoint(input)
    return Vector2.new(input.Position.X, input.Position.Y)
end

local function clampScalePosition(frame, desired)
    local camera = workspace.CurrentCamera
    if not camera then return desired end
    local view = camera.ViewportSize
    local absoluteSize = frame.AbsoluteSize
    local halfX = absoluteSize.X / 2
    local halfY = absoluteSize.Y / 2
    local x = desired.X.Scale * view.X + desired.X.Offset
    local y = desired.Y.Scale * view.Y + desired.Y.Offset
    local minX = halfX + 6
    local maxX = view.X - halfX - 6
    local minY = halfY + 6
    local maxY = view.Y - halfY - 6
    x = math.clamp(x, minX, math.max(minX, maxX))
    y = math.clamp(y, minY, math.max(minY, maxY))
    return UDim2.fromScale(x / view.X, y / view.Y)
end

local function moveButtonToScreenPoint(point)
    local camera = workspace.CurrentCamera
    if not camera then return end
    local view = camera.ViewportSize
    local pos = UDim2.fromScale(point.X / view.X, point.Y / view.Y)
    button.Position = clampScalePosition(button, pos)
end

button.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        buttonPress = true
        buttonDragging = false
        buttonStartInput = input
        buttonStartPoint = inputPoint(input)
        buttonStartPosition = button.Position
    end
end)

button.InputChanged:Connect(function(input)
    if buttonPress and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local now = inputPoint(input)
        if (now - buttonStartPoint).Magnitude >= dragThreshold then
            buttonDragging = true
            local delta = now - buttonStartPoint
            local camera = workspace.CurrentCamera
            if camera then
                local view = camera.ViewportSize
                local desired = UDim2.fromScale(
                    buttonStartPosition.X.Scale + delta.X / view.X,
                    buttonStartPosition.Y.Scale + delta.Y / view.Y
                )
                button.Position = desired
            end
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input == buttonStartInput and buttonPress then
        buttonPress = false
        if buttonDragging then
            button.Position = clampScalePosition(button, button.Position)
        else
            menuOpen = not menuOpen
            mainFrame.Visible = menuOpen
        end
        buttonStartInput = nil
    end
end)

local function addCornerDrag(areaPosition)
    local zone = make("TextButton", {
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Position = areaPosition,
        Size = UDim2.fromScale(0.085, 0.085),
        Text = "",
        AutoButtonColor = false,
        ZIndex = 25,
    }, mainFrame)

    local active = false
    local startPoint
    local startPos

    zone.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            active = true
            startPoint = inputPoint(input)
            startPos = mainFrame.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not active then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local now = inputPoint(input)
            local delta = now - startPoint
            local camera = workspace.CurrentCamera
            if camera then
                local view = camera.ViewportSize
                local nextPos = UDim2.fromScale(
                    startPos.X.Scale + delta.X / view.X,
                    startPos.Y.Scale + delta.Y / view.Y
                )
                mainFrame.Position = nextPos
            end
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if not active then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            active = false
        end
    end)
end

local function correctMainFrameIfFullyOffscreen()
    local camera = workspace.CurrentCamera
    if not camera or not mainFrame.Visible then return end
    local view = camera.ViewportSize
    local p = mainFrame.AbsolutePosition
    local s = mainFrame.AbsoluteSize
    local right = p.X + s.X
    local bottom = p.Y + s.Y
    local fullyLeft = right <= 0
    local fullyRight = p.X >= view.X
    local fullyTop = bottom <= 0
    local fullyBottom = p.Y >= view.Y
    if fullyLeft or fullyRight or fullyTop or fullyBottom then
        local targetX
        if fullyLeft then
            targetX = (s.X / 2 + 10) / view.X
        elseif fullyRight then
            targetX = (view.X - s.X / 2 - 10) / view.X
        else
            targetX = mainFrame.Position.X.Scale
        end
        local targetY = math.clamp(
            (p.Y + s.Y / 2) / view.Y,
            s.Y / 2 / view.Y + 0.01,
            1 - s.Y / 2 / view.Y - 0.01
        )
        TweenService:Create(mainFrame, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
            Position = UDim2.fromScale(math.clamp(targetX, 0.01, 0.99), targetY)
        }):Play()
    end
end
