local function new(className, properties, parent)
	local object = Instance.new(className)
	for property, value in pairs(properties or {}) do
		object[property] = value
	end
	object.Parent = parent
	return object
end

local function tween(object, duration, properties)
	local animation = TweenService:Create(
		object,
		TweenInfo.new(duration or 0.16, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
		properties
	)
	animation:Play()
	return animation
end

local function englishOnly(root)
	local function scan(object)
		if object:IsA("TextLabel") or object:IsA("TextButton") or object:IsA("TextBox") then
			pcall(function()
				object.AutoLocalize = false
			end)
		end
		for _, child in ipairs(object:GetChildren()) do
			scan(child)
		end
	end
	scan(root)
end

local function normalized(text)
	return string.lower(text or "")
end

local function searchOptions(query)
	local result = {}
	local needle = normalized(query)

	if needle == "" then
		return datasets[currentTab] or {}
	end

	for _, tabName in ipairs({"Main", "Auto Buy", "Farm", "Config", "Credits"}) do
		for _, item in ipairs(datasets[tabName] or {}) do
			local title = normalized(item.Title)
			local description = normalized(item.Description)
			if string.find(title, needle, 1, true) or string.find(description, needle, 1, true) then
				table.insert(result, item)
			end
		end
	end

	return result
end

local function render()
	clearResults()

	local query = SearchBox.Text or ""
	local items = searchOptions(query)

	if #items == 0 then
		if query ~= "" then
			emptyState("No matching options found.")
		else
			emptyState("This tab has no options yet.")
		end
		Results.CanvasPosition = Vector2.zero
		return
	end

	for index, item in ipairs(items) do
		createOption(item, index)
	end
end

local function setTab(name)
	currentTab = name

	for tabName, button in pairs(tabButtons) do
		local selected = tabName == name
		tween(button, 0.14, {
			BackgroundColor3 = selected and COLORS.GraySoft or COLORS.BlackSoft,
			BackgroundTransparency = selected and 0 or 0.15,
		})

		local icon = button:FindFirstChild("Icon")
		if icon then
			tween(icon, 0.14, {
				ImageColor3 = selected and COLORS.White or COLORS.SoftWhite,
			})
		end

		local label = button:FindFirstChild("Label")
		if label then
			tween(label, 0.14, {
				TextColor3 = selected and COLORS.White or COLORS.SoftWhite,
			})
		end
	end

	SearchBox.Text = ""
	SearchHint.Visible = true
	SearchClear.Visible = false
	render()
end

local launcherDragging = false
local launcherInput = nil
local launcherStartPointer = nil
local launcherStartPosition = nil
local launcherDragged = false

local function clampLauncher(position)
	local size = viewport()
	local current = viewport()
	local widthScale = math.clamp(launcherHolder.AbsoluteSize.X / math.max(1, size.X), 0, 0.50)
	local heightScale = math.clamp(launcherHolder.AbsoluteSize.Y / math.max(1, size.Y), 0, 0.50)

	local xMin = 0.01
	local xMax = math.max(xMin, 1 - widthScale - 0.01)
	local yMin = 0.02
	local yMax = math.max(yMin, 1 - heightScale - 0.02)

	return UDim2.fromScale(
		math.clamp(position.X.Scale, xMin, xMax),
		math.clamp(position.Y.Scale, yMin, yMax)
	)
end

local function beginLauncherDrag(input)
	launcherDragging = true
	launcherInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
	launcherStartPointer = input.Position
	launcherStartPosition = launcherHolder.Position
	launcherDragged = false
end

local function updateLauncherDrag(input)
	if not launcherStartPointer or not launcherStartPosition then
		return
	end

	local delta = input.Position - launcherStartPointer
	if delta.Magnitude > 5 then
		launcherDragged = true
	end

	local size = viewport()
	if size.X <= 0 or size.Y <= 0 then
		return
	end

	local target = UDim2.fromScale(
		launcherStartPosition.X.Scale + delta.X / size.X,
		launcherStartPosition.Y.Scale + delta.Y / size.Y
	)

	launcherHolder.Position = clampLauncher(target)
end

local function endLauncherDrag()
	launcherDragging = false
	launcherInput = nil
	launcherStartPointer = nil
	launcherStartPosition = nil
end

launcher.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		beginLauncherDrag(input)
	end
end)

local menuDragging = false
local menuDragInput = nil
local menuDragStartPointer = nil
local menuStartPosition = nil
local menuDidDrag = false
local menuPointerDown = 0

local cornerSize = UDim2.fromScale(0.075, 0.11)

local function createDragCorner(name, position, anchor)
	local corner = new("TextButton", {
		Name = name,
		AnchorPoint = anchor,
		Position = position,
		Size = cornerSize,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		AutoButtonColor = false,
		Text = "",
		Active = true,
		Selectable = false,
		ZIndex = 30,
	}, Menu)

	corner.MouseEnter:Connect(function()
		if not menuDragging then
			tween(menuStroke, 0.10, {
				Color = COLORS.White,
				Transparency = 0.04,
			})
		end
	end)

	corner.MouseLeave:Connect(function()
		if not menuDragging then
			tween(menuStroke, 0.10, {
				Color = COLORS.GrayStrong,
				Transparency = 0.10,
			})
		end
	end)

	corner.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			menuDragging = false
			menuDidDrag = false
			menuPointerDown = os.clock()
			menuDragInput = input.UserInputType == Enum.UserInputType.Touch and input or nil
			menuDragStartPointer = input.Position
			menuStartPosition = Menu.Position
		end
	end)

	return corner
end

createDragCorner("TopLeft", UDim2.fromScale(0, 0), Vector2.new(0, 0))
createDragCorner("TopRight", UDim2.fromScale(1, 0), Vector2.new(1, 0))
createDragCorner("BottomLeft", UDim2.fromScale(0, 1), Vector2.new(0, 1))
createDragCorner("BottomRight", UDim2.fromScale(1, 1), Vector2.new(1, 1))

local function updateMenuDrag(input)
	if not menuDragStartPointer or not menuStartPosition then
		return
	end

	local delta = input.Position - menuDragStartPointer
	if delta.Magnitude < 5 then
		return
	end

	if os.clock() - menuPointerDown < 0.045 then
		return
	end

	menuDidDrag = true
	menuDragging = true

	local size = viewport()
	if size.X <= 0 or size.Y <= 0 then
		return
	end

	Menu.Position = UDim2.fromScale(
		menuStartPosition.X.Scale + delta.X / size.X,
		menuStartPosition.Y.Scale + delta.Y / size.Y
	)
end

local function snapMenuIfFullyOutside()
	local size = viewport()
	local position = Menu.AbsolutePosition
	local absoluteSize = Menu.AbsoluteSize

	local fullyLeft = position.X + absoluteSize.X < 0
	local fullyRight = position.X > size.X
	local fullyTop = position.Y + absoluteSize.Y < 0
	local fullyBottom = position.Y > size.Y

	local targetX = Menu.Position.X.Scale
	local targetY = Menu.Position.Y.Scale

	if fullyLeft then
		targetX = 0.02
	elseif fullyRight then
		targetX = 0.98
	end

	if fullyTop then
		targetY = 0.05
	elseif fullyBottom then
		targetY = 0.95
	end

	if fullyLeft or fullyRight or fullyTop or fullyBottom then
		tween(Menu, 0.22, {
			Position = UDim2.fromScale(targetX, targetY),
		})
	end
end

local function endMenuDrag()
	if menuDragging then
		snapMenuIfFullyOutside()
	end

	menuDragging = false
	menuDragInput = nil
	menuDragStartPointer = nil
	menuStartPosition = nil

	tween(menuStroke, 0.12, {
		Color = COLORS.GrayStrong,
		Transparency = 0.10,
	})
end

local function toggleMenu()
	if not ScreenGui.Parent then
		return
	end

	if Menu.Visible then
		Menu.Visible = false
		ConfirmOverlay.Visible = false
	else
		Menu.Visible = true
		ConfirmOverlay.Visible = false
		setTab(currentTab)
	end
end

launcher.Activated:Connect(function()
	if launcherDragged then
		launcherDragged = false
		return
	end
	toggleMenu()
end)

UserInputService.InputChanged:Connect(function(input)
	if launcherDragging then
		if (launcherInput and input == launcherInput) or (not launcherInput and input.UserInputType == Enum.UserInputType.MouseMovement) then
			updateLauncherDrag(input)
		end
	end

	if menuDragInput and input == menuDragInput then
		updateMenuDrag(input)
	elseif menuDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	elseif menuDragStartPointer and not menuDragInput and input.UserInputType == Enum.UserInputType.MouseMovement then
		updateMenuDrag(input)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		if launcherDragging then
			endLauncherDrag()
		end

		if menuDragging then
			endMenuDrag()
		end
	end
end)

RunService.RenderStepped:Connect(function()
	if not ScreenGui.Parent then
		return
	end
	updateGlobalScale()
end)

englishOnly(ScreenGui)
setTab("Main")
Menu.Visible = false
ConfirmOverlay.Visible = false
