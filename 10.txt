local function create(className, properties, parent)
    local object = Instance.new(className)
    for property, value in pairs(properties or {}) do
        object[property] = value
    end
    object.Parent = parent
    return object
end

local function preventTranslation(object)
    pcall(function()
        object.AutoLocalize = false
    end)
end

local function corner(parent, radius)
    return create("UICorner", {
        CornerRadius = UDim.new(radius, 0),
    }, parent)
end

local function outline(parent, color, thickness, transparency)
    return create("UIStroke", {
        Color = color,
        Thickness = thickness,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function tween(object, duration, properties, easingStyle, easingDirection)
    local info = TweenInfo.new(
        duration or 0.16,
        easingStyle or Enum.EasingStyle.Quint,
        easingDirection or Enum.EasingDirection.Out
    )
    local animation = TweenService:Create(object, info, properties)
    animation:Play()
    return animation
end

local function viewport()
    local camera = workspace.CurrentCamera
    return camera and camera.ViewportSize or Vector2.new(1280, 720)
end

local function clamp(value, minimum, maximum)
    return math.clamp(value, minimum, maximum)
end

local function setTabVisual(tabName, selected, pressed)
    local data = tabButtons[tabName]
    if not data then
        return
    end

    local background
    if selected then
        background = COLORS.Gray
    elseif pressed then
        background = COLORS.Gray2
    else
        background = COLORS.Black2
    end

    tween(data.Button, 0.10, {
        BackgroundColor3 = background,
        BackgroundTransparency = selected and 0 or 0.12,
    })

    if selected then
        data.Icon.ImageColor3 = COLORS.White
        data.Label.TextColor3 = COLORS.White
    else
        data.Icon.ImageColor3 = COLORS.SoftWhite
        data.Label.TextColor3 = COLORS.SoftWhite
    end
end

        Name = "Icon",

        Size = UDim2.fromScale(0.18, 0.56),
        BackgroundTransparency = 1,
        Image = info.Icon,
        ImageColor3 = COLORS.SoftWhite,
        ScaleType = Enum.ScaleType.Fit,
        ZIndex = 36,
    }, Button)

    local Label = create("TextLabel", {
        Name = "Label",
        Position = UDim2.fromScale(0.30, 0),
        Size = UDim2.fromScale(0.62, 1),
        BackgroundTransparency = 1,
        Text = info.Name,
        TextColor3 = COLORS.SoftWhite,
        Font = Enum.Font.GothamSemibold,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
        ZIndex = 36,
    }, Button)

    preventTranslation(Button)
    preventTranslation(Label)

    tabButtons[info.Name] = {
        Button = Button,
        Icon = Icon,
        Label = Label,
    }

    Button.MouseEnter:Connect(function()
        if selectedTab ~= info.Name then
            setTabVisual(info.Name, false, true)
        end
    end)

    Button.MouseLeave:Connect(function()
        if selectedTab ~= info.Name then
            setTabVisual(info.Name, false, false)
        end
    end)

    Button.InputBegan:Connect(function(input)
        if selectedTab == info.Name then
            return
        end
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            setTabVisual(info.Name, false, true)
        end
    end)

    Button.Activated:Connect(function()
        selectedTab = info.Name

        for _, other in ipairs(TAB_INFO) do
            setTabVisual(other.Name, other.Name == selectedTab, false)
        end

        -- Always rebuild the content from the currently selected tab.
        -- Credits is the only tab with real options at this stage.
        clearResults()

        if selectedTab == "Credits" and SearchBox.Text == "" then
            addCreditCard()
        elseif SearchBox.Text ~= "" then
            addEmptyState("No Matching Options")
        else
            addEmptyState("No Options")
        end
    end)
end

-- =========================================================
-- Runtime cleanup registry
-- Future gameplay options can register a reset callback here so
-- confirming the close restores their previous state before the GUI is destroyed.
-- =========================================================

local cleanupTasks = {}

local function registerCleanup(callback)
    if typeof(callback) ~= "function" then
        return function() end
    end

    table.insert(cleanupTasks, callback)

    local active = true
    return function()
        if not active then
            return
        end
        active = false

        for index = #cleanupTasks, 1, -1 do
            if cleanupTasks[index] == callback then
                table.remove(cleanupTasks, index)
                break
            end
        end
    end
end

local function runCleanupTasks()
    for index = #cleanupTasks, 1, -1 do
        local callback = cleanupTasks[index]
        cleanupTasks[index] = nil
        pcall(callback)
    end
end

local menuOpen = false

local function setMenuVisible(value)
    menuOpen = value
    Menu.Visible = value
    if not value then
        ConfirmationOverlay.Visible = false
    end

local function openConfirmation()
    if not menuOpen then
        return
    end

    ConfirmationOverlay.Visible = true
    Confirmation.Size = UDim2.fromScale(0.34, 0.24)
    tween(Confirmation, 0.16, {
        Size = UDim2.fromScale(0.36, 0.255),
    })
end

local function closeConfirmation()
    ConfirmationOverlay.Visible = false
end

local lastCloseRequest = 0

local function requestClose()
    if not menuOpen then
        return
    end

    local now = os.clock()
    if now - lastCloseRequest < 0.08 then
        return
    end
    lastCloseRequest = now

    Confirmation.Size = UDim2.fromScale(0.36, 0.255)
    openConfirmation()
end

CloseButton.Activated:Connect(requestClose)

-- Additional coordinate-based fallback for mouse and touch.
-- This makes the entire visible CloseButton rectangle a reliable hit region,
-- even when Roblox's GUI input routing behaves unexpectedly.
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed or not menuOpen then
        return
    end

    local isPointerPress =
        input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch

    if not isPointerPress then
        return
    end

    local pointerPosition = input.Position
    local absolutePosition = CloseButton.AbsolutePosition
    local absoluteSize = CloseButton.AbsoluteSize

    local insideX = pointerPosition.X >= absolutePosition.X
        and pointerPosition.X <= absolutePosition.X + absoluteSize.X

    local insideY = pointerPosition.Y >= absolutePosition.Y
        and pointerPosition.Y <= absolutePosition.Y + absoluteSize.Y

    if insideX and insideY then
        requestClose()
    end
end)

CancelButton.Activated:Connect(function()
    closeConfirmation()
end)

ConfirmButton.Activated:Connect(function()
    runCleanupTasks()
    ScreenGui:Destroy()
end)

            BackgroundColor3 = COLORS.Gray,
            BackgroundTransparency = 0.12,
        })
    end)
end

-- =========================================================
-- Launcher drag
-- Hold to move; release without movement to toggle.
-- Drag remains screen-local and does not move the game camera.
-- =========================================================

local launcherDragging = false
local launcherMoved = false
local launcherInput = nil
local launcherStartInput = nil
local launcherStartPosition = nil

local function setLauncherPosition(x, y, smooth)
    x = clamp(x, 0.04, 0.96)
    y = clamp(y, 0.04, 0.96)

    local target = UDim2.fromScale(x, y)
    if smooth then
        tween(Launcher, 0.10, {Position = target})
    else
        Launcher.Position = target
    end
end

local function updateLauncher(input)
    if not launcherDragging or not launcherStartInput or not launcherStartPosition then
        return
    end

    local view = viewport()
    if view.X <= 0 or view.Y <= 0 then
        return
    end

    local delta = input.Position - launcherStartInput
    if delta.Magnitude > 6 then
        launcherMoved = true
    end

    setLauncherPosition(
        launcherStartPosition.X.Scale + delta.X / view.X,
        launcherStartPosition.Y.Scale + delta.Y / view.Y,
        false
    )
end

LauncherButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        launcherDragging = true
        launcherMoved = false
        launcherInput = input
        launcherStartInput = input.Position
        launcherStartPosition = Launcher.Position
    end
end)

LauncherButton.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        launcherInput = input
    end
end)

-- =========================================================
-- Main menu drag: only the four corner hit zones can start drag.
-- Touch requires a hold/movement, so a simple tap is not treated as drag.
-- =========================================================

local menuDragging = false
local menuMoved = false
local menuInput = nil
local menuStartInput = nil
local menuStartPosition = nil
local menuPressTime = 0
local menuDragSide = "Left"
local menuLastY = 0.5

local cornerHitSize = UDim2.fromScale(0.065, 0.090)
local dragHandles = {}

local function makeDragHandle(name, position, anchorPoint)
    local handle = create("TextButton", {
        Name = name,
        AnchorPoint = anchorPoint,
        Position = position,
        Size = cornerHitSize,
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        AutoButtonColor = false,
        Text = "",
        Active = true,
        ZIndex = 39,
    }, Menu)
    preventTranslation(handle)
    dragHandles[name] = handle
    return handle
end

makeDragHandle("TopLeftDrag", UDim2.fromScale(0, 0), Vector2.new(0, 0))
makeDragHandle("TopRightDrag", UDim2.fromScale(1, 0), Vector2.new(1, 0))
makeDragHandle("BottomLeftDrag", UDim2.fromScale(0, 1), Vector2.new(0, 1))
makeDragHandle("BottomRightDrag", UDim2.fromScale(1, 1), Vector2.new(1, 1))

local function beginMenuDrag(input)
    menuDragging = false
    menuMoved = false
    menuInput = input
    menuStartInput = input.Position
    menuStartPosition = Menu.Position
    menuPressTime = os.clock()
end

for _, handle in pairs(dragHandles) do
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            beginMenuDrag(input)
        end
    end)

    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            menuInput = input
        end
    end)
end

local function updateMenuDrag(input)
    if not menuStartInput or not menuStartPosition then
        return
    end

    local view = viewport()
    if view.X <= 0 or view.Y <= 0 then
        return
    end

    local delta = input.Position - menuStartInput

    if delta.Magnitude > 6 and (os.clock() - menuPressTime) >= 0.05 then
        menuMoved = true
        menuDragging = true
    end

    if not menuDragging then
        return
    end

    -- Allow temporary movement outside the viewport.
    -- Recovery is checked only on release, exactly as requested.
    local newX = clamp(menuStartPosition.X.Scale + delta.X / view.X, -0.75, 1.75)
    local newY = clamp(menuStartPosition.Y.Scale + delta.Y / view.Y, -0.75, 1.75)

    Menu.Position = UDim2.fromScale(newX, newY)
end

local function menuIsFullyOutside()
    local view = viewport()
    local position = Menu.AbsolutePosition
    local size = Menu.AbsoluteSize

    local left = position.X
    local right = position.X + size.X
    local top = position.Y
    local bottom = position.Y + size.Y

    return right <= 0 or left >= view.X or bottom <= 0 or top >= view.Y
end

local function recoverMenu()
    local y = clamp(menuLastY, 0.16, 0.84)
    local x = menuDragSide == "Right" and 0.62 or 0.38

    local target = UDim2.fromScale(x, y)

    tween(Menu, 0.18, {Position = target})
end

local function endMenuDrag()
    if not menuInput then
        return
    end

    if menuDragging then
        if menuIsFullyOutside() then
            recoverMenu()
        else
            local x = Menu.Position.X.Scale
            local y = Menu.Position.Y.Scale
            menuLastY = y
            menuDragSide = x < 0.5 and "Left" or "Right"
        end
    end

    menuDragging = false
    menuMoved = false
    menuInput = nil
    menuStartInput = nil
    menuStartPosition = nil
end

UserInputService.InputChanged:Connect(function(input)
    if launcherDragging and launcherInput then
        if input == launcherInput or input.UserInputType == Enum.UserInputType.MouseMovement then
            updateLauncher(input)
        end
    end

    if menuInput then
        if input == menuInput or input.UserInputType == Enum.UserInputType.MouseMovement then
            updateMenuDrag(input)
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    if launcherDragging then
        local shouldToggle = not launcherMoved
        launcherDragging = false
        launcherInput = nil
        launcherStartInput = nil
        launcherStartPosition = nil

        if shouldToggle then
            setMenuVisible(not menuOpen)
        else
            local p = Launcher.Position
            setLauncherPosition(p.X.Scale, p.Y.Scale, true)
        end
    end

    if menuInput then
        endMenuDrag()
    end
end)

-- =========================================================
-- Viewport maintenance: preserve Scale-based sizing/position.
-- A partially visible menu is not forcibly repositioned.
-- =========================================================

local function connectViewport(camera)
    if not camera then
        return
    end

    camera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
        -- Clamp the launcher so it cannot remain outside its normal safe zone.
        setLauncherPosition(Launcher.Position.X.Scale, Launcher.Position.Y.Scale, false)
    end)
end

connectViewport(workspace.CurrentCamera)

workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    connectViewport(workspace.CurrentCamera)
end)
