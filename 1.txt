local function frame(parent, name, size, position, backgroundColor, transparency)
	local object = Instance.new("Frame")

local function viewport()
	local camera = workspace.CurrentCamera
	return camera and camera.ViewportSize or Vector2.new(1920, 1080)
end

local function clamp(value, low, high)
	return math.max(low, math.min(high, value))
end

local function addAspect(parent, ratio)
	local object = Instance.new("UIAspectRatioConstraint")
	object.AspectRatio = ratio
	object.DominantAxis = Enum.DominantAxis.Width
	object.Parent = parent
	return object
end

local function setTabState(name)
	for tabName, tabButton in pairs(tabButtons) do
		local active = tabName == name

		tabButton.BackgroundColor3 = active and COLORS.Selected or COLORS.Black

		local indicator = tabButton:FindFirstChild("Indicator")
		local tabIcon = tabButton:FindFirstChild("Icon")
		local tabText = tabButton:FindFirstChild("Text")

		if indicator then
			indicator.BackgroundTransparency = active and 0 or 1
		end

		if tabIcon then
			tabIcon.ImageColor3 = active and COLORS.White or COLORS.Gray
		end

		if tabText then
			tabText.TextColor3 = active and COLORS.White or COLORS.SoftWhite
		end
	end
end

local function createToggle(parent, name, state, order)
	local row = frame(
		parent,
		name .. "Row",
		UDim2.fromScale(0.96, 0.11),
		UDim2.fromScale(0.02, 0),
		COLORS.PanelSoft
	)
	row.LayoutOrder = order

	local label = text(
		row,
		"Label",
		name,
		UDim2.fromScale(0.70, 0.78),
		UDim2.fromScale(0.035, 0.11),
		Enum.Font.GothamMedium,
		COLORS.White
	)

	local toggle = button(
		row,
		"Toggle",
		UDim2.fromScale(0.15, 0.52),
		UDim2.fromScale(0.81, 0.24)
	)
	toggle.BackgroundColor3 = state and COLORS.White or COLORS.DarkGray
	addAspect(toggle, 1.90)

	local knob = frame(
		toggle,
		"Knob",
		UDim2.fromScale(0.42, 0.76),
		state and UDim2.fromScale(0.52, 0.12) or UDim2.fromScale(0.06, 0.12),
		state and COLORS.Black or COLORS.White
	)

	local function setState(newState)
		state = newState

		TweenService:Create(
			toggle,
			TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
			{BackgroundColor3 = state and COLORS.White or COLORS.DarkGray}
		):Play()

		TweenService:Create(
			knob,
			TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
			{
				Position = state and UDim2.fromScale(0.52, 0.12) or UDim2.fromScale(0.06, 0.12),
				BackgroundColor3 = state and COLORS.Black or COLORS.White,
			}
		):Play()
	end

	toggle.Activated:Connect(function()
		setState(not state)
	end)

	return row, setState
end

--==================================================
-- Generic smooth drag using only Scale for Position
--==================================================

local function scalePositionFromAbsolute(x, y)
	local vp = viewport()
	return UDim2.fromScale(
		x / math.max(vp.X, 1),
		y / math.max(vp.Y, 1)
	)
end

local function installLauncherDrag()
	local dragging = false
	local moved = false
	local startInput
	local startAbs

	Launcher.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		dragging = true
		moved = false
		startInput = input.Position
		startAbs = Launcher.AbsolutePosition

		local moveConnection
		local endConnection

		moveConnection = UserInputService.InputChanged:Connect(function(changed)
			if not dragging then
				return
			end

			if changed.UserInputType ~= Enum.UserInputType.MouseMovement
				and changed.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			local delta = changed.Position - startInput

			if delta.Magnitude > 6 then
				moved = true
			end

			local vp = viewport()
			local size = Launcher.AbsoluteSize
			local paddingX = vp.X * 0.006
			local paddingY = vp.Y * 0.006

			local x = clamp(
				startAbs.X + delta.X,
				paddingX,
				math.max(paddingX, vp.X - size.X - paddingX)
			)

			local y = clamp(
				startAbs.Y + delta.Y,
				paddingY,
				math.max(paddingY, vp.Y - size.Y - paddingY)
			)

			Launcher.Position = scalePositionFromAbsolute(x, y)
		end)

		endConnection = UserInputService.InputEnded:Connect(function(ended)
			if ended.UserInputType ~= Enum.UserInputType.MouseButton1
				and ended.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			dragging = false

			if moveConnection then
				moveConnection:Disconnect()
			end

			if endConnection then
				endConnection:Disconnect()
			end

			if not moved then
				if Menu.Visible then
					Menu.Visible = false
				else
					Menu.Visible = true
				end
			end
		end)
	end)
end

installLauncherDrag()

--==================================================
-- Menu dragging: only corner handles + hold
-- Partial off-screen is allowed.
-- If the whole menu leaves the screen, it returns to
-- the nearest visible edge while preserving the axis.
--==================================================

local function menuIsFullyOutside()
	local vp = viewport()
	local p = Menu.AbsolutePosition
	local s = Menu.AbsoluteSize

	return p.X + s.X <= 0
		or p.X >= vp.X
		or p.Y + s.Y <= 0
		or p.Y >= vp.Y
end

local function recoverMenu()
	local vp = viewport()
	local p = Menu.AbsolutePosition
	local s = Menu.AbsoluteSize

	local x = p.X
	local y = p.Y

	if p.X + s.X <= 0 then
		x = -s.X + (vp.X * 0.008)
	elseif p.X >= vp.X then
		x = vp.X - (vp.X * 0.008)
	end

	if p.Y + s.Y <= 0 then
		y = -s.Y + (vp.Y * 0.008)
	elseif p.Y >= vp.Y then
		y = vp.Y - (vp.Y * 0.008)
	end

	Menu.Position = scalePositionFromAbsolute(
		x + (s.X * 0.5),
		y + (s.Y * 0.5)
	)
end

local cornerSizeX = 0.075
local cornerSizeY = 0.075

local corners = {
	TopLeft = UDim2.fromScale(0, 0),
	TopRight = UDim2.fromScale(1 - cornerSizeX, 0),
	BottomLeft = UDim2.fromScale(0, 1 - cornerSizeY),
	BottomRight = UDim2.fromScale(1 - cornerSizeX, 1 - cornerSizeY),
}

local function addMenuCorner(name, position)
	local handle = button(
		Menu,
		name,
		UDim2.fromScale(cornerSizeX, cornerSizeY),
		position
	)
	handle.ZIndex = 90
	return handle
end

local function installMenuCornerDrag(handle)
	handle.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local dragging = true
		local startInput = input.Position
		local startMenuAbs = Menu.AbsolutePosition

		local moveConnection
		local endConnection

		moveConnection = UserInputService.InputChanged:Connect(function(changed)
			if not dragging then
				return
			end

			if changed.UserInputType ~= Enum.UserInputType.MouseMovement
				and changed.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			local delta = changed.Position - startInput
			local x = startMenuAbs.X + delta.X
			local y = startMenuAbs.Y + delta.Y

			Menu.Position = scalePositionFromAbsolute(
				x + (Menu.AbsoluteSize.X * 0.5),
				y + (Menu.AbsoluteSize.Y * 0.5)
			)
		end)

		endConnection = UserInputService.InputEnded:Connect(function(ended)
			if ended.UserInputType ~= Enum.UserInputType.MouseButton1
				and ended.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			dragging = false

			if moveConnection then
				moveConnection:Disconnect()
			end

			if endConnection then
				endConnection:Disconnect()
			end

			if menuIsFullyOutside() then
				recoverMenu()
			end
		end)
	end)
end

for name, position in pairs(corners) do
	installMenuCornerDrag(addMenuCorner(name, position))
end

--==================================================
-- Resolution / viewport adaptation
-- All element Position and Size values use Scale.
-- No UDim2 pixel offsets are used.
-- UIScale stays at 1 because Scale is the sizing system.
--==================================================

local function keepLauncherVisible()
	if not Launcher.Parent then
		return
	end

	local vp = viewport()
	local p = Launcher.AbsolutePosition
	local s = Launcher.AbsoluteSize
	local padX = vp.X * 0.006
	local padY = vp.Y * 0.006

	local x = clamp(p.X, padX, math.max(padX, vp.X - s.X - padX))
	local y = clamp(p.Y, padY, math.max(padY, vp.Y - s.Y - padY))

	Launcher.Position = scalePositionFromAbsolute(x, y)
end

local function adaptViewport()
	keepLauncherVisible()

	if Menu.Parent and Menu.Visible and menuIsFullyOutside() then
		recoverMenu()
	end
end

task.defer(adaptViewport)

local camera = workspace.CurrentCamera
if camera then
	camera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
		task.defer(adaptViewport)
	end)
end

-- Keep visible English text fixed against automatic localization.
for _, descendant in ipairs(ScreenGui:GetDescendants()) do
	disableLocalization(descendant)
end
