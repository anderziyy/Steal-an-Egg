local function noLocalize(instance)
	pcall(function()
		instance.AutoLocalize = false
	end)
end

local function roundRect(parent, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(radius, 0)
	corner.Parent = parent
	return corner
end

local function stroke(parent, color, thickness, transparency)
	local uiStroke = Instance.new("UIStroke")
	uiStroke.Color = color
	uiStroke.Thickness = thickness
	uiStroke.Transparency = transparency or 0
	uiStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	uiStroke.Parent = parent
	return uiStroke
end

local function addScaleConstraint(parent, ratio)
	local constraint = Instance.new("UIAspectRatioConstraint")
	constraint.AspectRatio = ratio
	constraint.Parent = parent
	return constraint
end

local function tween(instance, duration, props, style, direction)
	local info = TweenInfo.new(
		duration or 0.15,
		style or Enum.EasingStyle.Quart,
		direction or Enum.EasingDirection.Out
	)
	local t = TweenService:Create(instance, info, props)
	t:Play()
	return t
end

local function newText(parent, text, font, size, color, alignX)
	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.BorderSizePixel = 0
	label.Text = text
	label.Font = font or Enum.Font.Gotham
	label.TextSize = size or 14
	label.TextColor3 = color or C.White
	label.TextXAlignment = alignX or Enum.TextXAlignment.Left
	label.TextYAlignment = Enum.TextYAlignment.Center
	label.TextWrapped = false
	label.RichText = false
	label.TextTruncate = Enum.TextTruncate.AtEnd
	label.Active = false
	label.Selectable = false
	noLocalize(label)
	label.Parent = parent
	return label
end

local selectedTabName = TABS[1].Name
local tabButtons = {}

local function updateTabVisuals(selectedName)
	for name, data in pairs(tabButtons) do
		local button = data.Button
		local icon = data.Icon
		local label = data.Label

		if name == selectedName then
			button.BackgroundColor3 = C.Gray
			icon.ImageColor3 = C.White
			label.TextColor3 = C.White
		else
			button.BackgroundColor3 = C.Dark
			icon.ImageColor3 = C.SoftWhite
			label.TextColor3 = C.SoftWhite
		end
	end
end

	label.Position = UDim2.fromScale(0.34, 0.16)
	label.Size = UDim2.fromScale(0.60, 0.68)

	tabButtons[tab.Name] = {
		Button = button,
		Icon = icon,
		Label = label,
	}

	local hovering = false

	button.MouseEnter:Connect(function()
		hovering = true
		if selectedTabName ~= tab.Name then
			tween(button, 0.10, {BackgroundColor3 = C.Gray2})
		end
	end)

	button.MouseLeave:Connect(function()
		hovering = false
		if selectedTabName ~= tab.Name then
			tween(button, 0.10, {BackgroundColor3 = C.Dark})
		end
	end)

	button.MouseButton1Down:Connect(function()
		if selectedTabName ~= tab.Name then
			tween(button, 0.08, {BackgroundColor3 = C.Gray3})
		end
	end)

	button.MouseButton1Click:Connect(function()
		selectedTabName = tab.Name
		updateTabVisuals(selectedTabName)

		-- The option system is intentionally empty for now.
		emptyTitle.Text = "No Options"
		emptyDescription.Text = "There are no options available in this tab."
	end)
end

updateTabVisuals(selectedTabName)

local function refreshCanvasSizes()
	task.defer(function()
		local tabHeight = tabLayout.AbsoluteContentSize.Y
		local contentHeight = contentLayout.AbsoluteContentSize.Y

		tabScroll.CanvasSize = UDim2.fromScale(0, math.max(1, tabHeight / math.max(1, tabScroll.AbsoluteSize.Y)))
		contentScroll.CanvasSize = UDim2.fromScale(0, math.max(1, contentHeight / math.max(1, contentScroll.AbsoluteSize.Y)))
	end)
end

contentLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(refreshCanvasSizes)
tabLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(refreshCanvasSizes)

-- =========================================================

local function clamp(v, minValue, maxValue)
	return math.clamp(v, minValue, maxValue)
end

local draggingToggle = false
local toggleDragStart = nil
local toggleStartPos = nil
local toggleMoved = false
local toggleInputObject = nil

local function setTogglePosition(scaleX, scaleY, animate)
	local x = clamp(scaleX, 0.035, 0.965)
	local y = clamp(scaleY, 0.045, 0.955)

	local target = UDim2.fromScale(x, y)
	if animate then
		tween(toggleShadow, 0.10, {Position = target})
	else
		toggleShadow.Position = target
	end
end

local function finishToggleDrag()
	if not draggingToggle then
		return
	end

	draggingToggle = false

	local x = toggleShadow.Position.X.Scale
	local y = toggleShadow.Position.Y.Scale

	if x <= 0.08 then
		lastHiddenSide = "Left"
	elseif x >= 0.92 then
		lastHiddenSide = "Right"
	end

	setTogglePosition(x, y, true)
end

toggleButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		draggingToggle = true
		toggleMoved = false
		toggleDragStart = input.Position
		toggleStartPos = toggleShadow.Position
		toggleInputObject = input
	end
end)

toggleButton.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then
		toggleInputObject = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not draggingToggle or not toggleInputObject then
		return
	end

	if input ~= toggleInputObject and input.UserInputType ~= Enum.UserInputType.MouseMovement then
		return
	end

	local current = input.Position
	local delta = current - toggleDragStart

	local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1920, 1080)
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end

	local deltaScaleX = delta.X / viewport.X
	local deltaScaleY = delta.Y / viewport.Y

	if math.abs(delta.X) + math.abs(delta.Y) > 6 then
		toggleMoved = true
	end

	local startX = toggleStartPos.X.Scale
	local startY = toggleStartPos.Y.Scale

	setTogglePosition(startX + deltaScaleX, startY + deltaScaleY, false)
end)

UserInputService.InputEnded:Connect(function(input)
	if not draggingToggle then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		local shouldToggle = not toggleMoved
		finishToggleDrag()

		if shouldToggle then
			toggleMenu()
		end
	end
end)

-- =========================================================
-- Menu drag: only the corners can be held and dragged
-- =========================================================

local cornerHandles = {}
local menuDragging = false
local menuDragStart = nil
local menuStartPos = nil
local activeCorner = nil
local menuMoved = false
local menuInputObject = nil

local CORNER_SIZE = 0.085

local function buildCorner(name, position, anchorPoint)
	local corner = Instance.new("TextButton")
	corner.Name = name
	corner.AnchorPoint = anchorPoint
	corner.Position = position
	corner.Size = UDim2.fromScale(CORNER_SIZE, CORNER_SIZE)
	corner.BackgroundTransparency = 1
	corner.BorderSizePixel = 0
	corner.Text = ""
	corner.AutoButtonColor = false
	corner.TextTransparency = 1
	corner.ZIndex = 45
	corner.Parent = menu
	noLocalize(corner)
	addScaleConstraint(corner, 1)
	table.insert(cornerHandles, corner)
	return corner
end

local topLeft = buildCorner("TopLeftDragHandle", UDim2.fromScale(0, 0), Vector2.new(0, 0))
local topRight = buildCorner("TopRightDragHandle", UDim2.fromScale(1, 0), Vector2.new(1, 0))
local bottomLeft = buildCorner("BottomLeftDragHandle", UDim2.fromScale(0, 1), Vector2.new(0, 1))
local bottomRight = buildCorner("BottomRightDragHandle", UDim2.fromScale(1, 1), Vector2.new(1, 1))

local function visibleEnough()
	local cam = workspace.CurrentCamera
	if not cam then
		return true
	end

	local viewport = cam.ViewportSize
	local menuSize = menu.AbsoluteSize

	if viewport.X <= 0 or viewport.Y <= 0 or menuSize.X <= 0 or menuSize.Y <= 0 then
		return true
	end

	local pos = menu.AbsolutePosition

	local left = pos.X
	local right = pos.X + menuSize.X
	local top = pos.Y
	local bottom = pos.Y + menuSize.Y

	-- "Fully outside" means no visible pixel area remains.
	local overlapsX = right > 0 and left < viewport.X
	local overlapsY = bottom > 0 and top < viewport.Y

	return overlapsX and overlapsY
end

local function restoreMenuFromOutside()
	local y = clamp(lastHiddenY, 0.16, 0.84)
	local x

	if lastHiddenSide == "Right" then
		x = 0.62
	else
		x = 0.38
	end

	menu.Position = UDim2.fromScale(x, y)
	menuShadow.Position = UDim2.fromScale(x, y)

	tween(menu, 0.18, {Position = UDim2.fromScale(x, y)})
	tween(menuShadow, 0.18, {Position = UDim2.fromScale(x, y)})
end

local function setMenuPosition(scaleX, scaleY, animate)
	local x = scaleX
	local y = scaleY

	-- Safe region in normal cases. This keeps the menu entirely visible.
	x = clamp(x, 0.38, 0.62)
	y = clamp(y, 0.38, 0.62)

	local target = UDim2.fromScale(x, y)

	if animate then
		tween(menu, 0.12, {Position = target})
		tween(menuShadow, 0.12, {Position = target})
	else
		menu.Position = target
		menuShadow.Position = target
	end
end

local function beginMenuDrag(input, corner)
	menuDragging = true
	menuMoved = false
	activeCorner = corner
	menuDragStart = input.Position
	menuStartPos = menu.Position
	menuInputObject = input
end

local function endMenuDrag()
	if not menuDragging then
		return
	end

	menuDragging = false

	if not visibleEnough() then
		restoreMenuFromOutside()
	else
		local x = menu.Position.X.Scale
		local y = menu.Position.Y.Scale

		lastHiddenY = y
		if x < 0.5 then
			lastHiddenSide = "Left"
		else
			lastHiddenSide = "Right"
		end

		setMenuPosition(x, y, true)
	end

	activeCorner = nil
	menuInputObject = nil
end

for _, handle in ipairs(cornerHandles) do
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			beginMenuDrag(input, handle)
		end
	end)

	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			menuInputObject = input
		end
	end)
end

UserInputService.InputChanged:Connect(function(input)
	if not menuDragging or not menuInputObject then
		return
	end

	if input ~= menuInputObject and input.UserInputType ~= Enum.UserInputType.MouseMovement then
		return
	end

	local viewport = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(1920, 1080)
	if viewport.X <= 0 or viewport.Y <= 0 then
		return
	end

	local delta = input.Position - menuDragStart
	local deltaX = delta.X / viewport.X
	local deltaY = delta.Y / viewport.Y

	if math.abs(delta.X) + math.abs(delta.Y) > 7 then
		menuMoved = true
	end

	local x = menuStartPos.X.Scale + deltaX
	local y = menuStartPos.Y.Scale + deltaY

	-- During dragging, allow the panel to move toward/outside an edge.
	-- The exact recovery happens on release only if the whole menu is outside.
	x = clamp(x, -0.70, 1.70)
	y = clamp(y, -0.70, 1.70)

	menu.Position = UDim2.fromScale(x, y)
	menuShadow.Position = UDim2.fromScale(x, y)
end)

UserInputService.InputEnded:Connect(function(input)
	if not menuDragging then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then
		endMenuDrag()
	end
end)

-- =========================================================
-- Search filtering
-- =========================================================

local function updateSearch()
	local query = string.lower(searchText.Text or "")
	clearSearch.Visible = query ~= ""

	if query == "" then
		searchText.PlaceholderText = "Search..."
		emptyCard.Visible = true
		return
	end

	searchText.PlaceholderText = ""
	-- No option dataset exists yet, so the empty state remains visible.
	emptyCard.Visible = true
	emptyTitle.Text = "No Matching Options"
	emptyDescription.Text = "No options match \"" .. searchText.Text .. "\"."
end

searchText:GetPropertyChangedSignal("Text"):Connect(updateSearch)

clearSearch.MouseEnter:Connect(function()
	tween(clearSearch, 0.10, {TextColor3 = C.LightGray})
end)

clearSearch.MouseLeave:Connect(function()
	tween(clearSearch, 0.10, {TextColor3 = C.White})
end)

clearSearch.MouseButton1Click:Connect(function()
	searchText.Text = ""
	searchText:CaptureFocus()
	updateSearch()
end)

-- =========================================================
-- Responsive handling
-- =========================================================

local function keepToggleInBounds()
	local x = toggleShadow.Position.X.Scale
	local y = toggleShadow.Position.Y.Scale
	setTogglePosition(x, y, false)
end

local function keepMenuInBounds()
	if not menu.Visible then
		return
	end

	if not visibleEnough() then
		restoreMenuFromOutside()
		return
	end

	-- Do not force a menu back when even a portion remains visible.
	local x = menu.Position.X.Scale
	local y = menu.Position.Y.Scale

	if x < 0.05 or x > 0.95 or y < 0.05 or y > 0.95 then
		-- Keep the user's visible placement, but move gently toward a usable center.
		local targetX = clamp(x, 0.10, 0.90)
		local targetY = clamp(y, 0.10, 0.90)
		menu.Position = UDim2.fromScale(targetX, targetY)
		menuShadow.Position = UDim2.fromScale(targetX, targetY)
	end
end

local camera = workspace.CurrentCamera

if camera then
	camera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
		task.defer(function()
			keepToggleInBounds()
			keepMenuInBounds()
			refreshCanvasSizes()
		end)
	end)
end

workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
	local currentCamera = workspace.CurrentCamera
	if currentCamera then
		currentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
			task.defer(function()
				keepToggleInBounds()
				keepMenuInBounds()
				refreshCanvasSizes()
			end)
		end)
	end
end)
