local function makeText(parent, text, size, font, color, alignment)
	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.Size = size
	label.Text = text
	label.Font = font or Enum.Font.Gotham
	label.TextColor3 = color or COLORS.White
	label.TextSize = math.max(10, math.floor(workspace.CurrentCamera.ViewportSize.Y * 0.018))
	label.TextXAlignment = alignment or Enum.TextXAlignment.Left
	label.TextYAlignment = Enum.TextYAlignment.Center
	label.ClipsDescendants = true
	label.Parent = parent
	setNoLocalization(label)
	return label
end

local function clamp(value, minValue, maxValue)
	return math.max(minValue, math.min(maxValue, value))
end

local function viewport()

local function keepOnScreen(guiObject, extraMarginScale)
	local vp = viewport()
	local absSize = guiObject.AbsoluteSize
	local desired = guiObject.AbsolutePosition
	local marginX = vp.X * (extraMarginScale or 0.008)
	local marginY = vp.Y * (extraMarginScale or 0.008)

	local minX = marginX
	local minY = marginY
	local maxX = math.max(minX, vp.X - absSize.X - marginX)
	local maxY = math.max(minY, vp.Y - absSize.Y - marginY)

	local newX = clamp(desired.X, minX, maxX)
	local newY = clamp(desired.Y, minY, maxY)

	guiObject.Position = UDim2.fromScale(newX / vp.X, newY / vp.Y)
end

local function createScaleDrag(guiObject, dragHandle)
	local dragging = false
	local dragStart
	local startPosition

	local function begin(input)
		if dragging then
			return
		end

		dragging = true
		dragStart = input.Position
		startPosition = guiObject.Position

		local connection
		connection = UserInputService.InputChanged:Connect(function(changed)
			if not dragging then
				if connection then
					connection:Disconnect()
				end
				return
			end

			if changed.UserInputType ~= Enum.UserInputType.MouseMovement
				and changed.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			local vp = viewport()
			local delta = changed.Position - dragStart
			local deltaScaleX = delta.X / math.max(1, vp.X)
			local deltaScaleY = delta.Y / math.max(1, vp.Y)

			local targetX = startPosition.X.Scale + deltaScaleX
			local targetY = startPosition.Y.Scale + deltaScaleY

			local currentSizeX = guiObject.Size.X.Scale
			local currentSizeY = guiObject.Size.Y.Scale

			local padding = 0.006
			local minX = padding
			local minY = padding
			local maxX = 1 - currentSizeX - padding
			local maxY = 1 - currentSizeY - padding

			guiObject.Position = UDim2.fromScale(
				clamp(targetX, minX, math.max(minX, maxX)),
				clamp(targetY, minY, math.max(minY, maxY))
			)
		end)

		local endedConnection
		endedConnection = UserInputService.InputEnded:Connect(function(ended)
			if ended.UserInputType == Enum.UserInputType.MouseButton1
				or ended.UserInputType == Enum.UserInputType.Touch then
				dragging = false
				if connection then
					connection:Disconnect()
				end
				if endedConnection then
					endedConnection:Disconnect()
				end
				keepOnScreen(guiObject)
			end
		end)
	end

	dragHandle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			begin(input)
		end
	end)
end

-- Button drag versus click
local openDragging = false
local openDragStart = nil
local openStartPosition = nil
local openMoved = false

OpenButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		openDragging = true
		openDragStart = input.Position
		openStartPosition = OpenButton.Position
		openMoved = false

		local moveConnection
		local endConnection

		moveConnection = UserInputService.InputChanged:Connect(function(changed)
			if not openDragging then
				return
			end

			if changed.UserInputType ~= Enum.UserInputType.MouseMovement
				and changed.UserInputType ~= Enum.UserInputType.Touch then
				return
			end

			local delta = changed.Position - openDragStart
			if delta.Magnitude > 6 then
				openMoved = true
			end

			local vp = viewport()
			local dx = delta.X / math.max(1, vp.X)
			local dy = delta.Y / math.max(1, vp.Y)
			local sx = OpenButton.Size.X.Scale
			local sy = OpenButton.Size.Y.Scale
			local padding = 0.006

			local x = clamp(openStartPosition.X.Scale + dx, padding, 1 - sx - padding)
			local y = clamp(openStartPosition.Y.Scale + dy, padding, 1 - sy - padding)

			OpenButton.Position = UDim2.fromScale(x, y)
		end)

		endConnection = UserInputService.InputEnded:Connect(function(ended)
			if ended.UserInputType == Enum.UserInputType.MouseButton1
				or ended.UserInputType == Enum.UserInputType.Touch then

				openDragging = false

				if moveConnection then
					moveConnection:Disconnect()
				end
				if endConnection then
					endConnection:Disconnect()
				end

				keepOnScreen(OpenButton)

				if not openMoved then
					-- Toggle is handled below after menu objects are created.
					OpenButton:SetAttribute("RequestToggle", true)
				end
			end
		end)
	end
end)

local function setTabVisual(tabName)
	for name, button in pairs(TabButtons) do
		local active = name == tabName
		button.BackgroundColor3 = active and COLORS.Selected or COLORS.Black
		local indicator = button:FindFirstChild("Indicator")
		local icon = button:FindFirstChild("Icon")
		local title = button:FindFirstChild("Title")
		if indicator then
			indicator.BackgroundTransparency = active and 0 or 1
		end
		if icon and icon:IsA("ImageLabel") then
			icon.ImageColor3 = active and COLORS.White or COLORS.Gray
		end
		if title and title:IsA("TextLabel") then
			title.TextColor3 = active and COLORS.White or COLORS.SoftWhite
		end
	end
end

local function createToggle(parent, titleText, initialState, order)
	local row = makeCornerlessFrame(parent, "ToggleRow_" .. tostring(order), UDim2.fromScale(0.96, 0.11), UDim2.fromScale(0.02, 0), COLORS.Panel2)
	row.LayoutOrder = order

	local label = makeText(row, titleText, UDim2.fromScale(0.67, 0.80), Enum.Font.GothamMedium, COLORS.White)
	label.Position = UDim2.fromScale(0.035, 0.10)
	label.TextScaled = true

	local toggle = Instance.new("TextButton")
	toggle.Size = UDim2.fromScale(0.15, 0.52)
	toggle.Position = UDim2.fromScale(0.81, 0.24)
	toggle.BackgroundColor3 = initialState and COLORS.White or COLORS.DarkGray
	toggle.BorderSizePixel = 0
	toggle.Text = ""
	toggle.AutoButtonColor = false
	toggle.Parent = row
	setNoLocalization(toggle)

	local toggleAspect = Instance.new("UIAspectRatioConstraint")
	toggleAspect.AspectRatio = 1.85
	toggleAspect.DominantAxis = Enum.DominantAxis.Width
	toggleAspect.Parent = toggle

	local knob = Instance.new("Frame")
	knob.Size = UDim2.fromScale(0.42, 0.76)
	knob.Position = initialState and UDim2.fromScale(0.52, 0.12) or UDim2.fromScale(0.06, 0.12)
	knob.BackgroundColor3 = initialState and COLORS.Black or COLORS.White
	knob.BorderSizePixel = 0
	knob.Parent = toggle

	local function setState(state)
		initialState = state
		TweenService:Create(toggle, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			BackgroundColor3 = state and COLORS.White or COLORS.DarkGray
		}):Play()
		TweenService:Create(knob, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			Position = state and UDim2.fromScale(0.52, 0.12) or UDim2.fromScale(0.06, 0.12),
			BackgroundColor3 = state and COLORS.Black or COLORS.White,
		}):Play()
	end

	toggle.Activated:Connect(function()
		setState(not initialState)
	end)

	return row, toggle, setState
end

local closeMenuButton = true

local function showMenu()
	if closeMenuButton then
		MainMenu.Visible = true
		MainMenu.Position = UDim2.fromScale(0.50, 0.52)
		TweenService:Create(MainMenu, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			Position = UDim2.fromScale(0.50, 0.50)
		}):Play()

end

local function hideMenu()
	MainMenu.Visible = false
	Overlay.Visible = false
end

local function toggleMenu()
	if Overlay.Visible then
		return
	end
	if MainMenu.Visible then
		hideMenu()
	else
		showMenu()
	end
end

end)

CloseButton.Activated:Connect(function()
	if MainMenu.Visible then
		Overlay.Visible = true
	end
end)

CancelButton.Activated:Connect(function()
	Overlay.Visible = false
end)

ConfirmButton.Activated:Connect(function()
	closeMenuButton = false
	Overlay:Destroy()
	MainMenu:Destroy()
	OpenButton:Destroy()
	ScreenGui:Destroy()
end)

-- Drag only from corners: invisible active corner handles.
local function createCornerDragHandle(name, position)
	local handle = Instance.new("TextButton")
	handle.Name = name
	handle.Size = UDim2.fromScale(0.075, 0.075)
	handle.Position = position
	handle.BackgroundTransparency = 1
	handle.Text = ""
	handle.AutoButtonColor = false
	handle.ZIndex = 50
	handle.Parent = MainMenu
	setNoLocalization(handle)
	return handle
end

local TopLeftHandle = createCornerDragHandle("TopLeftDrag", UDim2.fromScale(0, 0))
local TopRightHandle = createCornerDragHandle("TopRightDrag", UDim2.fromScale(0.925, 0))
local BottomLeftHandle = createCornerDragHandle("BottomLeftDrag", UDim2.fromScale(0, 0.925))
local BottomRightHandle = createCornerDragHandle("BottomRightDrag", UDim2.fromScale(0.925, 0.925))

createScaleDrag(MainMenu, TopLeftHandle)
createScaleDrag(MainMenu, TopRightHandle)
createScaleDrag(MainMenu, BottomLeftHandle)
createScaleDrag(MainMenu, BottomRightHandle)

-- Keep the main menu on-screen after viewport/resolution changes.
local function reClampMenu()
	if not MainMenu.Parent then
		return
	end

	local vp = viewport()
	local size = MainMenu.AbsoluteSize
	local pos = MainMenu.AbsolutePosition
	local marginX = vp.X * 0.006
	local marginY = vp.Y * 0.006

	local x = clamp(pos.X, marginX, math.max(marginX, vp.X - size.X - marginX))
	local y = clamp(pos.Y, marginY, math.max(marginY, vp.Y - size.Y - marginY))

	if vp.X > 0 and vp.Y > 0 then
		MainMenu.Position = UDim2.fromScale((x + size.X / 2) / vp.X, (y + size.Y / 2) / vp.Y)
	end
end

local camera = workspace.CurrentCamera
if camera then
	camera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
		task.defer(function()
			if OpenButton.Parent then
				keepOnScreen(OpenButton)
			end
			if MainMenu.Parent and MainMenu.Visible then
				reClampMenu()
			end
		end)
	end)
end

-- Keep button's square visual consistent on all devices.
local function updateResponsiveSizing()
	if not OpenButton.Parent then
		return
	end

	local vp = viewport()
	local shortest = math.min(vp.X, vp.Y)

	-- Size remains scale-based; UIScale compensates only for extreme viewport ratios.
	-- The object itself never uses pixel Offset sizing.
	if shortest < 600 then
		OpenScale.Scale = 1
	elseif shortest < 900 then
		OpenScale.Scale = 1
	else
		OpenScale.Scale = 1
	end
end

updateResponsiveSizing()
