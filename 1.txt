local function makeCorner(parent, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(radius, 0)
	c.Parent = parent
	return c
end

local function makeStroke(parent, color, thickness, transparency)
	local s = Instance.new("UIStroke")
	s.Color = color
	s.Thickness = thickness
	s.Transparency = transparency or 0
	s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	s.Parent = parent
	return s
end

local function makeAspect(parent, ratio)
	local a = Instance.new("UIAspectRatioConstraint")
	a.AspectRatio = ratio
	a.Parent = parent
	return a
end

local function makeText(parent, text, font, textSize, color, xAlignment)
	local l = Instance.new("TextLabel")
	l.BackgroundTransparency = 1
	l.BorderSizePixel = 0
	l.Text = text
	l.Font = font or Enum.Font.Gotham
	l.TextSize = textSize or 14
	l.TextColor3 = color or COLORS.White
	l.TextXAlignment = xAlignment or Enum.TextXAlignment.Left
	l.TextYAlignment = Enum.TextYAlignment.Center
	l.TextWrapped = false
	l.TextTruncate = Enum.TextTruncate.AtEnd
	l.RichText = false
	l.Active = false
	l.Selectable = false
	disableLocalization(l)
	l.Parent = parent
	return l
end

local function playTween(instance, duration, properties, easingStyle, easingDirection)
	local info = TweenInfo.new(
		duration or 0.15,
		easingStyle or Enum.EasingStyle.Quart,
		easingDirection or Enum.EasingDirection.Out
	)
	local tw = TweenService:Create(instance, info, properties)
	tw:Play()
	return tw
end

local function clamp(value, minimum, maximum)
	return math.clamp(value, minimum, maximum)
end

local selectedTab = TABS[1].Name
local tabData = {}

local function setTabVisual(name, state)
	local data = tabData[name]
	if not data then
		return
	end

	if state == "selected" then
		playTween(data.Button, 0.10, {BackgroundColor3 = COLORS.Gray})
		data.Icon.ImageColor3 = COLORS.White
		data.Label.TextColor3 = COLORS.White
	elseif state == "pressed" then
		playTween(data.Button, 0.08, {BackgroundColor3 = COLORS.Gray3})
	elseif state == "hover" then
		playTween(data.Button, 0.10, {BackgroundColor3 = COLORS.Gray2})
	else
		playTween(data.Button, 0.10, {BackgroundColor3 = COLORS.Dark})
		data.Icon.ImageColor3 = COLORS.SoftWhite
		data.Label.TextColor3 = COLORS.SoftWhite
	end
end

local function refreshTabs()
	for _, tab in ipairs(TABS) do
		if tab.Name == selectedTab then
			setTabVisual(tab.Name, "selected")
		else
			setTabVisual(tab.Name, "normal")
		end
	end
end

for index, tab in ipairs(TABS) do
	local button = Instance.new("TextButton")
	button.Name = tab.Name:gsub("%s+", "") .. "Tab"
	button.BackgroundColor3 = COLORS.Dark

	button.AutoButtonColor = false
	button.Size = UDim2.fromScale(0.98, 0.15)
	button.LayoutOrder = index
	button.ZIndex = 35
	button.Parent = tabScroll
	disableLocalization(button)
	makeCorner(button, 0.055)

	local iconBox = Instance.new("Frame")
	iconBox.Name = "IconBox"
	iconBox.BackgroundColor3 = COLORS.Gray2
	iconBox.BorderSizePixel = 0
	iconBox.Position = UDim2.fromScale(0.055, 0.15)
	iconBox.Size = UDim2.fromScale(0.205, 0.70)
	iconBox.ZIndex = 36
	iconBox.Parent = button
	makeAspect(iconBox, 1)
	makeCorner(iconBox, 0.10)

	local icon = Instance.new("ImageLabel")
	icon.Name = "Icon"
	icon.BackgroundTransparency = 1
	icon.BorderSizePixel = 0
	icon.Image = tab.Icon
	icon.ImageColor3 = COLORS.SoftWhite
	icon.ScaleType = Enum.ScaleType.Fit
	icon.Position = UDim2.fromScale(0.15, 0.15)
	icon.Size = UDim2.fromScale(0.70, 0.70)
	icon.ZIndex = 37
	icon.Parent = iconBox

	local label = makeText(
		button,
		tab.Name,
		Enum.Font.GothamSemibold,
		11,
		COLORS.SoftWhite,
		Enum.TextXAlignment.Left
	)
	label.Position = UDim2.fromScale(0.335, 0.15)
	label.Size = UDim2.fromScale(0.61, 0.70)

	tabData[tab.Name] = {
		Button = button,
		Icon = icon,
		Label = label,
	}

	button.MouseEnter:Connect(function()
		if selectedTab ~= tab.Name then
			setTabVisual(tab.Name, "hover")
		end
	end)

	button.MouseLeave:Connect(function()
		if selectedTab ~= tab.Name then
			setTabVisual(tab.Name, "normal")
		end
	end)

	button.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			if selectedTab ~= tab.Name then
				setTabVisual(tab.Name, "pressed")
			end
		end
	end)

	button.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			if selectedTab ~= tab.Name then
				setTabVisual(tab.Name, "normal")
			end
		end
	end)

	button.MouseButton1Click:Connect(function()
		selectedTab = tab.Name
		refreshTabs()
		emptyTitle.Text = "No Options"
		emptyDescription.Text = "There are no options available in this tab."
	end)
end

refreshTabs()

-- Visibility state
-- ==========================================================

local menuVisible = false
local confirmationVisible = false

local function setMenuVisible(value)
	menuVisible = value
	menu.Visible = value
	menuShadow.Visible = value

	if not value and confirmationVisible then
		confirmationVisible = false
		confirmationOverlay.Visible = false
	end
end

local function showConfirmation()
	if not menuVisible then
		return
	end

	confirmationVisible = true
	confirmationOverlay.Visible = true
	confirmation.Size = UDim2.fromScale(0.33, 0.24)
	playTween(confirmation, 0.14, {
		Size = UDim2.fromScale(0.35, 0.255)
	})
end

local function hideConfirmation()
	confirmationVisible = false
	playTween(confirmation, 0.10, {
		Size = UDim2.fromScale(0.33, 0.24)
	})
	task.delay(0.10, function()
		if confirmationVisible == false and confirmationOverlay.Parent then
			confirmationOverlay.Visible = false
		end
	end)
end

local function deleteEverything()
	confirmationVisible = false
	menuVisible = false
	gui:Destroy()
end

-- ==========================================================

local function refreshSearchState()
	local text = searchBox.Text or ""
	local hasText = text ~= ""

	clearButton.Visible = hasText

	if hasText then
		searchBox.PlaceholderText = ""
		emptyTitle.Text = "No Matching Options"
		emptyDescription.Text = "No options match \"" .. text .. "\"."
	else
		searchBox.PlaceholderText = "Search..."
		emptyTitle.Text = "No Options"
		emptyDescription.Text = "There are no options available in this tab."
	end
end

searchBox:GetPropertyChangedSignal("Text"):Connect(refreshSearchState)

clearButton.MouseEnter:Connect(function()
	clearButton.TextColor3 = COLORS.LightGray
end)

clearButton.MouseLeave:Connect(function()
	clearButton.TextColor3 = COLORS.White
end)

clearButton.MouseButton1Click:Connect(function()
	searchBox.Text = ""
	searchBox:CaptureFocus()
end)

-- ==========================================================
-- Dragging: floating button
-- ==========================================================

local toggleDragging = false
local toggleDragStart
local toggleStartPosition
local toggleInput
local toggleMoved = false
local lastToggleSide = "Left"
local lastToggleY = 0.165

local function setTogglePosition(x, y, smooth)
	x = clamp(x, 0.038, 0.962)
	y = clamp(y, 0.045, 0.955)

	local destination = UDim2.fromScale(x, y)
	if smooth then
		playTween(toggleShadow, 0.10, {Position = destination})
	else
		toggleShadow.Position = destination
	end
end

toggleButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		toggleDragging = true
		toggleMoved = false
		toggleDragStart = input.Position
		toggleStartPosition = toggleShadow.Position
		toggleInput = input
	end
end)

toggleButton.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then
		toggleInput = input
	end
end)

local function finishToggleDrag()
	if not toggleDragging then
		return
	end

	toggleDragging = false

	local x = toggleShadow.Position.X.Scale
	local y = toggleShadow.Position.Y.Scale

	if x <= 0.15 then
		lastToggleSide = "Left"
	elseif x >= 0.85 then
		lastToggleSide = "Right"
	end

	lastToggleY = y
	setTogglePosition(x, y, true)
end

-- ==========================================================
-- Dragging: main menu, corners only
-- ==========================================================

local menuDragging = false
local menuDragStart
local menuStartPosition
local menuInput
local menuMoved = false
local lastMenuSide = "Left"
local lastMenuY = 0.5

local cornerSize = 0.09
local dragHandles = {}

local function createDragHandle(name, position, anchor)
	local h = Instance.new("TextButton")
	h.Name = name
	h.AnchorPoint = anchor
	h.Position = position
	h.Size = UDim2.fromScale(cornerSize, cornerSize)
	h.BackgroundTransparency = 1
	h.BorderSizePixel = 0
	h.Text = ""
	h.TextTransparency = 1
	h.AutoButtonColor = false
	h.ZIndex = 49
	h.Parent = menu
	disableLocalization(h)
	makeAspect(h, 1)
	table.insert(dragHandles, h)
	return h
end

createDragHandle("TopLeftDragHandle", UDim2.fromScale(0, 0), Vector2.new(0, 0))
createDragHandle("TopRightDragHandle", UDim2.fromScale(1, 0), Vector2.new(1, 0))
createDragHandle("BottomLeftDragHandle", UDim2.fromScale(0, 1), Vector2.new(0, 1))
createDragHandle("BottomRightDragHandle", UDim2.fromScale(1, 1), Vector2.new(1, 1))

local function menuHasVisiblePixels()
	local camera = workspace.CurrentCamera
	if not camera then
		return true
	end

	local viewport = camera.ViewportSize
	local absolutePosition = menu.AbsolutePosition
	local absoluteSize = menu.AbsoluteSize

	if viewport.X <= 0 or viewport.Y <= 0 then
		return true
	end

	local left = absolutePosition.X
	local right = absolutePosition.X + absoluteSize.X
	local top = absolutePosition.Y
	local bottom = absolutePosition.Y + absoluteSize.Y

	local intersectsX = right > 0 and left < viewport.X
	local intersectsY = bottom > 0 and top < viewport.Y

	return intersectsX and intersectsY
end

local function recoverMenuFromSide()
	local x = lastMenuSide == "Right" and 0.62 or 0.38
	local y = clamp(lastMenuY, 0.18, 0.82)
	local destination = UDim2.fromScale(x, y)

	playTween(menu, 0.18, {Position = destination})
	playTween(menuShadow, 0.18, {Position = destination})
end

local function finishMenuDrag()
	if not menuDragging then
		return
	end

	menuDragging = false

	if not menuHasVisiblePixels() then
		recoverMenuFromSide()
	else
		local x = menu.Position.X.Scale
		local y = menu.Position.Y.Scale

		lastMenuY = y

		if x < 0.5 then
			lastMenuSide = "Left"
		else
			lastMenuSide = "Right"
		end
	end

	menuInput = nil
end

for _, handle in ipairs(dragHandles) do
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then

			menuDragging = true
			menuMoved = false
			menuDragStart = input.Position
			menuStartPosition = menu.Position
			menuInput = input
		end
	end)

	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			menuInput = input
		end
	end)
end

-- ==========================================================
-- Global input
-- ==========================================================

UserInputService.InputChanged:Connect(function(input)
	-- Floating button drag
	if toggleDragging and toggleInput then
		if input == toggleInput or input.UserInputType == Enum.UserInputType.MouseMovement then
			local camera = workspace.CurrentCamera
			if camera then
				local viewport = camera.ViewportSize
				if viewport.X > 0 and viewport.Y > 0 then
					local delta = input.Position - toggleDragStart
					local dx = delta.X / viewport.X
					local dy = delta.Y / viewport.Y

					if math.abs(delta.X) + math.abs(delta.Y) > 7 then
						toggleMoved = true
					end

					local newX = toggleStartPosition.X.Scale + dx
					local newY = toggleStartPosition.Y.Scale + dy
					setTogglePosition(newX, newY, false)
				end
			end
		end
	end

	-- Main menu drag
	if menuDragging and menuInput then
		if input == menuInput or input.UserInputType == Enum.UserInputType.MouseMovement then
			local camera = workspace.CurrentCamera
			if camera then
				local viewport = camera.ViewportSize
				if viewport.X > 0 and viewport.Y > 0 then
					local delta = input.Position - menuDragStart
					local dx = delta.X / viewport.X
					local dy = delta.Y / viewport.Y

					if math.abs(delta.X) + math.abs(delta.Y) > 7 then
						menuMoved = true
					end

					-- Deliberately allows temporary full exit; recovery happens only
					-- when the entire menu is outside the visible viewport.
					local newX = clamp(menuStartPosition.X.Scale + dx, -0.75, 1.75)
					local newY = clamp(menuStartPosition.Y.Scale + dy, -0.75, 1.75)

					menu.Position = UDim2.fromScale(newX, newY)
					menuShadow.Position = UDim2.fromScale(newX, newY)
				end
			end
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		if toggleDragging then
			local shouldToggle = not toggleMoved
			finishToggleDrag()

			if shouldToggle then
				setMenuVisible(not menuVisible)
			end
		end

		if menuDragging then
			finishMenuDrag()
		end
	end
end)

toggleButton.MouseEnter:Connect(function()
	playTween(toggleShadow, 0.12, {BackgroundColor3 = COLORS.Gray})
	playTween(toggleButton, 0.12, {Size = UDim2.fromScale(0.975, 0.975)})
end)

toggleButton.MouseLeave:Connect(function()
	playTween(toggleShadow, 0.12, {BackgroundColor3 = COLORS.Black})
	playTween(toggleButton, 0.12, {Size = UDim2.fromScale(0.94, 0.94)})
end)

-- ==========================================================
-- Responsive viewport maintenance
-- ==========================================================

local function maintainToggle()
	setTogglePosition(toggleShadow.Position.X.Scale, toggleShadow.Position.Y.Scale, false)
end

local function maintainMenu()
	if not menuVisible or not menuHasVisiblePixels() then
		return
	end

	-- The panel is only automatically recovered after a complete disappearance.
	-- A partially visible panel remains where the user left it.
end

local function connectCamera(camera)
	if not camera then
		return
	end

	camera:GetPropertyChangedSignal("ViewportSize"):Connect(function()
		task.defer(function()
			maintainToggle()
			maintainMenu()
		end)
	end)
end

connectCamera(workspace.CurrentCamera)

workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
	connectCamera(workspace.CurrentCamera)
end)

-- Initial state
setTogglePosition(0.075, 0.165, false)
setMenuVisible(false)
refreshSearchState()
