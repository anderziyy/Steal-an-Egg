local function new(className, properties)
    local object = Instance.new(className)
    for property, value in pairs(properties or {}) do

local antiAFK = false
local antiAFKThread = nil

local function getRootPart()
    local character = LocalPlayer.Character
    return character and character:FindFirstChild("HumanoidRootPart"), character and character:FindFirstChildOfClass("Humanoid")
end

function enableAntiAFK()
    if antiAFK then return end
    antiAFK = true
    antiAFKThread = task.spawn(function()
        while antiAFK do
            task.wait(8)
            if not antiAFK then break end
            local root, humanoid = getRootPart()
            if root and humanoid and humanoid.Health > 0 then
                local original = humanoid.MoveDirection
                humanoid:Move(Vector3.new(1, 0, 0), true)
                task.wait(0.35)
                humanoid:Move(Vector3.zero, true)
                task.wait(0.35)
                if original.Magnitude > 0 then
                    humanoid:Move(original, true)
                end
            end
            task.wait(8)
        end
    end)
end

function disableAntiAFK()
    antiAFK = false
    antiAFKThread = nil
end

local noclipEnabled = false
local noclipConnection = RunService.Stepped:Connect(function()
    if not noclipEnabled then return end
    local character = LocalPlayer.Character
    if not character then return end
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = false
        end
    end
end)

function restoreCollision()
    local character = LocalPlayer.Character
    if not character then return end
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            part.CanCollide = true
        end
    end
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.5)
    if noclipEnabled then
        local character = LocalPlayer.Character
        if character then
            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end
    end
end)

local dragState = {
    active = false,
    input = nil,
    startPointer = nil,
    startScale = nil,
}

local function getGuiPoint(input)
    if input.UserInputType == Enum.UserInputType.Touch then
        return Vector2.new(input.Position.X, input.Position.Y)
    end
    return UserInputService:GetMouseLocation()
end

local cornerHandles = {}

    TL = Vector2.new(0, 0),
    TR = Vector2.new(1, 0),
    BL = Vector2.new(0, 1),
    BR = Vector2.new(1, 1),
}

for key, anchor in pairs(cornerCoordinates) do
    local handle = new("TextButton", {

        Size = UDim2.fromScale(0.085, 0.105),
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false,
        ZIndex = 15,
        Parent = menu,
    })
    cornerHandles[key] = handle
end

local function startMenuDrag(input)
    if modal.Visible then return end
    if dragState.active then return end
    local pointer = getGuiPoint(input)
    dragState.active = true
    dragState.input = input
    dragState.startPointer = pointer
    dragState.startScale = menu.Position
end

for _, handle in pairs(cornerHandles) do
    handle.InputBegan:Connect(function(input)
        local valid = input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
        if valid then
            startMenuDrag(input)
        end
    end)
end

UserInputService.InputChanged:Connect(function(input)
    if not dragState.active then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end

    local camera = workspace.CurrentCamera
    if not camera then return end

    local pointer = getGuiPoint(input)
    local deltaPixels = pointer - dragState.startPointer
    local viewport = camera.ViewportSize
    local dx = viewport.X > 0 and deltaPixels.X / viewport.X or 0
    local dy = viewport.Y > 0 and deltaPixels.Y / viewport.Y or 0

    local startPos = dragState.startScale
    menu.Position = UDim2.fromScale(startPos.X.Scale + dx, startPos.Y.Scale + dy)
end)

UserInputService.InputEnded:Connect(function(input)
    if not dragState.active then return end
    local sameTouch = dragState.input and input == dragState.input
    local mouseRelease = input.UserInputType == Enum.UserInputType.MouseButton1
    if sameTouch or mouseRelease then
        dragState.active = false
        dragState.input = nil
        dragState.startPointer = nil
        dragState.startScale = nil
    end
end)

local function keepMenuRecoverable()
    if not menu.Visible or dragState.active then return end
    local camera = workspace.CurrentCamera
    if not camera then return end

    local viewport = camera.ViewportSize
    local pos = menu.AbsolutePosition
    local size = menu.AbsoluteSize

    local fullyLeft = pos.X + size.X <= 0
    local fullyRight = pos.X >= viewport.X
    local fullyTop = pos.Y + size.Y <= 0
    local fullyBottom = pos.Y >= viewport.Y

    if not (fullyLeft or fullyRight or fullyTop or fullyBottom) then
        return
    end

    local x = menu.Position.X.Scale
    local y = menu.Position.Y.Scale

    if fullyLeft then
        x = math.clamp((size.X / 2) / math.max(viewport.X, 1), 0.01, 0.16)
    elseif fullyRight then
        x = 1 - math.clamp((size.X / 2) / math.max(viewport.X, 1), 0.01, 0.16)
    end

    if fullyTop then
        y = math.clamp((size.Y / 2) / math.max(viewport.Y, 1), 0.01, 0.16)
    elseif fullyBottom then
        y = 1 - math.clamp((size.Y / 2) / math.max(viewport.Y, 1), 0.01, 0.16)
    end

    menu.Position = UDim2.fromScale(x, y)
end

RunService.RenderStepped:Connect(keepMenuRecoverable)

local openerDrag = {
    active = false,
    moved = false,
    pointer = nil,
    startPointer = nil,
    startPosition = nil,
}

local function getOpenerPosition()
    return opener.Position.X.Scale, opener.Position.Y.Scale
end

opener.InputBegan:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    openerDrag.active = true
    openerDrag.moved = false
    openerDrag.pointer = input
    openerDrag.startPointer = getGuiPoint(input)
    openerDrag.startPosition = opener.Position
end)

UserInputService.InputChanged:Connect(function(input)
    if not openerDrag.active then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    local camera = workspace.CurrentCamera
    if not camera then return end

    local point = getGuiPoint(input)
    local delta = point - openerDrag.startPointer
    if delta.Magnitude > math.min(camera.ViewportSize.X, camera.ViewportSize.Y) * 0.006 then
        openerDrag.moved = true
    end

    local dx = camera.ViewportSize.X > 0 and delta.X / camera.ViewportSize.X or 0
    local dy = camera.ViewportSize.Y > 0 and delta.Y / camera.ViewportSize.Y or 0
    local start = openerDrag.startPosition
    opener.Position = UDim2.fromScale(start.X.Scale + dx, start.Y.Scale + dy)
end)

local function recoverOpener()
    local camera = workspace.CurrentCamera
    if not camera then return end
    local viewport = camera.ViewportSize
    local pos = opener.AbsolutePosition
    local size = opener.AbsoluteSize
    local fullyLeft = pos.X + size.X <= 0
    local fullyRight = pos.X >= viewport.X
    local fullyTop = pos.Y + size.Y <= 0
    local fullyBottom = pos.Y >= viewport.Y
    if not (fullyLeft or fullyRight or fullyTop or fullyBottom) then
        return
    end

    local x = opener.Position.X.Scale
    local y = opener.Position.Y.Scale
    local halfX = (size.X / math.max(viewport.X, 1)) / 2
    local halfY = (size.Y / math.max(viewport.Y, 1)) / 2
    if fullyLeft then x = math.clamp(halfX, 0.005, 0.08) end
    if fullyRight then x = 1 - math.clamp(halfX, 0.005, 0.08) end
    if fullyTop then y = math.clamp(halfY, 0.08, 0.18) end
    if fullyBottom then y = 1 - math.clamp(halfY, 0.08, 0.18) end
    opener.Position = UDim2.fromScale(x, y)
end

RunService.RenderStepped:Connect(recoverOpener)

UserInputService.InputEnded:Connect(function(input)
    if not openerDrag.active then return end
    local sameTouch = openerDrag.pointer and input == openerDrag.pointer
    local mouseRelease = input.UserInputType == Enum.UserInputType.MouseButton1
    if sameTouch or mouseRelease then
        local moved = openerDrag.moved
        openerDrag.active = false
        openerDrag.pointer = nil
        if not moved then
            menu.Visible = not menu.Visible
        end
    end
end)

local activeTab = "Main"

local function styleTab(name)
    for tabName, button in pairs(tabButtons) do
        local selected = tabName == name
        TweenService:Create(
            button,
            TweenInfo.new(0.13, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {BackgroundColor3 = selected and GRAY or PANEL_2}
        ):Play()
    end
end

local function showTab(name)
    activeTab = name
    for pageName, page in pairs(pageObjects) do
        page.Visible = false
    end
    pageObjects[name:gsub("%s+", "")].Visible = true
    styleTab(name)
    searchBox.Text = ""
    clearSearch.Visible = false
    refreshSearch()
end

for tabName, button in pairs(tabButtons) do
    button.Activated:Connect(function()
        showTab(tabName)
    end)
end

function refreshSearch()
    local query = string.lower(searchBox.Text or "")
    clearSearch.Visible = query ~= ""

    for _, entry in ipairs(allEntries) do
        if entry.Object and entry.Object.Parent then
            local belongsToActivePage = false
            local ancestor = entry.Object
            while ancestor and ancestor ~= pages do
                if ancestor == pageObjects[activeTab:gsub("%s+", "")] then
                    belongsToActivePage = true
                    break
                end
                ancestor = ancestor.Parent
            end
            if query == "" then
                entry.Object.Visible = true
            else
                entry.Object.Visible = belongsToActivePage and string.find(entry.Terms, query, 1, true) ~= nil
            end
        end
    end
end

searchBox:GetPropertyChangedSignal("Text"):Connect(refreshSearch)
clearSearch.Activated:Connect(function()
    searchBox.Text = ""
    searchBox:CaptureFocus()
end)
